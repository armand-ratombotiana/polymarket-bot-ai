    57→def _valid_token(authorization: str | None) -> bool:
    58→    if not settings.api_token:
    59→        return False
    60→    scheme, _, creds = (authorization or "").partition(" ")
    61→    if scheme.lower() != "bearer" or not creds:
    62→        return False
    63→    return hmac.compare_digest(creds, settings.api_token)

    66→# ── WebSocket Connection Manager ──────────────────────────────────────────────
    67→
    68→class ConnectionManager:
    69→    def __init__(self) -> None:
    70→        self.active: list[WebSocket] = []
    71→
    72→    async def connect(self, ws: WebSocket) -> None:
    73→        await ws.accept()
    74→        self.active.append(ws)
    75→        log.info("WS client connected — total %d", len(self.active))
    76→
    77→    def disconnect(self, ws: WebSocket) -> None:
    78→        if ws in self.active:
    79→            self.active.remove(ws)
    80→
    81→    async def broadcast(self, data: dict) -> None:
    82→        dead = []
    83→        payload = json.dumps(data, default=str)
    84→        for ws in list(self.active):
    85→            try:
    86→                await ws.send_text(payload)
    87→            except Exception:
    88→                dead.append(ws)
    89→        for ws in dead:
    90→            if ws in self.active:
    91→                self.active.remove(ws)

   100→async def _seed_markets(limit: int = 60) -> list[str]:
   101→    token_ids: list[str] = []
   102→    try:
   103→        markets = await gamma_client.get_markets(active=True, limit=limit, order="volume24hr")
   104→        for mkt in markets:
   105→            slug = mkt.get("slug") or mkt.get("groupItemTitle") or ""
   106→            ids = gamma_client.extract_token_ids(mkt)
   107→            for tid in ids:
   108→                token_ids.append(tid)
   109→                if slug:
   110→                    store.market_slugs[tid] = slug
   111→                # Ingest into vector store
   112→                vector_store.add_market(tid, mkt)
   113→
   114→        vector_store.build_index()
   115→        vector_store.save_to_disk()
   116→
   117→        if token_ids:
   118→            unique_ids = list(dict.fromkeys(token_ids))
   119→            log.info("Seeded %d unique tokens from %d Gamma markets", len(unique_ids), len(markets))
   120→            await store.log_event(f"📊 Seeded {len(unique_ids)} market tokens & built Vector Embeddings index")
   121→            return unique_ids
   122→    except Exception as e:
   123→        log.error("Market seeding failed: %s", e)
   124→        await store.log_event(f"⚠ Market seed failed: {e}")
   125→    return token_ids

   332→async def _build_snapshot() -> dict:
   333→    from core.safety import kill_switch_file_exists
   334→    books = []
   335→    async with store._lock:
   336→        for tid, book in store.order_books.items():
   337→            books.append({
   338→                "token_id": tid,
   339→                "slug": store.market_slugs.get(tid, tid[:14]),
   340→                "best_bid": book.best_bid,
   341→                "best_ask": book.best_ask,
   342→                "mid": book.mid,
   343→                "spread": book.spread,
   344→                "updated_at": book.updated_at,
   345→            })
   346→
   347→    orders = []
   348→    for o in store.open_orders.values():
   349→        orders.append({
   350→            "order_id": o.order_id,
   351→            "token_id": o.token_id,
   352→            "slug": store.market_slugs.get(o.token_id, o.token_id[:14]),
   353→            "side": o.side.value,
   354→            "price": o.price,
   355→            "size": o.size,
   356→            "size_matched": o.size_matched,
   357→            "strategy": o.strategy,
   358→            "paper": o.paper,
   359→            "created_at": o.created_at,
   360→        })
   361→
   362→    positions = []
   363→    for tid, pos in store.positions.items():
   364→        if pos.yes_shares > 0 or pos.total_invested > 0 or pos.realised_pnl != 0:
   365→            # Mark-to-live-mid unrealized P&L for this position.
   366→            # Falls back to cost basis (avg_entry_price) when no live quote is tracked.
   367→            _book = store.order_books.get(tid)
   368→            current_price = _book.mid if _book and _book.mid is not None else pos.avg_entry_price
   369→            _pos_unrealized = 0.0
   370→            if pos.yes_shares > 0:
   371→                _pos_unrealized += (current_price - pos.avg_entry_price) * pos.yes_shares
   372→            if pos.no_shares > 0:
   373→                _pos_unrealized += ((1.0 - current_price) - pos.avg_entry_price) * pos.no_shares
   374→            positions.append({
   375→                "token_id": tid,
   376→                "slug": store.market_slugs.get(tid, tid[:14]),
   377→                "yes_shares": pos.yes_shares,
   378→                "avg_entry_price": pos.avg_entry_price,
   379→                "total_invested": pos.total_invested,
   380→                "realised_pnl": pos.realised_pnl,
   381→                "current_price": round(current_price, 4),
   382→                "unrealized_pnl": round(_pos_unrealized, 4),
   383→            })
   384→
   385→    trades = []
   386→    for t in store.trades[-50:]:
   387→        trades.append({
   388→            "trade_id": t.trade_id,
   389→            "token_id": t.token_id,
   390→            "slug": store.market_slugs.get(t.token_id, t.token_id[:14]),
   391→            "side": t.side.value,
   392→            "price": t.price,
   393→            "size": t.size,
   394→            "pnl": t.pnl,
   395→            "strategy": t.strategy,
   396→            "paper": t.paper,
   397→            "timestamp": t.timestamp,
   398→        })
   399→
   400→    events = await store.get_recent_events(50)
   401→    active_strats = list(strategy_registry.get_active_instances().keys())
   402→
   403→    return {

   407→        "kill_switch": store.kill_switch_active,
   408→        "kill_switch_durable": bool(kill_switch_file_exists()),

   410→        "observation_reason": risk_manager.observation_reason,

   417→        "recent_trades": trades,
   418→        "events": events,
   419→        "ml": {
   420→            "model_ready": ml_model.rf is not None,
   421→            "brier_score": ml_model.brier_score,
   422→            "roc_auc": ml_model.roc_auc,

   458→@app.middleware("http")
   459→async def enforce_api_auth(request: Request, call_next):
   460→    """Fail-closed bearer-token auth on every route except the liveness probe and OPTIONS preflight."""
   461→    if request.method == "OPTIONS":
   462→        return await call_next(request)
   463→
   464→    origin = request.headers.get("origin")
   465→    cors_allowed = bool(origin and (origin in settings.cors_origin_list or "*" in settings.cors_origin_list or not settings.cors_origin_list))
   466→    cors_headers = {}
   467→    if origin and cors_allowed:
   468→        cors_headers["access-control-allow-origin"] = origin
   469→        cors_headers["access-control-allow-credentials"] = "true"
   470→        cors_headers["access-control-allow-headers"] = "Authorization, Content-Type"
   471→        cors_headers["access-control-allow-methods"] = "GET, POST, PUT, DELETE, OPTIONS"
   472→
   473→    path = request.url.path
   474→    if path in PUBLIC_PATHS or path.startswith("/api/health"):
   475→        response = await call_next(request)
   476→        if origin and cors_allowed:
   477→            response.headers["access-control-allow-origin"] = origin
   478→            response.headers["access-control-allow-credentials"] = "true"
   479→        return response
   480→
   481→    if not settings.api_token:
   482→        return JSONResponse(
   483→            status_code=503,
   484→            content={"detail": "API authentication not configured — set API_TOKEN in .env", "code": "AUTH_NOT_CONFIGURED"},
   485→            headers=cors_headers,
   486→        )
   487→    if not _valid_token(request.headers.get("authorization")):
   488→        return JSONResponse(
   489→            status_code=401,
   490→            content={"detail": "Unauthorized — missing or invalid API token"},
   491→            headers=cors_headers,
   492→        )
   493→
   494→    response = await call_next(request)
   495→    if origin and cors_allowed:
   496→        response.headers["access-control-allow-origin"] = origin
   497→        response.headers["access-control-allow-credentials"] = "true"
   498→    return response

   568→@app.get("/api/analytics", tags=["system"])
   569→async def get_analytics():
   570→    trades = store.trades
   571→    total_trades = len(trades)
   572→    # Only trades that actually closed a position (pnl != 0) determine win rate.
   573→    closed_trades = [t for t in trades if t.pnl != 0]
   574→    winning_trades = sum(1 for t in closed_trades if t.pnl > 0)
   575→    losing_trades = sum(1 for t in closed_trades if t.pnl < 0)
   576→    win_rate = (winning_trades / len(closed_trades)) if closed_trades else 0.0
   577→    total_vol = sum(t.price * t.size for t in trades)
   578→
   579→    # Wilson 95% confidence interval for the win rate (sample-size honest).
   580→    n = len(closed_trades)
   581→    z = 1.96
   582→    ci_low = ci_high = None
   583→    if n > 0:
   584→        p = win_rate
   585→        denom = 1 + z * z / n
   586→        centre = (p + z * z / (2 * n)) / denom
   587→        half = z * ((p * (1 - p) / n + z * z / (4 * n * n)) ** 0.5) / denom
   588→        ci_low = max(0.0, centre - half)
   589→        ci_high = min(1.0, centre + half)
   590→
   591→    wins = sum(t.pnl for t in closed_trades if t.pnl > 0)
   592→    losses = sum(t.pnl for t in closed_trades if t.pnl < 0)
   593→    profit_factor = wins / max(1e-9, -losses) if losses else (None if not wins else float("inf"))
   594→
   595→    # Average win / loss and per-trade expectancy.
   596→    # Expectancy = (win_rate * avg_win) + (loss_rate * avg_loss) where avg_loss is negative.
   597→    avg_win = (wins / winning_trades) if winning_trades else 0.0
   598→    avg_loss = (losses / losing_trades) if losing_trades else 0.0  # negative or zero
   599→    loss_rate = 1.0 - win_rate
   600→    expectancy = (win_rate * avg_win) + (loss_rate * avg_loss)
   601→    sharpe_ratio = ml_model.sharpe_ratio
   602→
   603→    # Unrealized P&L: mark open positions to the live order-book mid when present.
   604→    unrealized_pnl = 0.0
   605→    for p in store.positions.values():
   606→        if p.current_exposure <= 0.001:
   607→            continue
   608→        book = store.order_books.get(p.token_id)
   609→        mark = book.mid if book and book.mid is not None else None
   610→        if mark is None:
   611→            mark = p.avg_entry_price  # cost-basis mark (no live quote)
   612→        unrealized_pnl += (mark - p.avg_entry_price) * p.yes_shares
   613→        unrealized_pnl += ((1.0 - mark) - p.avg_entry_price) * p.no_shares
   614→
   615→    realized_pnl = store.daily_pnl
   616→    net_pnl = realized_pnl + unrealized_pnl
   617→    equity = store.paper_balance
   618→    max_drawdown_dollars = max(0.0, store.peak_equity - equity)
   619→    max_drawdown_pct = (max_drawdown_dollars / store.peak_equity) if store.peak_equity > 0 else 0.0
   620→
   621→    exp = compute_exposure()
   622→    deployable = float(MAX_DEPLOYABLE_CAPITAL)
   623→    risk_utilization = min(1.0, exp["maximum_remaining_loss"] / deployable) if deployable > 0 else 0.0
   624→
   625→    # Data freshness: seconds since the newest tracked order-book update.
   626→    books = store.order_books.values()
   627→    freshness = max((time.time() - b.updated_at for b in books), default=0.0)
   628→
   629→    return {
   630→        "equity": round(equity, 2),
   631→        "realized_pnl": round(realized_pnl, 2),
   632→        "unrealized_pnl": round(unrealized_pnl, 2),
   633→        "net_pnl": round(net_pnl, 2),
   634→        "total_trades": total_trades,
   635→        "winning_trades": winning_trades,
   636→        "losing_trades": losing_trades,
   637→        "closed_trades": len(closed_trades),
   638→        "open_trades": total_trades - len(closed_trades),
   639→        "win_rate": round(win_rate, 4),
   640→        "win_rate_ci_low": round(ci_low, 4) if ci_low is not None else None,
   641→        "win_rate_ci_high": round(ci_high, 4) if ci_high is not None else None,
   642→        "profit_factor": round(profit_factor, 2) if isinstance(profit_factor, float) else profit_factor,
   643→        "avg_win": round(avg_win, 4),
   644→        "avg_loss": round(avg_loss, 4),
   645→        "expectancy": round(expectancy, 4),
   646→        "sharpe_ratio": round(sharpe_ratio, 4) if sharpe_ratio is not None else None,
   647→        "max_drawdown_dollars": round(max_drawdown_dollars, 2),
   648→        "max_drawdown_pct": round(max_drawdown_pct, 4),
   649→        "total_volume_usdc": round(total_vol, 2),
   650→        "open_exposure": round(exp["maximum_remaining_loss"], 2),
   651→        "open_position_count": exp["open_position_count"],
   652→        "pending_order_capital": round(exp["reserved_for_pending_orders"], 2),
   653→        "risk_utilization": round(risk_utilization, 4),
   654→        "mode": settings.trading_mode,
   655→        "data_freshness_seconds": round(freshness, 1),
   656→        "peak_equity": round(store.peak_equity, 2),
   657→        "active_strategies": list(strategy_registry.get_active_instances().keys()),
   658→    }

   718→
   719→@app.get("/api/ai/predict/{token_id}", tags=["ai"])
   720→async def get_ai_prediction(token_id: str):
   721→    """
   722→    Return the ML ensemble's directional view for a single YES token.
   723→
   724→    Reads market metadata from `market_discovery.catalog` (the universal
   725→    catalog maintained by `core/market_discovery.py`) — NOT from
   726→    `store.market_info` (that attribute does not exist on the DataStore).
   727→
   728→    Response:
   729→      * p_yes            — calibrated ensemble P(YES) ∈ (0.01, 0.99)
   730→      * confidence       — |p_yes - 0.5| * 2  ∈ [0, 1]
   731→      * market_mid       — live book mid price (or None if book is empty)
   732→      * edge             — p_yes - market_mid (positive ⇒ model is more
   733→                           bullish than the market; negative ⇒ bearish)
   734→      * edge_bps         — edge expressed in basis points
   735→      * recommended_action — BUY | SELL | HOLD (threshold-gated)
   736→      * thresholds       — the gates used so callers can audit the decision
   737→      * market           — the catalog metadata used to build the feature vector
   738→      * model_status     — whether the ensemble was actually trained
   739→
   740→    A HOLD recommendation does NOT imply no edge — it means the edge was
   741→    below the configured conviction threshold. Always inspect `edge` /
   742→    `edge_bps` before deciding to act.
   743→    """
   744→    # ── R10: pull metadata from market_discovery.catalog (NOT store.market_info) ──
   745→    from core.market_discovery import market_discovery
   746→
   747→    catalog_record = market_discovery.catalog.get(token_id)
   748→    if not catalog_record:
   749→        raise HTTPException(
   750→            status_code=404,
   751→            detail=f"token_id '{token_id}' not present in market_discovery.catalog — "
   752→                   "wait for the next catalog sync or call /api/markets/coverage",
   753→        )
   754→
   755→    # extract_features expects a dict with Gamma-style keys (volume24hr, volume,
   756→    # liquidity|liquidityNum, endDate|end_date_iso|endDateIso). The catalog
   757→    # record uses normalized snake_case keys, so bridge both shapes here.
   758→    market_for_features = dict(catalog_record)
   759→    market_for_features.setdefault(
   760→        "volume24hr", catalog_record.get("volume_24h", 0.0)
   761→    )
   762→    market_for_features.setdefault(
   763→        "volume", catalog_record.get("total_volume", 0.0)
   764→    )
   765→    # liquidity key already matches; bridge endDate aliases just in case.
   766→    if "endDate" not in market_for_features and catalog_record.get("end_date"):
   767→        market_for_features["endDate"] = catalog_record["end_date"]
   768→
   769→    # ── Live order book (needed for the microstructure feature vector) ──
   770→    book = await store.get_order_book(token_id)
   771→    if book is None:
   772→        # Hint the poller to prioritize this token, then surface a 502.
   773→        book_poller.prioritize_tokens([token_id])
   774→        raise HTTPException(
   775→            status_code=502,
   776→            detail=f"no live order book for token '{token_id}' — poller prioritized; retry shortly",
   777→        )
   778→
   779→    market_mid = book.mid
   780→    best_bid = book.best_bid
   781→    best_ask = book.best_ask
   782→
   783→    # ── Feature extraction + ensemble prediction ──
   784→    from ml.features import extract_features
   785→
   786→    features = extract_features(market_for_features, book)
   787→    if features is None:
   788→        raise HTTPException(
   789→            status_code=422,
   790→            detail=(
   791→                "feature extraction returned None — book mid is missing or "
   792→                "outside the tradeable (0.001, 0.999) band; cannot score"
   793→            ),
   794→        )
   795→
   796→    p_yes, confidence = ml_model.predict(features, token_id=token_id)
   797→    p_yes = float(p_yes)
   798→    confidence = float(confidence)
   799→
   800→    # ── Edge vs market mid ──
   801→    if market_mid is not None:
   802→        edge = float(p_yes) - float(market_mid)
   803→        edge_bps = round(edge * 10_000.0, 1)
   804→    else:
   805→        edge = None
   806→        edge_bps = None
   807→
   808→    # ── Recommendation thresholds ──
   809→    # Edge-based conviction gate (mirrors the spirit of strategies/signal_trader.py
   810→    # but expressed in terms of the model-vs-market edge rather than the absolute
   811→    # p_yes level, so a model that sees a 50/50 coin priced at 0.20 correctly
   812→    # surfaces a BUY rather than a HOLD just because |p_yes − 0.5| is small).
   813→    #
   814→    #   BUY  : edge ≥ +2ct (200 bps) AND confidence ≥ 0.10
   815→    #   SELL : edge ≤ −2ct (−200 bps) AND confidence ≥ 0.10
   816→    #   HOLD : otherwise (insufficient edge OR insufficient model conviction)
   817→    MIN_EDGE_CT = 0.02          # 2 cents
   818→    MIN_CONFIDENCE = 0.10
   819→
   820→    recommended_action = "HOLD"
   821→    if edge is None:
   822→        action_reason = "market mid unavailable — cannot compute edge"
   823→    elif confidence < MIN_CONFIDENCE:
   824→        recommended_action = "HOLD"
   825→        action_reason = (
   826→            f"confidence {confidence:.3f} < {MIN_CONFIDENCE:.2f} — model is too "
   827→            f"uncertain to act despite edge={edge*100:+.2f}ct"
   828→        )
   829→    elif edge >= MIN_EDGE_CT:
   830→        recommended_action = "BUY"
   831→        action_reason = (
   832→            f"edge=+{edge*100:.2f}ct ≥ +{MIN_EDGE_CT*100:.0f}ct AND "
   833→            f"confidence={confidence:.3f} ≥ {MIN_CONFIDENCE:.2f}"
   834→        )
   835→    elif edge <= -MIN_EDGE_CT:
   836→        recommended_action = "SELL"
   837→        action_reason = (
   838→            f"edge={edge*100:.2f}ct ≤ −{MIN_EDGE_CT*100:.0f}ct AND "
   839→            f"confidence={confidence:.3f} ≥ {MIN_CONFIDENCE:.2f}"
   840→        )
   841→    else:
   842→        recommended_action = "HOLD"
   843→        action_reason = (
   844→            f"|edge|={abs(edge)*100:.2f}ct below ±{MIN_EDGE_CT*100:.0f}ct "
   845→            f"conviction threshold"
   846→        )
   847→
   848→    # Compact market payload (avoid duplicating the full catalog record)
   849→    market_payload = {
   850→        "token_id": catalog_record.get("token_id"),
   851→        "event_id": catalog_record.get("event_id"),
   852→        "event_title": catalog_record.get("event_title"),
   853→        "question": catalog_record.get("question"),
   854→        "slug": catalog_record.get("slug") or store.market_slugs.get(token_id, ""),
   855→        "outcome": catalog_record.get("outcome"),
   856→        "category": catalog_record.get("category"),
   857→        "end_date": catalog_record.get("end_date"),
   858→        "status": catalog_record.get("status"),
   859→        "volume_24h": catalog_record.get("volume_24h"),
   860→        "total_volume": catalog_record.get("total_volume"),
   861→        "liquidity": catalog_record.get("liquidity"),
   862→        "last_synced": catalog_record.get("last_synced"),
   863→    }
   864→
   865→    return {
   866→        "token_id": token_id,
   867→        "p_yes": round(p_yes, 4),
   868→        "confidence": round(confidence, 4),
   869→        "market_mid": round(market_mid, 4) if market_mid is not None else None,
   870→        "best_bid": round(best_bid, 4) if best_bid is not None else None,
   871→        "best_ask": round(best_ask, 4) if best_ask is not None else None,
   872→        "spread": round(book.spread, 4) if book.spread is not None else None,
   873→        "edge": round(edge, 4) if edge is not None else None,
   874→        "edge_bps": edge_bps,
   875→        "recommended_action": recommended_action,
   876→        "action_reason": action_reason,
   877→        "thresholds": {
   878→            "min_edge_cents": MIN_EDGE_CT * 100.0,
   879→            "min_confidence": MIN_CONFIDENCE,
   880→        },
   881→        "model_status": {
   882→            "model_ready": ml_model.rf is not None,
   883→            "model_version": model_registry.active_version,
   884→            "brier_score": ml_model.brier_score,
   885→            "roc_auc": ml_model.roc_auc,
   886→            "ece": ml_model.ece,
   887→            "n_online_updates": ml_model._n_updates,
   888→        },
   889→        "market": market_payload,
   890→        "book_updated_at": book.updated_at,
   891→        "timestamp": time.time(),
   892→    }

   897→@app.get("/api/history/ohlcv/{token_id}", tags=["markets"])
   898→async def get_market_ohlcv(token_id: str, resolution: str = "5m", count: int = 40):
   899→    """Return OHLCV candlestick bars for visual charting.
   900→
   901→    Priority:
   902→      1. Real candles from TimescaleDB continuous aggregates (market.price_candle_*)
   903→         when TimescaleDB is connected and rows exist for this token — labeled synthetic=False.
   904→      2. Seeded random-walk anchored to live mid when no stored candles exist —
   905→         explicitly labeled synthetic=True so callers always know the data source.
   906→    """
   907→    book = await store.get_order_book(token_id)
   908→    mid = (book.mid if book else 0.5) or 0.5
   909→    slug = store.market_slugs.get(token_id, token_id[:14])
   910→    step_sec = 60 if resolution == "1m" else 300 if resolution == "5m" else 3600
   911→    view_map = {"1m": "market.price_candle_1m", "5m": "market.price_candle_5m", "1h": "market.price_candle_1h"}
   912→    pg_view = view_map.get(resolution, "market.price_candle_5m")
   913→
   914→    # ── Attempt real TimescaleDB continuous-aggregate candles ─────────────────
   915→    from core.timescale_db import timescale_db
   916→    if timescale_db._is_postgres and timescale_db._pool:
   917→        try:
   918→            async with timescale_db._pool.acquire() as conn:
   919→                rows = await conn.fetch(
   920→                    f"""
   921→                    SELECT bucket, open, high, low, close, vwap, tick_count
   922→                    FROM {pg_view}
   923→                    WHERE token_id = $1
   924→                    ORDER BY bucket DESC
   925→                    LIMIT $2;
   926→                    """,
   927→                    token_id,
   928→                    count,
   929→                )
   930→            if rows:
   931→                bars = [
   932→                    {
   933→                        "timestamp": float(r["bucket"].timestamp()),
   934→                        "open": round(float(r["open"]), 4),
   935→                        "high": round(float(r["high"]), 4),
   936→                        "low": round(float(r["low"]), 4),
   937→                        "close": round(float(r["close"]), 4),
   938→                        "volume": float(r["tick_count"]),
   939→                        "vwap": round(float(r["vwap"]), 4) if r["vwap"] is not None else None,
   940→                    }
   941→                    for r in reversed(rows)
   942→                ]
   943→                return {
   944→                    "token_id": token_id,
   945→                    "slug": slug,
   946→                    "resolution": resolution,
   947→                    "bars": bars,
   948→                    "count": len(bars),
   949→                    "synthetic": False,
   950→                    "source": pg_view,
   951→                    "data_age_seconds": round(time.time() - bars[-1]["timestamp"], 1) if bars else None,
   952→                }
   953→        except Exception as e:
   954→            log.debug("[ohlcv] TimescaleDB candle query failed for %s: %s", token_id, e)
   955→
   956→    # ── Synthetic fallback: seeded random-walk anchored to live mid ───────────
   957→    rng = np.random.RandomState(abs(hash(token_id + resolution)) % (2**31))
   958→    now = time.time()
   959→
   960→    bars = []
   961→    curr_price = max(mid * (1.0 + rng.uniform(-0.06, 0.06)), 0.05)
   962→    for i in range(count):
   963→        ts = now - (count - i) * step_sec
   964→        drift = rng.uniform(-0.012, 0.012)
   965→        open_p = curr_price
   966→        close_p = max(min(open_p + drift, 0.98), 0.02)
   967→        high_p = max(open_p, close_p) + abs(rng.uniform(0.001, 0.008))
   968→        low_p = min(open_p, close_p) - abs(rng.uniform(0.001, 0.008))
   969→        vol = float(rng.uniform(500, 15000))
   970→        bars.append({
   971→            "timestamp": ts,
   972→            "open": round(open_p, 4),
   973→            "high": round(high_p, 4),
   974→            "low": round(low_p, 4),
   975→            "close": round(close_p, 4),
   976→            "volume": round(vol, 1),
   977→        })
   978→        curr_price = close_p
   979→
   980→    if bars:
   981→        bars[-1]["close"] = round(mid, 4)
   982→
   983→    return {
   984→        "token_id": token_id,
   985→        "slug": slug,
   986→        "resolution": resolution,
   987→        "bars": bars,
   988→        "count": len(bars),
   989→        "synthetic": True,
   990→        "synthetic_kind": "seeded_random_walk",
   991→        "disclaimer": "Synthetic bars anchored to live mid — no stored history for this token yet",
   992→    }

  1230→class PositionCloseRequest(BaseModel):

  1232→

  1236→    of a position incrementally, or set `dry_run=true` to preview the
  1237→    marketable close without submitting an order.

  1240→    dry_run: bool = False

  1242→
  1243→@app.post("/api/positions/{token_id}/close", tags=["trading"])
  1244→async def close_position(token_id: str, req: PositionCloseRequest):
  1245→    """
  1246→    One-click marketable close of an open position.
  1247→
  1248→    Long YES positions are closed by submitting a SELL order at the current
  1249→    `best_bid` (a marketable limit — any resting bid ≥ best_bid is matched
  1250→    immediately). Long NO positions are closed by submitting a BUY order at
  1251→    the current `best_ask` (symmetric — covers the synthetic short).
  1252→
  1253→    Risk checks (`risk_manager.check_order`) are applied exactly as for a
  1254→    manual `/api/trade`, and the order flows through `paper_sim` when
  1255→    `settings.paper_trade` is true, or `clob_client` otherwise.
  1256→
  1257→    Use `dry_run=true` to preview the fill price, share count, and
  1258→    estimated realised P&L without submitting an order.
  1259→    """
  1260→    async with store._lock:
  1261→        pos = store.positions.get(token_id)
  1262→        # Snapshot the values we need before releasing the lock so we don't
  1263→        # hold store._lock across the (potentially blocking) order placement.
  1264→        if pos is None:
  1265→            long_yes = 0.0
  1266→            long_no = 0.0
  1267→            avg_entry_price = 0.0
  1268→            total_invested = 0.0
  1269→            realised_pnl = 0.0
  1270→            strategy = ""
  1271→        else:
  1272→            long_yes = float(pos.yes_shares)
  1273→            long_no = float(pos.no_shares)
  1274→            avg_entry_price = float(pos.avg_entry_price)
  1275→            total_invested = float(pos.total_invested)
  1276→            realised_pnl = float(pos.realised_pnl)
  1277→            strategy = pos.strategy or ""
  1278→
  1279→    if long_yes <= 0.0 and long_no <= 0.0:
  1280→        raise HTTPException(
  1281→            status_code=404,
  1282→            detail=f"no open position for token '{token_id}' — nothing to close",
  1283→        )
  1284→
  1285→    # ── Determine side, price, and size from the live book ──
  1286→    book = await store.get_order_book(token_id)
  1287→    if book is None or (book.best_bid is None and book.best_ask is None):
  1288→        # Treat a missing or fully empty book the same — surface 502 and hint
  1289→        # the poller to reprioritize this token so the next call succeeds.
  1290→        book_poller.prioritize_tokens([token_id])
  1291→        raise HTTPException(
  1292→            status_code=502,
  1293→            detail=f"no live order book for token '{token_id}' — poller prioritized; retry shortly",
  1294→        )
  1295→
  1296→    if long_yes > 0.0:
  1297→        # Close a long YES position: SELL into the bid ladder at best_bid.
  1298→        side = Side.SELL
  1299→        if book.best_bid is None:
  1300→            raise HTTPException(
  1301→                status_code=502,
  1302→                detail="cannot close long YES position — best_bid is empty (no resting bids)",
  1303→            )
  1304→        close_price = float(book.best_bid)
  1305→        available_shares = long_yes
  1306→    else:
  1307→        # Close a long NO position: BUY at best_ask (covers the synthetic short).
  1308→        side = Side.BUY
  1309→        if book.best_ask is None:
  1310→            raise HTTPException(
  1311→                status_code=502,
  1312→                detail="cannot close long NO position — best_ask is empty (no resting asks)",
  1313→            )
  1314→        close_price = float(book.best_ask)
  1315→        available_shares = long_no
  1316→
  1317→    # Cap the close size if the caller requested a partial scale-out.
  1318→    if req.max_size_shares is not None:
  1319→        size_shares = min(float(req.max_size_shares), available_shares)
  1320→    else:
  1321→        size_shares = available_shares
  1322→
  1323→    if size_shares <= 0.0:
  1324→        raise HTTPException(
  1325→            status_code=400,
  1326→            detail="requested close size resolves to 0 shares — nothing to close",
  1327→        )
  1328→
  1329→    # ── Estimated P&L preview (uses cost basis for long YES positions) ──
  1330→    estimated_pnl = 0.0
  1331→    if side == Side.SELL and avg_entry_price > 0.0:
  1332→        # P&L = (sell_price - avg_entry) * shares  — for YES side only.
  1333→        estimated_pnl = (close_price - avg_entry_price) * size_shares
  1334→    # (NO side P&L requires the YES/NO parity identity and is computed at fill
  1335→    #  time by store.record_fill / paper_sim._execute_fill; we don't fabricate
  1336→    #  a number here.)
  1337→
  1338→    notional_usdc = close_price * size_shares
  1339→    slug = store.market_slugs.get(token_id, token_id[:12])
  1340→
  1341→    if req.dry_run:
  1342→        return {
  1343→            "status": "dry_run",
  1344→            "token_id": token_id,
  1345→            "slug": slug,
  1346→            "side": side.value,
  1347→            "price": round(close_price, 4),
  1348→            "size_shares": round(size_shares, 4),
  1349→            "notional_usdc": round(notional_usdc, 2),
  1350→            "estimated_pnl": round(estimated_pnl, 2),
  1351→            "best_bid": round(float(book.best_bid), 4) if book.best_bid is not None else None,
  1352→            "best_ask": round(float(book.best_ask), 4) if book.best_ask is not None else None,
  1353→            "book_updated_at": book.updated_at,
  1354→            "paper_trade": settings.paper_trade,
  1355→            "remaining_position": {
  1356→                "yes_shares": max(0.0, long_yes - (size_shares if side == Side.SELL else 0.0)),
  1357→                "no_shares": max(0.0, long_no - (size_shares if side == Side.BUY else 0.0)),
  1358→                "avg_entry_price": avg_entry_price,
  1359→                "total_invested_before": total_invested,
  1360→                "realised_pnl_before": realised_pnl,
  1361→            },
  1362→            "note": "dry_run=true — no order submitted",
  1363→        }
  1364→
  1365→    # ── Risk gate (same path as /api/trade) ──
  1366→    provisional = Order(
  1367→        order_id="close-pre-check",
  1368→        token_id=token_id,
  1369→        side=side,
  1370→        price=close_price,
  1371→        size=size_shares,
  1372→        strategy=strategy or "manual_close",
  1373→        paper=settings.paper_trade,
  1374→    )
  1375→    allowed, reason = await risk_manager.check_order(provisional)
  1376→    if not allowed:
  1377→        await store.log_event(f"⚠ Risk block [close]: {reason}")
  1378→        raise HTTPException(status_code=400, detail=f"Risk rejection: {reason}")
  1379→
  1380→    # ── Submit the marketable close order ──
  1381→    args = OrderArgs(
  1382→        token_id=token_id,
  1383→        price=close_price,
  1384→        side=side,
  1385→        size=size_shares,
  1386→        # FOK = fill-or-kill: a marketable close should either fill completely
  1387→        # at the quoted top-of-book price or be rejected (no partial leaves).
  1388→        order_type="FOK",
  1389→    )
  1390→
  1391→    if settings.paper_trade:
  1392→        order = await paper_sim.create_order(args, strategy=strategy or "manual_close")
  1393→    else:
  1394→        from core.clob_client import clob_client
  1395→        order = await clob_client.create_order(args)
  1396→        if order:
  1397→            order_obj = Order(
  1398→                order_id=order.get("orderID") or order.get("order_id", f"close-{token_id[:8]}"),
  1399→                token_id=token_id,
  1400→                side=side,
  1401→                price=close_price,
  1402→                size=size_shares,
  1403→                strategy=strategy or "manual_close",
  1404→                paper=False,
  1405→            )
  1406→            await store.add_order(order_obj)
  1407→            order = order_obj
  1408→
  1409→    if not order:
  1410→        raise HTTPException(
  1411→            status_code=400,
  1412→            detail="failed to submit close order — exchange rejected or paper_sim error",
  1413→        )
  1414→
  1415→    # ── Audit trail entry ──
  1416→    try:
  1417→        await audit_logger.log_event(
  1418→            category="trading",
  1419→            event_type="position_close",
  1420→            details=(
  1421→                f" Marketable close: {side.value} {size_shares:.4f} @ {close_price:.4f} "
  1422→                f"(${notional_usdc:.2f}) [est P&L ${estimated_pnl:+.2f}]"
  1423→            ),
  1424→            token_id=token_id,
  1425→            slug=slug,
  1426→            pnl=estimated_pnl,
  1427→            strategy=strategy or "manual_close",
  1428→        )
  1429→    except Exception as e:
  1430→        log.debug("[positions/close] audit log write failed: %s", e)
  1431→
  1432→    await store.log_event(
  1433→        f"🚪 Position close: {side.value} {slug} {size_shares:.2f} @ {close_price:.4f} "
  1434→        f"(${notional_usdc:.2f}) [est P&L ${estimated_pnl:+.2f}]"
  1435→    )
  1436→
  1437→    return {
  1438→        "status": "submitted",
  1439→        "token_id": token_id,
  1440→        "slug": slug,
  1441→        "order_id": getattr(order, "order_id", None),
  1442→        "side": side.value,
  1443→        "price": round(close_price, 4),
  1444→        "size_shares": round(size_shares, 4),
  1445→        "notional_usdc": round(notional_usdc, 2),
  1446→        "estimated_pnl": round(estimated_pnl, 2),
  1447→        "best_bid": round(float(book.best_bid), 4) if book.best_bid is not None else None,
  1448→        "best_ask": round(float(book.best_ask), 4) if book.best_ask is not None else None,
  1449→        "book_updated_at": book.updated_at,
  1450→        "paper_trade": settings.paper_trade,
  1451→        "remaining_position": {

  1635→@app.post("/api/ml/learn", tags=["ml"])

  1637→    """Feed a resolved ground-truth outcome into the online SGD learner.
  1638→
  1639→    1. Backfills outcome labels in both DB backends (TimescaleDB + SQLite).
  1640→    2. Fetches the most recent stored feature vector for this token.
  1641→    3. Calls ml_model.update() to incrementally train the SGD online learner.
  1642→    """
  1643→    from core.timescale_db import timescale_db
  1644→
  1645→    # Step 1: persist outcome label
  1646→    updated = timescale_db.mark_resolved_outcomes(token_id, resolved_yes=resolved_yes)
  1647→

  1649→    import json as _json
  1650→    import sqlite3
  1651→
  1652→    import numpy as np
  1653→    features = None

  1655→        with sqlite3.connect(timescale_db._sqlite_path) as conn:
  1656→            row = conn.execute(
  1657→                "SELECT features_json FROM ml_feature_store WHERE token_id = ? ORDER BY timestamp DESC LIMIT 1;",
  1658→                (token_id,),

  1661→            features = np.array(_json.loads(row[0]), dtype=np.float32)
  1662→    except Exception as e:
  1663→        log.warning("[api/ml/learn] Could not fetch feature vector for %s: %s", token_id, e)

  1670→            f"(update #{ml_model._n_updates})"

  1675→        )
  1676→
  1677→    return {
  1678→        "status": "updated",
  1679→        "token_id": token_id,
  1680→        "resolved_yes": resolved_yes,

  1799→@app.post("/api/arbitrage/execute", tags=["arbitrage"])
  1800→async def execute_arbitrage(req: ArbitrageExecuteRequest):
  1801→    """
  1802→    Execute a dual-leg Dutch-book arbitrage. Both legs pass the same risk gate
  1803→    and are hard-capped by the per-market ceiling. Live execution is only
  1804→    possible for real token ids; synthetic complementary legs are reported
  1805→    but not transmitted to the exchange.
  1806→    """
  1807→    from core.clob_client import clob_client
  1808→    from risk.manager import MAX_POSITION_PER_MARKET
  1809→
  1810→    size_usdc = min(float(req.size_usdc), float(MAX_POSITION_PER_MARKET))
  1811→    if size_usdc <= 0:
  1812→        raise HTTPException(status_code=400, detail="size_usdc must be positive")
  1813→
  1814→    results = []
  1815→    legs = [
  1816→        ("yes", req.token_id_yes),
  1817→        ("no", req.token_id_no),
  1818→    ]
  1819→    for leg, token_id in legs:
  1820→        book = store.order_books.get(token_id)
  1821→        price = book.best_ask if book and book.best_ask else 0.50
  1822→        shares = size_usdc / max(price, 0.01)
  1823→
  1824→        provisional = Order(
  1825→            order_id=f"arb-{leg}-pre",
  1826→            token_id=token_id,
  1827→            side=Side.BUY,
  1828→            price=price,
  1829→            size=shares,
  1830→            strategy="arb_scanner",
  1831→            paper=settings.paper_trade,
  1832→        )
  1833→        allowed, reason = await risk_manager.check_order(provisional)
  1834→        if not allowed:
  1835→            results.append({"leg": leg, "token_id": token_id, "status": "REJECTED", "reason": reason})
  1836→            continue
  1837→
  1838→        if settings.paper_trade:
  1839→            order = await paper_sim.create_order(
  1840→                OrderArgs(token_id=token_id, price=price, side=Side.BUY, size=shares),
  1841→                strategy="arb_scanner",
  1842→            )
  1843→            status = "PLACED_PAPER"
  1844→        else:
  1845→            if token_id.endswith("_no"):
  1846→                results.append({"leg": leg, "token_id": token_id, "status": "SKIPPED", "reason": "synthetic complementary token — not transmissible"})
  1847→                continue
  1848→            resp = await clob_client.create_order(
  1849→                OrderArgs(token_id=token_id, price=price, side=Side.BUY, size=shares)
  1850→            )
  1851→            if resp is None:
  1852→                results.append({"leg": leg, "token_id": token_id, "status": "FAILED", "reason": "exchange rejected order"})
  1853→                continue
  1854→            order = Order(
  1855→                order_id=resp.get("orderID") or resp.get("order_id", "unknown"),
  1856→                token_id=token_id,
  1857→                side=Side.BUY,
  1858→                price=price,
  1859→                size=shares,
  1860→                strategy="arb_scanner",
  1861→                paper=False,
  1862→            )
  1863→            await store.add_order(order)
  1864→            status = "PLACED_LIVE"
  1865→
  1866→        results.append({"leg": leg, "token_id": token_id, "status": status, "order_id": order.order_id})
  1867→
  1868→    slug = store.market_slugs.get(req.token_id_yes, req.token_id_yes[:12])
  1869→    await store.log_event(f"⚡ Manual arb execute: {slug} (${size_usdc:.2f}/leg): {[r['status'] for r in results]}")
  1870→    return {"status": "processed", "size_usdc": size_usdc, "legs": results, "slug": slug}
