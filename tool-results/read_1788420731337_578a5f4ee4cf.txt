     1→"""
     2→core/live_safety_gate.py — God Mode §82 Live Trading Safety Gate.
     3→
     4→Ten-check staged validation that MUST pass before live trading is enabled.
     5→Every check is fail-closed: an exception inside a check is recorded as a
     6→failed check (with the exception text in ``detail``) rather than crashing
     7→the gate — the gate's contract is to always return a verdict, never raise.
     8→
     9→Checks (in staged order — each must pass before live trading is authorised):
    10→
    11→  ┌────┬──────────────────────────────┬───────────────────────────────────────────┐
    12→  │ #  │ check id                      │ pass condition                           │
    13→  ├────┼──────────────────────────────┼───────────────────────────────────────────┤
    14→  │ 1  │ paper_mode_24h                │ trading_mode == "paper" AND              │
    15→  │    │                               │ (now - session_start) >= 24h             │
    16→  │ 2  │ positive_expectancy          │ closed_positions avg_pnl > 0 (count > 0) │
    17→  │ 3  │ max_drawdown_under_2usd       │ current drawdown < $2.00                 │
    18→  │ 4  │ win_rate_over_50pct           │ closed_positions win_rate > 0.50         │
    19→  │ 5  │ min_20_closed_trades          │ closed_positions count >= 20             │
    20→  │ 6  │ ml_trained_on_real_data       │ ml_model fitted AND training_source      │
    21→  │    │                               │ contains "real" AND n_real_samples > 0   │
    22→  │ 7  │ drift_healthy                 │ drift_detector.drift_status == HEALTHY  │
    23→  │ 8  │ kill_switch_tested            │ audit trail has ≥1 kill_switch_activated │
    24→  │    │                               │ AND ≥1 kill_switch_deactivated event,    │
    25→  │    │                               │ OR durable marker file present            │
    26→  │ 9  │ risk_limits_verified          │ risk_manager.status_report() shows all   │
    27→  │    │                               │ limits healthy (kill switch clear,        │
    28→  │    │                               │ exposure reconciled, within all caps)    │
    29→  │ 10 │ api_credentials_configured    │ settings.has_credentials AND             │
    30→  │    │                               │ settings.has_api_keys                    │
    31→  └────┴──────────────────────────────┴───────────────────────────────────────────┘
    32→
    33→The HTTP layer (``api/server.py``) calls ``register_routes(app)`` at startup
    34→to expose:
    35→
    36→  GET  /api/live/readiness   — run all 10 checks, return {passed, checks, ...}
    37→  POST /api/live/enable      — attempt to flip live mode on (only succeeds if
    38→                               all 10 checks pass; requires confirm=true)
    39→"""

   107→async def check_live_readiness() -> dict[str, Any]:
   108→    """
   109→    Run all 10 God Mode §82 staged checks and return a verdict dict::
   110→
   111→        {
   112→            "passed": bool,                  # True only if ALL 10 checks pass
   113→            "checks": [ {id, name, passed, severity, threshold, value, detail}, ... ],
   114→            "passed_count": int,
   115→            "total_count": int,
   116→            "blocking_checks": [str, ...],   # ids of failed checks (empty if passed)
   117→            "checked_at": float,             # epoch seconds
   118→        }
   119→
   120→    Never raises — a check that throws records itself as failed (with the
   121→    exception text in ``detail``) so the gate always returns a verdict. This
   122→    is the live-trading safety contract: the gate must answer, even when a
   123→    dependency is broken.
   124→    """
   125→    checks: list[dict[str, Any]] = []
   126→    checks.append(await _check_paper_mode())
   127→    checks.append(await _check_positive_expectancy())
   128→    checks.append(await _check_max_drawdown())
   129→    checks.append(await _check_win_rate())
   130→    checks.append(await _check_closed_trades())
   131→    checks.append(await _check_ml_real_data())
   132→    checks.append(await _check_drift_healthy())
   133→    checks.append(await _check_kill_switch_tested())
   134→    checks.append(await _check_risk_limits())
   135→    checks.append(await _check_api_credentials())
   136→
   137→    passed_count = sum(1 for c in checks if c.get("passed"))
   138→    blocking = [c["id"] for c in checks if not c.get("passed")]
   139→    return {
   140→        "passed": len(blocking) == 0,
   141→        "checks": checks,
   142→        "passed_count": passed_count,
   143→        "total_count": len(checks),
   144→        "blocking_checks": blocking,
   145→        "checked_at": time.time(),
   146→    }

   149→async def get_live_safety_report() -> dict[str, Any]:
   150→    """
   151→    Richer report wrapping ``check_live_readiness`` with mode context,
   152→    thresholds, and operator guidance. The shape::
   153→
   154→        {
   155→            "generated_at": float,
   156→            "gate": "God Mode §82 — Live Trading Safety Gate",
   157→            "readiness": { … check_live_readiness output … },
   158→            "mode_context": {
   159→                "trading_mode": str,
   160→                "paper_trade": bool,
   161→                "live_trading_enabled": bool,
   162→                "has_credentials": bool,
   163→                "has_api_keys": bool,
   164→                "kill_switch_active": bool,
   165→                "kill_switch_durable": bool,
   166→            },
   167→            "thresholds": { … §82 constants … },
   168→            "guidance": str,
   169→        }
   170→
   171→    Designed as the single payload an operator dashboard polls to render the
   172→    full "are we ready to go live?" posture.
   173→    """
   174→    readiness = await check_live_readiness()
   175→
   176→    mode_context: dict[str, Any]
   177→    try:
   178→        from config import settings
   179→        from core.data_store import store
   180→        from core.safety import kill_switch_file_exists
   181→
   182→        mode_context = {
   183→            "trading_mode": settings.trading_mode,
   184→            "paper_trade": bool(settings.paper_trade),
   185→            "live_trading_enabled": bool(settings.live_trading_enabled),
   186→            "has_credentials": bool(settings.has_credentials),
   187→            "has_api_keys": bool(settings.has_api_keys),
   188→            "kill_switch_active": bool(getattr(store, "kill_switch_active", False)),
   189→            "kill_switch_durable": bool(kill_switch_file_exists()),
   190→        }
   191→    except Exception as e:
   192→        log.error("[live_safety_gate] mode context gathering failed: %s", e)
   193→        mode_context = {"error": f"mode context unavailable: {e}"}
   194→
   195→    return {
   196→        "generated_at": time.time(),
   197→        "gate": "God Mode §82 — Live Trading Safety Gate",
   198→        "readiness": readiness,
   199→        "mode_context": mode_context,
   200→        "thresholds": {
   201→            "paper_mode_min_seconds": PAPER_MODE_MIN_SECONDS,
   202→            "min_closed_trades": MIN_CLOSED_TRADES,
   203→            "min_win_rate": MIN_WIN_RATE,
   204→            "max_live_drawdown_usd": MAX_LIVE_DRAWDOWN_USD,
   205→            "drift_healthy_status": DRIFT_HEALTHY_STATUS,
   206→            "check_order": list(CHECK_ORDER),
   207→        },
   208→        "guidance": (
   209→            "All 10 staged checks must pass before live trading is enabled. "
   210→            "POST /api/live/enable with confirm=true to attempt activation; "
   211→            "the endpoint refuses and returns HTTP 409 with the blocking "
   212→            "check list if any check fails. The in-memory mode flip is "
   213→            "supplemented (not replaced) by setting TRADING_MODE=live + "
   214→            "LIVE_TRADING_ENABLED=true in .env and restarting for durable "
   215→            "activation across process restarts."
   216→        ),
   217→    }

   220→# ── Individual staged checks ────────────────────────────────────────────────
   221→# Each returns a dict: {id, name, passed, severity, threshold, value, detail}.
   222→# Local imports keep a broken dependency from crashing the whole gate — the
   223→# try/except envelope converts any failure into a recorded failed check.
   224→
   225→async def _check_paper_mode() -> dict[str, Any]:
   226→    """Check #1: paper_mode ≥ 24h — current continuous paper session length."""
   227→    name = "Paper mode soak ≥ 24h"
   228→    threshold = f"trading_mode=='paper' AND session_age_s >= {PAPER_MODE_MIN_SECONDS:.0f}"
   229→    try:
   230→        from config import settings
   231→        from core.data_store import store
   232→
   233→        mode = settings.trading_mode
   234→        session_start = float(getattr(store, "session_start", time.time()))
   235→        age_s = max(0.0, time.time() - session_start)
   236→        in_paper = mode == "paper"
   237→        passed = in_paper and age_s >= PAPER_MODE_MIN_SECONDS
   238→        if passed:
   239→            detail = f"paper session running {age_s/3600.0:.2f}h (≥24h)"
   240→        elif not in_paper:
   241→            detail = (
   242→                f"not currently in paper mode (trading_mode={mode!r}); "
   243→                f"the §82 gate requires a continuous ≥24h paper soak first. "
   244→                f"Note: session_start resets on every process restart, so the "
   245→                f"measured age reflects only the current continuous session."
   246→            )
   247→        else:
   248→            detail = (
   249→                f"paper session only {age_s/3600.0:.2f}h old "
   250→                f"(need ≥{PAPER_MODE_MIN_SECONDS/3600.0:.0f}h); "
   251→                f"session_start resets on restart."
   252→            )
   253→        return {
   254→            "id": CHECK_PAPER_MODE, "name": name, "passed": passed,
   255→            "severity": SEVERITY_BLOCKING, "threshold": threshold,
   256→            "value": {"trading_mode": mode, "session_age_seconds": round(age_s, 1)},
   257→            "detail": detail,
   258→        }
   259→    except Exception as e:
   260→        log.error("[live_safety_gate] paper_mode check failed: %s", e)
   261→        return _failed(CHECK_PAPER_MODE, name, threshold, f"check raised: {e}")

   264→async def _check_positive_expectancy() -> dict[str, Any]:
   265→    """Check #2: positive expectancy — avg PnL per closed trade > 0."""
   266→    name = "Positive expectancy (avg PnL > 0)"
   267→    threshold = "closed_positions.avg_pnl > 0 (count > 0)"
   268→    try:
   269→        from core.closed_positions import closed_positions
   270→
   271→        stats = await closed_positions.get_closed_stats()
   272→        count = int(stats.get("count", 0))
   273→        avg_pnl = float(stats.get("avg_pnl", 0.0) or 0.0)
   274→        # Expectancy = mean PnL per trade = (win_rate*avg_win)+(loss_rate*avg_loss).
   275→        # stats.avg_pnl is exactly this (total_pnl/count), so the >0 test is the
   276→        # canonical positive-expectancy gate.
   277→        passed = count > 0 and avg_pnl > 0.0
   278→        if passed:
   279→            detail = f"avg_pnl=${avg_pnl:+.4f} across {count} closed trade(s) — positive expectancy"
   280→        elif count == 0:
   281→            detail = "no closed trades yet — expectancy undefined until ≥1 trade closes"
   282→        else:
   283→            detail = f"avg_pnl=${avg_pnl:+.4f} across {count} closed trade(s) — non-positive expectancy"
   284→        return {
   285→            "id": CHECK_POSITIVE_EXPECTANCY, "name": name, "passed": passed,
   286→            "severity": SEVERITY_BLOCKING, "threshold": threshold,
   287→            "value": {"count": count, "avg_pnl": round(avg_pnl, 4)},
   288→            "detail": detail,
   289→        }
   290→    except Exception as e:
   291→        log.error("[live_safety_gate] positive_expectancy check failed: %s", e)
   292→        return _failed(CHECK_POSITIVE_EXPECTANCY, name, threshold, f"check raised: {e}")

   295→async def _check_max_drawdown() -> dict[str, Any]:

   299→    try:

   303→        if passed:
   304→            detail = f"current drawdown ${dd:.2f} < ${MAX_LIVE_DRAWDOWN_USD:.2f} live gate"
   305→        else:
   306→            detail = (
   307→                f"current drawdown ${dd:.2f} >= ${MAX_LIVE_DRAWDOWN_USD:.2f} live gate "
   308→                f"(risk engine hard breaker is ${report.get('max_drawdown_limit', 0.0):.2f})"
   309→            )
   310→        return {
   311→            "id": CHECK_MAX_DRAWDOWN, "name": name, "passed": passed,
   312→            "severity": SEVERITY_BLOCKING, "threshold": threshold,
   313→            "value": {
   314→                "drawdown_dollars": round(dd, 4),
   315→                "max_drawdown_limit": float(report.get("max_drawdown_limit", 0.0) or 0.0),
   316→            },
   317→            "detail": detail,
   318→        }
   319→    except Exception as e:
   320→        log.error("[live_safety_gate] max_drawdown check failed: %s", e)

   457→async def _check_kill_switch_tested() -> dict[str, Any]:
   458→    """Check #8: kill switch has been exercised (activated then deactivated).
   459→
   460→    Primary signal: the audit trail carries at least one
   461→    ``kill_switch_activated`` AND at least one ``kill_switch_deactivated``
   462→    event (proving the operator ran the full activate→deactivate cycle, so the
   463→    durable marker file + in-memory flag + cancel-all-orders path are all
   464→    known-good).
   465→
   466→    Escape hatch: a durable marker file at ``KILL_SWITCH_TESTED_PATH`` lets an
   467→    operator assert "tested" when the audit evidence is unavailable (e.g. on a
   468→    fresh deploy where the switch was exercised before audit logging began or
   469→    the audit DB was rotated). The marker is the documented override, not the
   470→    default — the audit trail is the canonical evidence path.
   471→    """
   472→    name = "Kill switch tested (activate→deactivate exercised)"
   473→    threshold = "audit trail has ≥1 kill_switch_activated AND ≥1 kill_switch_deactivated, OR marker file present"
   474→    try:
   475→        # Escape hatch first — cheap filesystem check.
   476→        marker_present = False
   477→        try:
   478→            marker_present = KILL_SWITCH_TESTED_PATH.exists()
   479→        except OSError:
   480→            marker_present = False
   481→
   482→        activated_count = 0
   483→        deactivated_count = 0
   484→        last_activated_ts: float | None = None
   485→        last_deactivated_ts: float | None = None
   486→        try:
   487→            from core.audit_logger import audit_logger
   488→
   489→            # Pull a wide window of risk-category events so we don't miss an
   490→            # early-cycle test buried under later risk events. 500 is generous
   491→            # but bounded — the audit DB is append-only and risk events are
   492→            # rare (only fire on breaker trips / manual resets).
   493→            events = await audit_logger.get_recent_events(limit=500, category="risk")
   494→            for ev in events:
   495→                et = str(ev.get("event_type", "") or "")
   496→                if et == "kill_switch_activated":
   497→                    activated_count += 1
   498→                    ts = ev.get("timestamp")
   499→                    if ts is not None and (last_activated_ts is None or float(ts) > last_activated_ts):
   500→                        last_activated_ts = float(ts)
   501→                elif et == "kill_switch_deactivated":
   502→                    deactivated_count += 1
   503→                    ts = ev.get("timestamp")
   504→                    if ts is not None and (last_deactivated_ts is None or float(ts) > last_deactivated_ts):
   505→                        last_deactivated_ts = float(ts)
   506→        except Exception as e:
   507→            log.debug("[live_safety_gate] audit trail inspection failed: %s", e)
   508→
   509→        audit_evidence = activated_count >= 1 and deactivated_count >= 1
   510→        passed = audit_evidence or marker_present
   511→
   512→        if passed and audit_evidence:
   513→            ordered = (
   514→                last_activated_ts is not None
   515→                and last_deactivated_ts is not None
   516→                and last_deactivated_ts >= last_activated_ts
   517→            )
   518→            detail = (
   519→                f"audit trail: {activated_count} activation(s), "
   520→                f"{deactivated_count} deactivation(s); "
   521→                f"last deactivate {'followed' if ordered else 'preceded'} last activate"
   522→            )
   523→        elif passed and marker_present:
   524→            detail = (
   525→                f"operator marker file present at {KILL_SWITCH_TESTED_PATH} "
   526→                f"(audit trail had {activated_count} activate / {deactivated_count} "
   527→                f"deactivate — marker is the documented override)"
   528→            )
   529→        else:
   530→            detail = (
   531→                f"audit trail has {activated_count} activate / {deactivated_count} "
   532→                f"deactivate event(s); need ≥1 of each. Exercise the kill switch "
   533→                f"(activate via POST /api/risk/kill-switch, then reset via "
   534→                f"DELETE /api/risk/kill-switch) or create the marker file at "
   535→                f"{KILL_SWITCH_TESTED_PATH}."
   536→            )
   537→        return {
   538→            "id": CHECK_KILL_SWITCH_TESTED, "name": name, "passed": passed,
   539→            "severity": SEVERITY_BLOCKING, "threshold": threshold,
   540→            "value": {
   541→                "audit_activated_count": activated_count,
   542→                "audit_deactivated_count": deactivated_count,
   543→                "last_activated_at": last_activated_ts,
   544→                "last_deactivated_at": last_deactivated_ts,
   545→                "marker_file_present": marker_present,
   546→                "marker_path": str(KILL_SWITCH_TESTED_PATH),
   547→            },
   548→            "detail": detail,
   549→        }
   550→    except Exception as e:
   551→        log.error("[live_safety_gate] kill_switch_tested check failed: %s", e)
   552→        return _failed(CHECK_KILL_SWITCH_TESTED, name, threshold, f"check raised: {e}")

   655→# ── FastAPI route registration ───────────────────────────────────────────────
   656→
   657→# ── FastAPI request model (module-level so FastAPI resolves it as a body) ───
   658→# Mirrors the convention in ``api/server.py`` where every Pydantic request model
   659→# is defined at module scope (e.g. ``ObservationModeRequest``). Defining the
   660→# model inside ``register_routes`` made FastAPI treat the parameter as a query
   661→# arg (``loc: ["query","req"]``) instead of a JSON body.
   662→
   663→try:
   664→    from pydantic import BaseModel, Field
   665→
   666→    class EnableLiveRequest(BaseModel):
   667→        """Request body for ``POST /api/live/enable``."""
   668→
   669→        confirm: bool = Field(
   670→            default=False,
   671→            description="Must be true to authorise live-mode activation (defence against accidental clicks).",
   672→        )
   673→        reason: str = Field(
   674→            default="",
   675→            description="Operator justification for enabling live trading (recorded in the audit trail).",
   676→        )
   677→except Exception:  # pragma: no cover — pydantic is a hard project dep; this only trips in odd unit-test stubs
   678→    EnableLiveRequest = None  # type: ignore[assignment]

   681→# ── FastAPI route registration ───────────────────────────────────────────────
   682→
   683→def register_routes(app: Any) -> None:
   684→    """
   685→    Append live-safety-gate endpoints to a FastAPI app.
   686→
   687→    Endpoints (auth-protected by the caller's existing middleware):
   688→
   689→      GET /api/live/readiness
   690→          Run all 10 God Mode §82 staged checks and return the verdict dict
   691→          ({passed, checks, passed_count, total_count, blocking_checks,
   692→          checked_at}). Never 500s — a check that throws records itself as
   693→          failed.
   694→
   695→      POST /api/live/enable
   696→          Request body: {confirm: bool, reason: str}
   697→          Attempt to flip the bot into live trading mode. Requires
   698→          ``confirm=true`` (defence against accidental double-click). Runs
   699→          ``check_live_readiness`` first; if any check fails, returns HTTP 409
   700→          with the blocking-check list and the full readiness payload. On
   701→          success, flips the in-memory mode flags (live_trading_enabled=True,
   702→          trading_mode="live", paper_trade=False) and logs an audit event.
   703→
   704→          NOTE: the in-memory flip is sufficient for ``check_order`` to start
   705→          admitting live orders, but it does NOT persist across process
   706→          restarts. For durable activation, set TRADING_MODE=live and
   707→          LIVE_TRADING_ENABLED=true in .env and restart — the response
   708→          payload carries this guidance.
   709→    """
   710→    from fastapi import HTTPException
   711→
   712→    @app.get("/api/live/readiness", tags=["live"])
   713→    async def _live_readiness():
   714→        """Run all 10 §82 staged checks; return {passed, checks, blocking_checks, ...}."""
   715→        return await check_live_readiness()
   716→
   717→    @app.post("/api/live/enable", tags=["live"])
   718→    async def _enable_live(req: "EnableLiveRequest"):
   719→        """Attempt to enable live trading. Refuses (HTTP 409) if any §82 check fails."""
   720→        if not req.confirm:
   721→            raise HTTPException(
   722→                status_code=400,
   723→                detail="confirm=true is required to enable live trading (defence against accidental activation).",
   724→            )
   725→        readiness = await check_live_readiness()
   726→        if not readiness["passed"]:
   727→            # 409 Conflict: the request conflicts with the current safety state.
   728→            raise HTTPException(
   729→                status_code=409,
   730→                detail={
   731→                    "message": "Live trading NOT enabled — God Mode §82 safety gate failed.",
   732→                    "passed_count": readiness["passed_count"],
   733→                    "total_count": readiness["total_count"],
   734→                    "blocking_checks": readiness["blocking_checks"],
   735→                    "checks": readiness["checks"],
   736→                    "guidance": "Resolve every blocking check, then re-POST /api/live/enable.",
   737→                },
   738→            )
   739→        # All 10 checks passed — authorise live trading in-memory.
   740→        try:
   741→            from config import settings
   742→            from core.audit_logger import audit_logger
   743→            from core.data_store import store
   744→
   745→            settings.live_trading_enabled = True
   746→            settings.trading_mode = "live"
   747→            settings.paper_trade = False
   748→            await audit_logger.log_event(
   749→                category="system",
   750→                event_type="live_trading_enabled",
   751→                details=(
   752→                    f"Live trading enabled via §82 safety gate "
   753→                    f"(reason={req.reason or 'operator request'}). "
   754→                    f"In-memory flags flipped — set TRADING_MODE=live + "
   755→                    f"LIVE_TRADING_ENABLED=true in .env and restart for "
   756→                    f"durable activation."
   757→                ),
   758→            )
   759→            await store.log_event(
   760→                f"🔴 LIVE TRADING ENABLED via §82 safety gate — reason: {req.reason or 'operator request'}"
   761→            )
   762→            log.warning(
   763→                "[live_safety_gate] 🔴 LIVE TRADING ENABLED via safety gate — reason=%s",
   764→                req.reason or "operator request",
   765→            )
   766→            return {
   767→                "enabled": True,
   768→                "mode": settings.trading_mode,
   769→                "live_trading_enabled": settings.live_trading_enabled,
   770→                "paper_trade": settings.paper_trade,
   771→                "readiness": readiness,
   772→                "note": (
   773→                    "In-memory mode flags flipped — check_order will now admit live orders. "
   774→                    "Set TRADING_MODE=live + LIVE_TRADING_ENABLED=true in .env and restart "
   775→                    "for durable activation across process restarts."
   776→                ),
   777→            }
   778→        except Exception as e:
   779→            log.error("[live_safety_gate] live enable failed post-gate: %s", e)
   780→            raise HTTPException(
   781→                status_code=500,
   782→                detail=f"All §82 checks passed but the in-memory mode flip failed: {e}",
   783→            )
