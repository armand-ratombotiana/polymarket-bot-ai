   202→    def _note_write(self, table: str, elapsed_ms: float, ok: bool, err: Exception | None = None) -> None:
   203→        key = "inserts_ok" if ok else "inserts_failed"
   204→        if table in self._telemetry[key]:
   205→            self._telemetry[key][table] += 1
   206→            self._telemetry["write_time_ms"][table] += elapsed_ms
   207→        if err is not None:
   208→            self._telemetry["last_error"] = str(err)
   209→            self._telemetry["last_error_at"] = time.time()

   211→    async def _write_via_sqlite(self, table: str, sql: str, params: tuple) -> bool:
   212→        """Write to SQLite fallback and record telemetry."""
   213→        import sqlite3
   214→        started = time.perf_counter()
   215→
   216→        def _insert() -> None:
   217→            with sqlite3.connect(self._sqlite_path) as conn:
   218→                conn.execute(sql, params)
   219→
   220→        try:
   221→            loop = asyncio.get_running_loop()
   222→            await loop.run_in_executor(None, _insert)
   223→            self._note_write(table, (time.perf_counter() - started) * 1000, True)
   224→            return True
   225→        except Exception as e:
   226→            self._note_write(table, (time.perf_counter() - started) * 1000, False, e)
   227→            log.error("[timescale_db] SQLite write FAILED for %s: %s", table, e)
   228→            return False
   229→
   230→    async def _write(self, table: str, pg_sql: str, pg_params: tuple, sqlite_sql: str, sqlite_params: tuple) -> bool:
   231→        """Write to PostgreSQL primary first, SQLite fallback on failure."""
   232→        if self._is_postgres and self._pool:
   233→            started = time.perf_counter()
   234→            try:
   235→                async with self._pool.acquire() as conn:
   236→                    await conn.execute(pg_sql, *pg_params)
   237→                self._note_write(table, (time.perf_counter() - started) * 1000, True)
   238→                return True
   239→            except Exception as e:
   240→                self._note_write(table, (time.perf_counter() - started) * 1000, False, e)
   241→                log.error("[timescale_db] PostgreSQL write FAILED for %s: %s", table, e)
   242→        return await self._write_via_sqlite(table, sqlite_sql, sqlite_params)

   333→    async def record_feature_vector(
   334→        self,
   335→        token_id: str,
   336→        features: np.ndarray,
   337→        p_pred: float,
   338→        confidence: float,
   339→        outcome_resolved: int | None = None,
   340→    ) -> bool:
   341→        """Insert point-in-time feature snapshot and ML prediction."""
   342→        ts = time.time()
   343→        features_arr = np.asarray(features, dtype=float)
   344→        features_json = json.dumps(features_arr.tolist())
   345→        dt = datetime.datetime.fromtimestamp(ts, datetime.timezone.utc)
   346→
   347→        if self._is_postgres and self._pool:
   348→            try:
   349→                async with self._pool.acquire() as conn:
   350→                    # 1. Insert into feature.feature_snapshot
   351→                    snap_id = await conn.fetchval("""
   352→                        INSERT INTO feature.feature_snapshot (token_id, time, features_array, feature_names)
   353→                        VALUES ($1, $2, $3, $4)
   354→                        RETURNING snapshot_id;
   355→                    """, token_id, dt, features_arr.tolist(), [f"f_{i}" for i in range(len(features_arr))])
   356→
   357→                    # 2. Insert into ml.prediction
   358→                    await conn.execute("""
   359→                        INSERT INTO ml.prediction (
   360→                            model_id, token_id, time, feature_snapshot_id, raw_probability,
   361→                            calibrated_probability, confidence, actual_outcome
   362→                        )
   363→                        VALUES ($1, $2, $3, $4, $5, $6, $7, $8);
   364→                    """, "champion_ensemble_v1", token_id, dt, snap_id, p_pred, p_pred, confidence, outcome_resolved)
   365→                self._note_write("ml_feature_store", 1.0, True)
   366→                return True
   367→            except Exception as e:
   368→                self._note_write("ml_feature_store", 1.0, False, e)
   369→                log.error("[timescale_db] Feature store PostgreSQL write error: %s", e)
   370→
   371→        return await self._write_via_sqlite(
   372→            "ml_feature_store",
   373→            """
   374→                INSERT INTO ml_feature_store (timestamp, token_id, features_json, p_pred, confidence, outcome_resolved)
   375→                VALUES (?, ?, ?, ?, ?, ?);
   376→            """,
   377→            (ts, token_id, features_json, p_pred, confidence, outcome_resolved),
   378→        )

   380→    def record_prediction(self, features: np.ndarray, p_pred: float, confidence: float, token_id: str = "") -> None:
   381→        """Non-blocking recorder for ml_model.predict()."""
   382→        from ml.features import (
   383→            N_FEATURES,  # import here to avoid circular at module level
   384→        )
   385→        try:
   386→            features = np.asarray(features, dtype=np.float32)
   387→        except Exception:
   388→            features = np.zeros(N_FEATURES, dtype=np.float32)
   389→
   390→        async def _recorder() -> None:
   391→            try:
   392→                await self.record_feature_vector(
   393→                    token_id=token_id, features=features, p_pred=float(p_pred), confidence=float(confidence)
   394→                )
   395→            except Exception:
   396→                pass
   397→
   398→        try:
   399→            loop = asyncio.get_running_loop()
   400→            loop.create_task(_recorder())
   401→        except RuntimeError:
   402→            try:
   403→                asyncio.run(_recorder())
   404→            except Exception:
   405→                pass

   407→    def mark_resolved_outcomes(self, token_id: str, resolved_yes: bool) -> int:
   408→        """Update resolved outcome for token upon market resolution.
   409→
   410→        Writes to both backends:
   411→        1. TimescaleDB ml.prediction.actual_outcome (when PostgreSQL is connected)
   412→        2. SQLite ml_feature_store.outcome_resolved (always — canonical for training samples)
   413→        """
   414→        outcome = 1 if resolved_yes else 0
   415→        updated = 0
   416→
   417→        # ── 1. TimescaleDB (async pool — schedule as fire-and-forget task) ──
   418→        if self._is_postgres and self._pool:
   419→            async def _pg_update() -> None:
   420→                try:
   421→                    async with self._pool.acquire() as conn:
   422→                        await conn.execute(
   423→                            """
   424→                            UPDATE ml.prediction
   425→                            SET actual_outcome = $1
   426→                            WHERE token_id = $2 AND actual_outcome IS NULL;
   427→                            """,
   428→                            outcome, token_id,
   429→                        )
   430→                except Exception as e:
   431→                    log.error("[timescale_db] mark_resolved_outcomes PG update failed for %s: %s", token_id, e)
   432→
   433→            try:
   434→                loop = asyncio.get_running_loop()
   435→                loop.create_task(_pg_update())
   436→            except RuntimeError:
   437→                pass  # Not in an async context — SQLite path is sufficient
   438→
   439→        # ── 2. SQLite fallback (canonical training sample store) ──
   440→        try:
   441→            import sqlite3
   442→            with sqlite3.connect(self._sqlite_path) as conn:
   443→                cur = conn.execute(
   444→                    "UPDATE ml_feature_store SET outcome_resolved = ? WHERE token_id = ? AND outcome_resolved IS NULL;",
   445→                    (outcome, token_id),
   446→                )
   447→                updated = cur.rowcount
   448→        except Exception as e:
   449→            log.error("[timescale_db] mark_resolved_outcomes SQLite update failed: %s", e)
   450→
   451→        return updated

   453→    def fetch_recent_feature_vector(self, token_id: str) -> np.ndarray | None:
   454→        """
   455→        Retrieve the most recent feature vector for a token from the SQLite feature store.
   456→        Used by settlement.py to feed ground-truth outcomes back to the ML online learner.
   457→        Pads/trims to N_FEATURES so old stored vectors remain compatible.
   458→        """
   459→        from ml.features import N_FEATURES
   460→        try:
   461→            import sqlite3
   462→            with sqlite3.connect(self._sqlite_path) as conn:
   463→                cursor = conn.cursor()
   464→                cursor.execute(
   465→                    """
   466→                    SELECT features_json FROM ml_feature_store
   467→                    WHERE token_id = ? AND features_json IS NOT NULL
   468→                    ORDER BY timestamp DESC LIMIT 1;
   469→                    """,
   470→                    (token_id,),
   471→                )
   472→                row = cursor.fetchone()
   473→            if row:
   474→                arr = np.array(json.loads(row[0]), dtype=np.float32)
   475→                # Pad or trim to current N_FEATURES (handles legacy 32-feature vectors)
   476→                if len(arr) < N_FEATURES:
   477→                    arr = np.pad(arr, (0, N_FEATURES - len(arr)))
   478→                elif len(arr) > N_FEATURES:
   479→                    arr = arr[:N_FEATURES]
   480→                return arr
   481→        except Exception as e:
   482→            log.debug("[timescale_db] fetch_recent_feature_vector(%s): %s", token_id, e)
   483→        return None

   485→    def fetch_training_samples(self, min_samples: int = 100) -> tuple[np.ndarray | None, np.ndarray | None]:
   486→        """
   487→        Extract class-balanced training samples with verified ground-truth labels.
   488→
   489→        Uses stratified sampling (up to 2500 YES and 2500 NO outcomes) to prevent
   490→        class imbalance from biasing the model toward the majority resolution outcome.
   491→        Most-recent samples within each class are preferred.
   492→        """
   493→        from ml.features import N_FEATURES
   494→        try:
   495→            import sqlite3
   496→            with sqlite3.connect(self._sqlite_path) as conn:
   497→                cursor = conn.cursor()
   498→
   499→                # Fetch YES outcomes (resolved_yes = 1)
   500→                cursor.execute(
   501→                    """
   502→                    SELECT features_json, outcome_resolved
   503→                    FROM ml_feature_store
   504→                    WHERE outcome_resolved = 1 AND features_json IS NOT NULL
   505→                    ORDER BY timestamp DESC LIMIT 2500;
   506→                    """
   507→                )
   508→                yes_rows = cursor.fetchall()
   509→
   510→                # Fetch NO outcomes (resolved_yes = 0)
   511→                cursor.execute(
   512→                    """
   513→                    SELECT features_json, outcome_resolved
   514→                    FROM ml_feature_store
   515→                    WHERE outcome_resolved = 0 AND features_json IS NOT NULL
   516→                    ORDER BY timestamp DESC LIMIT 2500;
   517→                    """
   518→                )
   519→                no_rows = cursor.fetchall()
   520→
   521→            rows = yes_rows + no_rows
   522→            if len(rows) < min_samples:
   523→                return None, None
   524→
   525→            log.info(
   526→                "[timescale_db] Stratified training sample: %d YES + %d NO = %d total",
   527→                len(yes_rows), len(no_rows), len(rows),
   528→            )
   529→
   530→            X_list, y_list = [], []
   531→            for feat_str, outcome in rows:
   532→                try:
   533→                    arr = np.array(json.loads(feat_str), dtype=np.float32)
   534→                    # Pad/trim legacy vectors to current feature count
   535→                    if len(arr) < N_FEATURES:
   536→                        arr = np.pad(arr, (0, N_FEATURES - len(arr)))
   537→                    elif len(arr) > N_FEATURES:
   538→                        arr = arr[:N_FEATURES]
   539→                    X_list.append(arr)
   540→                    y_list.append(int(outcome))
   541→                except Exception:
   542→                    continue
   543→
   544→            if len(X_list) < min_samples:
   545→                return None, None
   546→
   547→            return np.array(X_list, dtype=np.float32), np.array(y_list, dtype=int)
   548→        except Exception as e:
   549→            log.warning("[timescale_db] Failed to fetch training samples: %s", e)
   550→            return None, None
   551→
   552→    def fetch_records(self, table: str, limit: int = 25) -> dict[str, Any]:
   553→        """Fetch records for database explorer."""
   554→        valid_tables = set(_TABLES)
   555→        if table not in valid_tables:
   556→            return {"is_success": False, "error": f"Invalid table {table}", "backend": self.backend_label}
   557→        try:
   558→            import sqlite3
   559→            with sqlite3.connect(self._sqlite_path) as conn:
   560→                conn.row_factory = sqlite3.Row
   561→                cursor = conn.cursor()
   562→                cursor.execute(f"SELECT * FROM {table} ORDER BY id DESC LIMIT ?", (limit,))
   563→                rows = [dict(r) for r in cursor.fetchall()]
   564→            return {
   565→                "is_success": True,
   566→                "table": table,
   567→                "records": rows,
   568→                "count": len(rows),
   569→                "backend": self.backend_label,
   570→            }
   571→        except Exception as e:
   572→            log.error("[timescale_db] fetch_records(%s) FAILED: %s", table, e)
   573→            return {
   574→                "is_success": False,
   575→                "table": table,
   576→                "records": [],
   577→                "count": 0,
   578→                "backend": self.backend_label,
   579→                "error": str(e),
   580→            }
   581→
   582→    @property
   583→    def backend_label(self) -> str:
   584→        return "postgres" if self._is_postgres else "sqlite"

   586→    def _count_rows(self, table: str) -> int:
   587→        try:
   588→            import sqlite3
   589→            with sqlite3.connect(self._sqlite_path) as conn:
   590→                cursor = conn.cursor()
   591→                cursor.execute(f"SELECT COUNT(*) FROM {table}")
   592→                return int(cursor.fetchone()[0])
   593→        except Exception:
   594→            return 0
