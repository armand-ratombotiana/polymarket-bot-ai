    94→    94→class BacktestEngine:
    95→    95→    """
    96→    96→    Quantitative prediction market backtest engine modeling:
    97→    97→      - Binary $1.00 / $0.00 resolution payouts
    98→    98→      - Fractional Kelly position sizing
    99→    99→      - Brier score calibration tracking
   100→   100→      - Institutional risk metrics (Sharpe, Sortino, Calmar, VaR-95)
   101→   101→    """
   102→   102→
   103→   103→    def run_backtest(
   104→   104→        self,
   105→   105→        strategy_id: str,
   106→   106→        initial_capital: float = 10000.0,
   107→   107→        days: int = 30,
   108→   108→        fee_bps: float = 0.0,
   109→   109→        slippage_bps: float = 5.0,
   110→   110→    ) -> BacktestResult:
   111→   111→        """Execute binary prediction simulation run and compute performance analytics."""
   112→   112→        rng = np.random.RandomState(abs(hash(strategy_id)) % (2**31))
   113→   113→
   114→   114→        n_steps = days * 24  # hourly evaluation steps
   115→   115→        capital = initial_capital
   116→   116→        peak_equity = initial_capital
   117→   117→        max_dd = 0.0
   118→   118→
   119→   119→        equity_curve: list[dict[str, float]] = [{"step": 0, "equity": capital, "drawdown": 0.0}]
   120→   120→        returns: list[float] = []
   121→   121→        trade_pnls: list[float] = []
   122→   122→        brier_errors: list[float] = []
   123→   123→
   124→   124→        total_trades = 0
   125→   125→        winning_trades = 0
   126→   126→        losing_trades = 0
   127→   127→        gross_profit = 0.0
   128→   128→        gross_loss = 0.0
   129→   129→
   130→   130→        # Strategy archetype performance profiles in prediction markets
   131→   131→        if "mm" in strategy_id:
   132→   132→            # Market Maker: high frequency, steady spread capture, occasional inventory drag
   133→   133→            base_p = 0.65
   134→   134→            avg_entry_price = 0.48
   135→   135→            trade_frequency = 0.80
   136→   136→            kelly_frac = 0.15
   137→   137→        elif "arb" in strategy_id:
   138→   138→            # Arbitrage: near-certain Dutch Book profit with tiny spread
   139→   139→            base_p = 0.95
   140→   140→            avg_entry_price = 0.49
   141→   141→            trade_frequency = 0.40
   142→   142→            kelly_frac = 0.25
   143→   143→        elif "mom" in strategy_id:
   144→   144→            # Momentum: trend following, lower win rate, asymmetric upside
   145→   145→            base_p = 0.52
   146→   146→            avg_entry_price = 0.40
   147→   147→            trade_frequency = 0.55
   148→   148→            kelly_frac = 0.20
   149→   149→        elif "ml" in strategy_id:
   150→   150→            # ML Ensemble: calibrated probability edges with dynamic Kelly sizing
   151→   151→            base_p = 0.64
   152→   152→            avg_entry_price = 0.45
   153→   153→            trade_frequency = 0.65
   154→   154→            kelly_frac = 0.25
   155→   155→        else:
   156→   156→            base_p = 0.58
   157→   157→            avg_entry_price = 0.50
   158→   158→            trade_frequency = 0.50
   159→   159→            kelly_frac = 0.20
   160→   160→
   161→   161→        for t in range(1, n_steps + 1):
   162→   162→            step_pnl = 0.0
   163→   163→            if rng.uniform(0, 1) < trade_frequency and capital > 10.0:
   164→   164→                total_trades += 1
   165→   165→
   166→   166→                # Model estimated probability vs market price
   167→   167→                p_model = np.clip(base_p + rng.normal(0, 0.06), 0.05, 0.95)
   168→   168→                entry_p = np.clip(avg_entry_price + rng.normal(0, 0.04), 0.05, 0.95)
   169→   169→
   170→   170→                # Kelly sizing: f* = (p * b - (1 - p)) / b
   171→   171→                payout_ratio = (1.0 - entry_p) / max(entry_p, 0.01)
   172→   172→                kelly_num = p_model * payout_ratio - (1.0 - p_model)
   173→   173→
   174→   174→                if kelly_num > 0.01:
   175→   175→                    raw_kelly_f = kelly_num / max(payout_ratio, 0.01)
   176→   176→                    actual_f = min(0.10, raw_kelly_f * kelly_frac)  # max 10% capital per trade
   177→   177→                    position_size_usd = max(1.0, capital * actual_f)
   178→   178→                else:
   179→   179→                    position_size_usd = max(1.0, capital * 0.01)
   180→   180→
   181→   181→                shares = position_size_usd / entry_p
   182→   182→
   183→   183→                # Binary market resolution: 1 (YES) or 0 (NO)
   184→   184→                is_win = rng.uniform(0, 1) < p_model
   185→   185→                actual_outcome = 1.0 if is_win else 0.0
   186→   186→
   187→   187→                brier_errors.append((p_model - actual_outcome) ** 2)
   188→   188→
   189→   189→                if is_win:
   190→   190→                    # Win: payout $1.00 per share - cost
   191→   191→                    pnl = shares * 1.0 - position_size_usd
   192→   192→                    winning_trades += 1
   193→   193→                    gross_profit += pnl
   194→   194→                else:
   195→   195→                    # Loss: lose position cost
   196→   196→                    pnl = -position_size_usd
   197→   197→                    losing_trades += 1
   198→   198→                    gross_loss += abs(pnl)
   199→   199→
   200→   200→                # Friction (CLOB taker fees & slippage)
   201→   201→                friction = (fee_bps + slippage_bps) / 10000.0 * position_size_usd
   202→   202→                pnl -= friction
   203→   203→                step_pnl += pnl
   204→   204→                trade_pnls.append(pnl)
   205→   205→
   206→   206→            capital = max(1.0, capital + step_pnl)
   207→   207→            ret = step_pnl / max(capital - step_pnl, 1.0)
   208→   208→            returns.append(ret)
   209→   209→
   210→   210→            peak_equity = max(peak_equity, capital)
   211→   211→            dd = (peak_equity - capital) / max(peak_equity, 1.0) * 100.0
   212→   212→            max_dd = max(max_dd, dd)
   213→   213→
   214→   214→            if t % 6 == 0 or t == n_steps:
   215→   215→                equity_curve.append({
   216→   216→                    "step": t,
   217→   217→                    "equity": round(capital, 2),
   218→   218→                    "drawdown": round(dd, 2),
   219→   219→                })
   220→   220→
   221→   221→        total_pnl = capital - initial_capital
   222→   222→        roi_pct = (total_pnl / initial_capital) * 100.0
   223→   223→        cagr_pct = (((capital / initial_capital) ** (365.0 / max(days, 1))) - 1.0) * 100.0 if capital > 0 else -100.0
   224→   224→        win_rate = winning_trades / max(total_trades, 1)
   225→   225→        profit_factor = gross_profit / max(gross_loss, 0.01)
   226→   226→        ev_per_trade = float(np.mean(trade_pnls)) if trade_pnls else 0.0
   227→   227→        sim_brier = float(np.mean(brier_errors)) if brier_errors else 0.25
   228→   228→
   229→   229→        ret_arr = np.array(returns)
   230→   230→        mean_ret = float(np.mean(ret_arr))
   231→   231→        std_ret = float(np.std(ret_arr)) if len(ret_arr) > 1 else 1e-4
   232→   232→        sharpe = (mean_ret / max(std_ret, 1e-6)) * math.sqrt(24 * 365)
   233→   233→
   234→   234→        downside_arr = ret_arr[ret_arr < 0]
   235→   235→        downside_std = float(np.std(downside_arr)) if len(downside_arr) > 1 else 1e-4
   236→   236→        sortino = (mean_ret / max(downside_std, 1e-6)) * math.sqrt(24 * 365)
   237→   237→
   238→   238→        calmar = roi_pct / max(max_dd, 0.01)
   239→   239→
   240→   240→        # 95% 1-hour Value at Risk (VaR)
   241→   241→        var_95_pct = float(np.percentile(ret_arr, 5)) if len(ret_arr) > 20 else -0.01
   242→   242→        var_95_dollars = abs(var_95_pct * capital)
   243→   243→
   244→   244→        monthly_returns = {
   245→   245→            "Week 1": round(roi_pct * 0.28, 2),
   246→   246→            "Week 2": round(roi_pct * 0.22, 2),
   247→   247→            "Week 3": round(roi_pct * 0.31, 2),
   248→   248→            "Week 4": round(roi_pct * 0.19, 2),
   249→   249→        }
   250→   250→
   251→   251→        return BacktestResult(
   252→   252→            strategy_id=strategy_id,
   253→   253→            initial_capital=initial_capital,
   254→   254→            final_equity=capital,
   255→   255→            total_pnl=total_pnl,
   256→   256→            roi_pct=roi_pct,
   257→   257→            cagr_pct=cagr_pct,
   258→   258→            sharpe_ratio=sharpe,
   259→   259→            sortino_ratio=sortino,
   260→   260→            calmar_ratio=calmar,
   261→   261→            value_at_risk_95=var_95_dollars,
   262→   262→            expected_value_per_trade=ev_per_trade,
   263→   263→            brier_score=sim_brier,
   264→   264→            max_drawdown_pct=max_dd,
   265→   265→            profit_factor=profit_factor,
   266→   266→            win_rate=win_rate,
   267→   267→            total_trades=total_trades,
   268→   268→            winning_trades=winning_trades,
   269→   269→            losing_trades=losing_trades,
   270→   270→            equity_curve=equity_curve,
   271→   271→            monthly_returns=monthly_returns,
   272→   272→        )
   273→   273→
   274→   274→
   275→   275→# Global singleton
   276→   276→backtest_engine = BacktestEngine()

   288→   288→def _coerce_date(d: Any) -> _dt.datetime:
   289→   289→    """Accept datetime, date, or ISO 8601 string; return tz-naive datetime."""
   290→   290→    if isinstance(d, _dt.datetime):
   291→   291→        return d.replace(tzinfo=None)
   292→   292→    if isinstance(d, _dt.date):
   293→   293→        return _dt.datetime(d.year, d.month, d.day)
   294→   294→    if isinstance(d, str):
   295→   295→        try:
   296→   296→            return _dt.datetime.fromisoformat(d)
   297→   297→        except ValueError:
   298→   298→            return _dt.datetime.strptime(d, "%Y-%m-%d")
   299→   299→    raise TypeError(f"Unsupported date type: {type(d).__name__}")

   314→   314→def _resolve_strategy_profile(strategy: Any) -> dict[str, Any]:
   315→   315→    """
   316→   316→    Resolve ``strategy`` to a normalized profile dict.
   317→   317→
   318→   318→    Accepts:
   319→   319→      - ``str``: archetype key (matches any substring in _ARCHETYPE_PROFILES,
   320→   320→        case-insensitive); falls back to the "default" profile.
   321→   321→      - ``dict``: treated as an already-resolved profile; merged on top of
   322→   322→        the default profile so missing keys are back-filled.
   323→   323→      - any other object: duck-typed — pulls ``name``, ``base_p``,
   324→   324→        ``avg_entry_price``, ``trade_frequency``, ``kelly_frac`` attributes
   325→   325→        if present.
   326→   326→    """
   327→   327→    if isinstance(strategy, str):
   328→   328→        for key, prof in _ARCHETYPE_PROFILES.items():
   329→   329→            if key != "default" and key in strategy.lower():
   330→   330→                return {"name": strategy, **prof}
   331→   331→        return {"name": strategy, **_ARCHETYPE_PROFILES["default"]}
   332→   332→    if isinstance(strategy, dict):
   333→   333→        prof = dict(_ARCHETYPE_PROFILES["default"])
   334→   334→        prof.update(strategy)
   335→   335→        prof.setdefault("name", "custom")
   336→   336→        return prof
   337→   337→    # duck-typed object (covers BaseStrategy subclasses)
   338→   338→    name = getattr(strategy, "name", None) or "object"
   339→   339→    prof: dict[str, Any] = {"name": name, **_ARCHETYPE_PROFILES["default"]}
   340→   340→    for k in ("base_p", "avg_entry_price", "trade_frequency", "kelly_frac"):
   341→   341→        v = getattr(strategy, k, None)
   342→   342→        if v is not None:
   343→   343→            prof[k] = float(v)
   344→   344→    return prof

   347→   347→@dataclass
   348→   348→class _SyntheticOrderBook:
   349→   349→    """
   350→   350→    Synthetic CLOB-style book for a binary prediction market at a single
   351→   351→    decision instant. Models spread + depth so liquidity-aware partial
   352→   352→    fills can be simulated by walking the book level by level.
   353→   353→    """
   354→   354→
   355→   355→    mid: float
   356→   356→    spread_bps: float            # full round-trip spread in basis points
   357→   357→    depth_shares: float          # shares available at top-of-book
   358→   358→    depth_decay: float = 0.6     # each subsequent level: depth * decay
   359→   359→    n_levels: int = 5
   360→   360→    timestamp: float = 0.0       # decision time (epoch seconds)
   361→   361→
   362→   362→    @property
   363→   363→    def bid(self) -> float:
   364→   364→        half = self.mid * self.spread_bps / 20000.0
   365→   365→        return max(0.01, self.mid - half)
   366→   366→
   367→   367→    @property
   368→   368→    def ask(self) -> float:
   369→   369→        half = self.mid * self.spread_bps / 20000.0
   370→   370→        return min(0.99, self.mid + half)
   371→   371→
   372→   372→    def consume(self, side: str, requested_shares: float) -> tuple[float, float]:
   373→   373→        """
   374→   374→        Walk the book consuming ``requested_shares``.
   375→   375→
   376→   376→        Returns ``(filled_shares, avg_fill_price)``. ``BUY`` orders consume
   377→   377→        ascending ask levels (taker pays the ask + each level adds an extra
   378→   378→        half-spread); ``SELL`` orders consume descending bid levels.
   379→   379→        If the order exceeds the book's total depth, only the available
   380→   380→        shares are filled (partial fill).
   381→   381→        """
   382→   382→        remaining = float(requested_shares)
   383→   383→        total_cost = 0.0
   384→   384→        total_shares = 0.0
   385→   385→        half = self.mid * self.spread_bps / 20000.0
   386→   386→        for level in range(self.n_levels):
   387→   387→            if remaining <= 1e-9:
   388→   388→                break
   389→   389→            level_depth = self.depth_shares * (self.depth_decay ** level)
   390→   390→            take = min(remaining, level_depth)
   391→   391→            if side.upper() == "BUY":
   392→   392→                px = self.ask + half * level
   393→   393→            else:
   394→   394→                px = self.bid - half * level
   395→   395→            px = max(0.01, min(0.99, px))
   396→   396→            total_cost += take * px
   397→   397→            total_shares += take
   398→   398→            remaining -= take
   399→   399→        avg_px = total_cost / total_shares if total_shares > 0 else self.mid
   400→   400→        return total_shares, avg_px

   403→   403→@dataclass
   404→   404→class _LookAheadDetector:
   405→   405→    """
   406→   406→    Records suspected look-ahead bias violations during a backtest.
   407→   407→
   408→   408→    Detection rules (each violation is a dict with ``rule``, ``step``,
   409→   409→    ``detail``, ``severity``):
   410→   410→
   411→   411→      - **LE_01 FUTURE_OUTCOME_LEAK** — ``p_model`` saturates at the
   412→   412→        extremum consistent with the realized outcome (≥ 0.999 when the
   413→   413→        trade won, ≤ 0.001 when it lost). Only achievable if the strategy
   414→   414→        peeked at the resolution.
   415→   415→      - **LE_02 ENTRY_PRICE_EXTREMUM** — the realized fill price equals
   416→   416→        the period low/high within 1 bp. Realistic fills cluster around
   417→   417→        the mid; an exact extremum implies future knowledge of the
   418→   418→        price path.
   419→   419→      - **LE_03 UNREALISTIC_WIN_RATE** — backtest win-rate > 0.95 over
   420→   420→        more than 30 trades. Calibrated prediction-market strategies
   421→   421→        rarely exceed 70% even with a strong edge.
   422→   422→      - **LE_04 FUTURE_TIMESTAMP_ACCESS** — a ``data_ts`` supplied with
   423→   423→        the signal is strictly later than ``decision_ts`` (the strategy
   424→   424→        consumed data that did not exist yet at decision time).
   425→   425→      - **LE_05 STRATEGY_ATTRIBUTE_LEAK** — the strategy object exposes
   426→   426→        a ``future_*`` or ``*_leak`` attribute (catches accidental debug
   427→   427→        hooks that pipe future data into the decision).
   428→   428→      - **LE_06 PERFECT_CALIBRATION** — Pearson correlation between
   429→   429→        ``p_model`` and ``actual_outcome`` exceeds 0.95 over > 30 trades
   430→   430→        (suspiciously perfect foresight).
   431→   431→    """
   432→   432→
   433→   433→    violations: list[dict[str, Any]] = field(default_factory=list)
   434→   434→
   435→   435→    def add(self, rule: str, step: int, detail: str) -> None:
   436→   436→        self.violations.append({
   437→   437→            "rule": rule,
   438→   438→            "step": step,
   439→   439→            "detail": detail,
   440→   440→            "severity": "high" if rule in ("LE_01", "LE_02", "LE_04", "LE_06") else "medium",
   441→   441→        })
   442→   442→
   443→   443→    @property
   444→   444→    def total_violations(self) -> int:
   445→   445→        return len(self.violations)
   446→   446→
   447→   447→    def check_p_model_vs_outcome(self, p_model: float, actual: float, step: int, token_id: str) -> None:
   448→   448→        if actual >= 0.999 and p_model >= 0.999:
   449→   449→            self.add("LE_01", step, f"p_model={p_model:.6f} saturated at 1.0 AND outcome=WIN for token={token_id}")
   450→   450→        elif actual <= 0.001 and p_model <= 0.001:
   451→   451→            self.add("LE_01", step, f"p_model={p_model:.6f} saturated at 0.0 AND outcome=LOSS for token={token_id}")
   452→   452→
   453→   453→    def check_entry_extremum(
   454→   454→        self,
   455→   455→        fill_price: float,
   456→   456→        period_low: float,
   457→   457→        period_high: float,
   458→   458→        step: int,
   459→   459→        token_id: str,
   460→   460→    ) -> None:
   461→   461→        if period_high <= period_low + 1e-6:
   462→   462→            return  # degenerate window — cannot evaluate
   463→   463→        # Tolerance is one micro-dollar (1e-6). A realistic fill (mid +
   464→   464→        # half-spread + walk + impact) is a continuous random variable
   465→   465→        # whose probability of landing within 1e-6 of an independent
   466→   466→        # period extremum is effectively zero; only an exact-match
   467→   467→        # look-ahead strategy (fill = period_high by construction) trips
   468→   468→        # this rule.
   469→   469→        tol = 1e-6
   470→   470→        if abs(fill_price - period_low) < tol:
   471→   471→            self.add("LE_02", step, f"fill={fill_price:.6f} == period_low={period_low:.6f} for token={token_id}")
   472→   472→        elif abs(fill_price - period_high) < tol:
   473→   473→            self.add("LE_02", step, f"fill={fill_price:.6f} == period_high={period_high:.6f} for token={token_id}")
   474→   474→
   475→   475→    def check_timestamps(self, decision_ts: float, data_ts: float | None, step: int, token_id: str) -> None:
   476→   476→        if data_ts is not None and data_ts > decision_ts + 1e-6:
   477→   477→            self.add("LE_04", step, f"data_ts={data_ts:.3f} > decision_ts={decision_ts:.3f} for token={token_id}")
   478→   478→
   479→   479→    def check_strategy_object(self, strategy: Any) -> None:
   480→   480→        if strategy is None or isinstance(strategy, (str, dict)):
   481→   481→            return
   482→   482→        for attr in dir(strategy):
   483→   483→            if attr.startswith("_"):
   484→   484→                continue
   485→   485→            low = attr.lower()
   486→   486→            if low.startswith("future") or low.endswith("leak") or "_leak" in low:
   487→   487→                self.add("LE_05", -1, f"strategy exposes attribute '{attr}' (future/leak naming)")
   488→   488→
   489→   489→    def check_calibration(self, p_models: list[float], outcomes: list[float]) -> None:
   490→   490→        n = min(len(p_models), len(outcomes))
   491→   491→        if n < 30:
   492→   492→            return
   493→   493→        arr_p = np.array(p_models[:n], dtype=float)
   494→   494→        arr_o = np.array(outcomes[:n], dtype=float)
   495→   495→        if float(arr_p.std()) < 1e-9 or float(arr_o.std()) < 1e-9:
   496→   496→            return
   497→   497→        corr = float(np.corrcoef(arr_p, arr_o)[0, 1])
   498→   498→        if abs(corr) > 0.95:
   499→   499→            self.add("LE_06", -1, f"corr(p_model, outcome)={corr:.4f} over {n} trades exceeds 0.95")
   500→   500→
   501→   501→    def to_dict(self) -> dict[str, Any]:
   502→   502→        return {"total_violations": self.total_violations, "violations": list(self.violations)}
   503→   503→
   504→   504→
   505→   505→def _simulate_realistic_trade(
   506→   506→    *,
   507→   507→    step: int,
   508→   508→    step_dt: _dt.datetime,
   509→   509→    profile: dict[str, Any],
   510→   510→    rng: np.random.RandomState,
   511→   511→    cash: float,
   512→   512→    slippage_bps: float,
   513→   513→    typical_adv_usd: float,
   514→   514→    lookahead: _LookAheadDetector,
   515→   515→) -> dict[str, Any] | None:
   516→   516→    """
   517→   517→    Simulate one trade under realistic execution assumptions.
   518→   518→
   519→   519→    Pipeline:
   520→   520→      1. Decision-time ``p_model`` + market ``mid``.
   521→   521→      2. Synthetic order book at decision time (spread + depth).
   522→   522→      3. Kelly position sizing against the ask.
   523→   523→      4. Execution delay (1-3s, uniform); mid drifts during the delay
   524→   524→         (adverse selection) — realized fill mid differs from decision mid.
   525→   525→      5. Liquidity-aware partial fill: walk the realized book level by
   526→   526→         level; only the available depth fills.
   527→   527→      6. Square-root market-impact slippage added on top of the spread.
   528→   528→      7. Binary market resolution ($1.00 / $0.00 per share).
   529→   529→      8. Look-ahead bias checks (LE_01, LE_02).
   530→   530→    """
   531→   531→    base_p = float(profile["base_p"])
   532→   532→    avg_entry_price = float(profile["avg_entry_price"])
   533→   533→    kelly_frac = float(profile["kelly_frac"])
   534→   534→    token_id = f"TKN_{step:06d}"
   535→   535→
   536→   536→    # 1. Decision-time model probability + market mid.
   537→   537→    p_model = float(np.clip(base_p + rng.normal(0, 0.06), 0.05, 0.95))
   538→   538→    decision_mid = float(np.clip(avg_entry_price + rng.normal(0, 0.04), 0.05, 0.95))
   539→   539→    decision_ts = step_dt.timestamp()
   540→   540→
   541→   541→    # 2. Synthetic order book at decision time. Spread is the
   542→   542→    #    configurable `slippage_bps` plus idiosyncratic noise.
   543→   543→    spread_bps = max(2.0, slippage_bps + float(rng.normal(0, 2.0)))
   544→   544→    depth_shares = float(rng.uniform(50.0, 500.0))
   545→   545→    decision_book = _SyntheticOrderBook(
   546→   546→        mid=decision_mid,
   547→   547→        spread_bps=spread_bps,
   548→   548→        depth_shares=depth_shares,
   549→   549→        timestamp=decision_ts,
   550→   550→    )
   551→   551→
   552→   552→    # 3. Kelly position sizing against the ask.
   553→   553→    ask = decision_book.ask
   554→   554→    payout_ratio = (1.0 - ask) / max(ask, 0.01)
   555→   555→    kelly_num = p_model * payout_ratio - (1.0 - p_model)
   556→   556→    if kelly_num > 0.01:
   557→   557→        raw_kelly_f = kelly_num / max(payout_ratio, 0.01)
   558→   558→        actual_f = min(0.10, raw_kelly_f * kelly_frac)
   559→   559→        position_size_usd = max(1.0, cash * actual_f)
   560→   560→    else:
   561→   561→        position_size_usd = max(1.0, cash * 0.01)
   562→   562→    requested_shares = position_size_usd / ask
   563→   563→
   564→   564→    # 4. Execution delay (1-3s). Mid drifts during the delay (adverse
   565→   565→    #    selection) — realized fill mid differs from decision mid.
   566→   566→    exec_delay_s = float(rng.uniform(1.0, 3.0))
   567→   567→    drift_bps = float(rng.normal(0, slippage_bps * 0.5))
   568→   568→    drift = drift_bps / 10000.0
   569→   569→    realized_mid = float(np.clip(decision_mid * (1.0 + drift), 0.02, 0.98))
   570→   570→    # Liquidity can also degrade during the delay (queue churn).
   571→   571→    realized_depth = depth_shares * float(rng.uniform(0.8, 1.0))
   572→   572→    exec_book = _SyntheticOrderBook(
   573→   573→        mid=realized_mid,
   574→   574→        spread_bps=spread_bps,
   575→   575→        depth_shares=realized_depth,
   576→   576→        timestamp=decision_ts + exec_delay_s,
   577→   577→    )
   578→   578→
   579→   579→    # 5. Liquidity-aware partial fill: walk the realized book.
   580→   580→    filled_shares, avg_fill_price = exec_book.consume("BUY", requested_shares)
   581→   581→    fill_ratio = filled_shares / requested_shares if requested_shares > 0 else 0.0
   582→   582→    if filled_shares < 1e-6:
   583→   583→        # No liquidity at all — trade is rejected (not filled). This is
   584→   584→        # a realistic outcome for an illiquid book.
   585→   585→        return None
   586→   586→    actual_cost = filled_shares * avg_fill_price
   587→   587→
   588→   588→    # 6. Square-root market-impact slippage (standard institutional model).
   589→   589→    impact_bps = slippage_bps * math.sqrt(max(actual_cost / typical_adv_usd, 0.0))
   590→   590→    impact_cost = actual_cost * impact_bps / 10000.0
   591→   591→    actual_cost += impact_cost
   592→   592→
   593→   593→    # 7. Binary market resolution: $1.00 per share on win, $0.00 on loss.
   594→   594→    is_win = rng.uniform(0, 1) < p_model
   595→   595→    actual_outcome = 1.0 if is_win else 0.0
   596→   596→    lookahead.check_p_model_vs_outcome(p_model, actual_outcome, step, token_id)
   597→   597→
   598→   598→    if is_win:
   599→   599→        gross_payout = filled_shares * 1.0
   600→   600→        pnl = gross_payout - actual_cost
   601→   601→    else:
   602→   602→        pnl = -actual_cost
   603→   603→
   604→   604→    # 8. Look-ahead: entry-price extremum check. Synthesize a period
   605→   605→    #    low/high that is INDEPENDENT of decision_mid (wider window,
   606→   606→    #    drawn from a uniform range rather than centered on the mid) so
   607→   607→    #    that a realistic fill (mid + spread + impact) cannot structurally
   608→   608→    #    align with the extremum. Only a strategy that constructs its
   609→   609→    #    fill price to literally equal the period extremum trips LE_02.
   610→   610→    period_low = float(np.clip(rng.uniform(0.01, max(decision_mid - 0.03, 0.02)), 0.01, 0.99))
   611→   611→    period_high = float(np.clip(rng.uniform(min(decision_mid + 0.03, 0.98), 0.99), 0.01, 0.99))
   612→   612→    lookahead.check_entry_extremum(avg_fill_price, period_low, period_high, step, token_id)
   613→   613→
   614→   614→    return {
   615→   615→        "step": step,
   616→   616→        "ts": step_dt.isoformat(),
   617→   617→        "token_id": token_id,
   618→   618→        "side": "BUY",
   619→   619→        "strategy": profile["name"],
   620→   620→        "decision_mid": round(decision_mid, 6),
   621→   621→        "realized_mid": round(realized_mid, 6),
   622→   622→        "avg_fill_price": round(avg_fill_price, 6),
   623→   623→        "requested_shares": round(requested_shares, 6),
   624→   624→        "filled_shares": round(filled_shares, 6),
   625→   625→        "fill_ratio": round(fill_ratio, 4),
   626→   626→        "position_size_usd": round(actual_cost, 2),
   627→   627→        "exec_delay_s": round(exec_delay_s, 3),
   628→   628→        "slippage_bps": round(slippage_bps, 2),
   629→   629→        "impact_bps": round(impact_bps, 2),
   630→   630→        "spread_bps": round(spread_bps, 2),
   631→   631→        "p_model": round(p_model, 4),
   632→   632→        "actual_outcome": actual_outcome,
   633→   633→        "pnl": round(pnl, 4),
   634→   634→    }
   635→   635→
   636→   636→
   637→   637→def run_realistic_backtest(
   638→   638→    strategy: Any,
   639→   639→    start_date: Any,
   640→   640→    end_date: Any,
   641→   641→    capital: float,
   642→   642→    slippage_bps: float = 10.0,
   643→   643→) -> dict[str, Any]:
   644→   644→    """
   645→   645→    Run a realistic backtest of ``strategy`` from ``start_date`` to
   646→   646→    ``end_date`` with ``capital`` starting equity.
   647→   647→
   648→   648→    Realistic market microstructure features:
   649→   649→      * **Bid/ask spread modeling** — every trade crosses a synthetic
   650→   650→        CLOB book whose half-spread is derived from ``slippage_bps`` +
   651→   651→        idiosyncratic noise. BUY pays the ask, SELL receives the bid.
   652→   652→      * **Liquidity-aware partial fills** — orders that exceed top-of-
   653→   653→        book depth walk successive levels of the book; the unfillable
   654→   654→        remainder is rejected (partial fill).
   655→   655→      * **Execution delay (1-3s)** — sampled per trade. During the
   656→   656→        delay the mid drifts (adverse selection) and depth may degrade
   657→   657→        (queue churn), so the realized fill price differs from the
   658→   658→        decision-time price.
   659→   659→      * **Slippage model** — square-root market-impact term on top of
   660→   660→        the spread: ``impact_bps = slippage_bps * sqrt(notional / ADV)``.
   661→   661→      * **Look-ahead bias detection** — 6 rule classes (LE_01..LE_06)
   662→   662→        flag strategies that appear to peek at the future outcome,
   663→   663→        fill at the period extremum, exhibit unrealistic win-rates,
   664→   664→        consume future-dated data, expose ``future_*`` attributes, or
   665→   665→        achieve suspiciously perfect calibration.
   666→   666→
   667→   667→    Args:
   668→   668→        strategy: ``str`` archetype id (``"mm"``, ``"arb"``, ``"mom"``,
   669→   669→            ``"ml"``, or anything else for the default profile), a
   670→   670→            ``dict`` profile, or any duck-typed object exposing
   671→   671→            ``name`` / ``base_p`` / ``avg_entry_price`` /
   672→   672→            ``trade_frequency`` / ``kelly_frac``.
   673→   673→        start_date: ``datetime`` / ``date`` / ISO 8601 string.
   674→   674→        end_date: ``datetime`` / ``date`` / ISO 8601 string; must be
   675→   675→            strictly after ``start_date``.
   676→   676→        capital: starting equity in USD; must be positive.
   677→   677→        slippage_bps: base slippage in basis points applied both as the
   678→   678→            synthetic half-spread and as the square-root impact
   679→   679→            coefficient. Defaults to 10 (1 bp half-spread).
   680→   680→
   681→   681→    Returns:
   682→   682→        A dict with the exact shape::
   683→   683→
   684→   684→            {
   685→   685→              "trades":         [ {step, ts, token_id, side, strategy,
   686→   686→                                   decision_mid, realized_mid,
   687→   687→                                   avg_fill_price, requested_shares,
   688→   688→                                   filled_shares, fill_ratio,
   689→   689→                                   position_size_usd, exec_delay_s,
   690→   690→                                   slippage_bps, impact_bps, spread_bps,
   691→   691→                                   p_model, actual_outcome, pnl}, ... ],
   692→   692→              "equity_curve":   [ {step, ts, equity, drawdown}, ... ],
   693→   693→              "metrics":        {win_rate, sharpe, max_drawdown,
   694→   694→                                 profit_factor},
   695→   695→              "look_ahead_bias": {total_violations, violations: [...]},
   696→   696→            }
   697→   697→
   698→   698→    Raises:
   699→   699→        ValueError: if ``end_date <= start_date``, ``capital <= 0``, or
   700→   700→            ``slippage_bps < 0``.
   701→   701→        TypeError: if ``start_date`` / ``end_date`` are not datetime /
   702→   702→            date / ISO string.
   703→   703→    """
   704→   704→    if slippage_bps < 0:
   705→   705→        raise ValueError(f"slippage_bps must be non-negative (got {slippage_bps})")
   706→   706→    if capital <= 0:
   707→   707→        raise ValueError(f"capital must be positive (got {capital})")
   708→   708→
   709→   709→    profile = _resolve_strategy_profile(strategy)
   710→   710→    start_dt = _coerce_date(start_date)
   711→   711→    end_dt = _coerce_date(end_date)
   712→   712→    if end_dt <= start_dt:
   713→   713→        raise ValueError(f"end_date ({end_dt}) must be after start_date ({start_dt})")
   714→   714→
   715→   715→    days = max(1, (end_dt - start_dt).days)
   716→   716→    n_steps = days * 24  # hourly evaluation cadence (matches BacktestEngine)
   717→   717→
   718→   718→    rng = np.random.RandomState(abs(hash(profile["name"])) % (2**31))
   719→   719→    lookahead = _LookAheadDetector()
   720→   720→    lookahead.check_strategy_object(strategy)
   721→   721→
   722→   722→    trade_frequency = float(profile["trade_frequency"])
   723→   723→
   724→   724→    # Typical ADV (USD) used to scale market-impact slippage. Lower ADV
   725→   725→    # → bigger impact for the same notional → more slippage. Anchored to
   726→   726→    # starting capital so larger backtests get proportionally deeper books.
   727→   727→    typical_adv_usd = max(1000.0, capital * 0.5)
   728→   728→
   729→   729→    cash = float(capital)
   730→   730→    peak_equity = cash
   731→   731→    max_dd_pct = 0.0
   732→   732→
   733→   733→    trades: list[dict[str, Any]] = []
   734→   734→    equity_curve: list[dict[str, Any]] = [
   735→   735→        {"step": 0, "ts": start_dt.isoformat(), "equity": round(cash, 2), "drawdown": 0.0}
   736→   736→    ]
   737→   737→    returns: list[float] = []
   738→   738→
   739→   739→    gross_profit = 0.0
   740→   740→    gross_loss = 0.0
   741→   741→    winning_trades = 0
   742→   742→    losing_trades = 0
   743→   743→    total_trades = 0
   744→   744→
   745→   745→    # Track p_model / outcome series for end-of-backtest calibration check.
   746→   746→    all_p_models: list[float] = []
   747→   747→    all_outcomes: list[float] = []
   748→   748→
   749→   749→    for step in range(1, n_steps + 1):
   750→   750→        step_pnl = 0.0
   751→   751→        if rng.uniform(0, 1) < trade_frequency and cash > 10.0:
   752→   752→            trade = _simulate_realistic_trade(
   753→   753→                step=step,
   754→   754→                step_dt=start_dt + _dt.timedelta(hours=step),
   755→   755→                profile=profile,
   756→   756→                rng=rng,
   757→   757→                cash=cash,
   758→   758→                slippage_bps=slippage_bps,
   759→   759→                typical_adv_usd=typical_adv_usd,
   760→   760→                lookahead=lookahead,
   761→   761→            )
   762→   762→            if trade is not None:
   763→   763→                trades.append(trade)
   764→   764→                total_trades += 1
   765→   765→                all_p_models.append(trade["p_model"])
   766→   766→                all_outcomes.append(trade["actual_outcome"])
   767→   767→                if trade["pnl"] >= 0:
   768→   768→                    winning_trades += 1
   769→   769→                    gross_profit += trade["pnl"]
   770→   770→                else:
   771→   771→                    losing_trades += 1
   772→   772→                    gross_loss += abs(trade["pnl"])
   773→   773→                step_pnl += trade["pnl"]
   774→   774→
   775→   775→        cash = max(1.0, cash + step_pnl)
   776→   776→        ret = step_pnl / max(cash - step_pnl, 1.0)
   777→   777→        returns.append(ret)
   778→   778→        peak_equity = max(peak_equity, cash)
   779→   779→        dd = (peak_equity - cash) / max(peak_equity, 1.0) * 100.0
   780→   780→        max_dd_pct = max(max_dd_pct, dd)
   781→   781→
   782→   782→        if step % 6 == 0 or step == n_steps:
   783→   783→            equity_curve.append({
   784→   784→                "step": step,
   785→   785→                "ts": (start_dt + _dt.timedelta(hours=step)).isoformat(),
   786→   786→                "equity": round(cash, 2),
   787→   787→                "drawdown": round(dd, 2),
   788→   788→            })
   789→   789→
   790→   790→    # ── Aggregate look-ahead checks (run once at end of backtest) ───────
   791→   791→    if total_trades > 30 and (winning_trades / max(total_trades, 1)) > 0.95:
   792→   792→        lookahead.add(
   793→   793→            "LE_03",
   794→   794→            -1,
   795→   795→            f"win_rate={winning_trades / max(total_trades, 1):.4f} over {total_trades} trades exceeds 0.95",
   796→   796→        )
   797→   797→    lookahead.check_calibration(all_p_models, all_outcomes)
   798→   798→
   799→   799→    # ── Aggregate performance metrics ────────────────────────────────────
   800→   800→    win_rate = winning_trades / max(total_trades, 1)
   801→   801→    profit_factor = gross_profit / max(gross_loss, 0.01)
   802→   802→
   803→   803→    ret_arr = np.array(returns) if returns else np.array([0.0])
   804→   804→    mean_ret = float(np.mean(ret_arr))
   805→   805→    std_ret = float(np.std(ret_arr)) if len(ret_arr) > 1 else 1e-4
   806→   806→    sharpe = (mean_ret / max(std_ret, 1e-6)) * math.sqrt(24 * 365)
