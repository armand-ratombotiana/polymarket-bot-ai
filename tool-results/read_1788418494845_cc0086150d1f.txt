   286→# 2. SQLite unavailable (DB_PATH -> /dev/null)
   287→# ────────────────────────────────────────────────────────────────────────────
   288→async def test_02_sqlite_unavailable_ledger_does_not_crash(caplog, monkeypatch):
   289→    """``DECISION_LEDGER_DB_PATH`` -> ``/dev/null`` — ledger writes fail
   290→    silently (logged at ERROR) and the strategy still returns a signal.
   291→
   292→    ``sqlite3.connect('/dev/null')`` succeeds but every ``CREATE TABLE`` /
   293→    ``INSERT`` fails with ``OperationalError`` (read-only device). The
   294→    decision ledger wraps every write in ``try/except``, logs at ERROR, and
   295→    returns normally — so the strategy's fire-and-forget ``_emit_ledger``
   296→    pattern never propagates the failure into the scan loop.
   297→    """
   298→    # ── (a) A fresh DecisionLedger pointed at /dev/null ──────────────────
   299→    broken_ledger = DecisionLedger(Path("/dev/null"))
   300→    with caplog.at_level("ERROR", logger="core.decision_ledger"):
   301→        # Must not raise even though the DB is unwritable.
   302→        await broken_ledger.record(
   303→            decision_id="dec-sqlite-test",
   304→            stage="PREDICTION",
   305→            token_id="TOK_SQLITE",
   306→            strategy="signal_trader",
   307→            pnl=0.0,
   308→            p_yes=0.7,
   309→            confidence=0.6,
   310→        )
   311→    # Graceful handling: the persistence error is logged at ERROR.
   312→    assert any(
   313→        "record failed" in r.message for r in caplog.records
   314→    ), "Expected a 'record failed' ERROR log when the SQLite DB is unavailable"
   315→
   316→    # ── (b) The global decision_ledger is also broken ───────────────────
   317→    # Monkeypatch the singleton's _db_path so the strategy's fire-and-forget
   318→    # writes hit /dev/null too. The strategy must still return a signal.
   319→    original_path = decision_ledger._db_path
   320→    monkeypatch.setattr(decision_ledger, "_db_path", Path("/dev/null"))
   321→
   322→    try:
   323→        strategy = SignalTraderStrategy()
   324→        book = _make_book(token_id="TOK_SQLITE")
   325→        await store.update_order_book(book)
   326→        features = __import__("numpy").zeros(38, dtype="float32")
   327→        features[0] = 0.56  # mid_price
   328→
   329→        # Mock the ML model to return a strong BUY signal so _ml_signal
   330→        # reaches the SIGNAL-stage ledger write (which will fail silently).
   331→        from unittest.mock import patch
   332→        with patch.object(ml_model, "predict", return_value=(0.85, 0.70)):
   333→            with caplog.at_level("ERROR", logger="core.decision_ledger"):
   334→                sig = strategy._ml_signal(
   335→                    "TOK_SQLITE", "sqlite-test", _basic_market(), book, features
   336→                )
   337→
   338→        # The strategy MUST still return a signal despite the broken ledger.
   339→        assert sig is not None, "Strategy must return a signal even when the ledger DB is broken"
   340→        assert sig.decision_id, "Signal must carry a decision_id even when the ledger is broken"
   341→
   342→        # Let the fire-and-forget ledger writes flush so their ERROR logs land.
   343→        await __import__("asyncio").sleep(0.3)
   344→
   345→        # The ledger writes failed but were logged (not raised).
   346→        ledger_errors = [
   347→            r for r in caplog.records if "record failed" in r.message
   348→        ]
   349→        assert ledger_errors, "Expected ledger write failures to be logged at ERROR"
   350→    finally:
   351→        # Restore the global ledger's path so subsequent tests use the temp DB.
   352→        decision_ledger._db_path = original_path
   353→

   489→# 7. Insufficient balance (paper_balance = 0)
   490→# ────────────────────────────────────────────────────────────────────────────
   491→async def test_07_insufficient_balance_does_not_crash(caplog):
   492→    """``store.paper_balance = 0`` — the order path completes without
   493→    crashing.
   494→
   495→    NOTE: The current ``InstitutionalRiskEngine.check_order`` uses
   496→    ``BANKROLL_BASELINE`` (USD 100) for sizing and checks
   497→    ``total_exposure`` against ``MAX_DEPLOYABLE_CAPITAL`` (USD 60), but
   498→    does NOT explicitly consult ``store.paper_balance`` before allowing an
   499→    order. This test pins the current behaviour: a zero-balance state does
   500→    not crash the system — the order either passes the existing exposure
   501→    gates (and is created, leaving balance tracking to record the
   502→    resulting negative balance) or is rejected by some other risk gate.
   503→    Either way, no unhandled exception propagates. A future hardening
   504→    could add an explicit paper-balance gate (out of scope for S11).
   505→    """
   506→    original_balance = store.paper_balance
   507→    original_kill = store.kill_switch_active
   508→    store.paper_balance = 0.0
   509→    store.kill_switch_active = False
   510→    order = None
   511→    try:
   512→        strategy = SignalTraderStrategy()
   513→        args = OrderArgs(
   514→            token_id="TOK_BROKE",
   515→            price=0.50,
   516→            side=Side.BUY,
   517→            size=2.0,  # cost = $1.00 — within per-market / deployable caps
   518→        )
   519→
   520→        with caplog.at_level("DEBUG", logger="strategies.base"):
   521→            # Must not raise — the system handles a zero paper_balance
   522→            # without crashing, even though no explicit balance gate exists.
   523→            order = await strategy.submit_order(args, decision_id="dec-broke-test")
   524→
   525→        # Graceful handling: either rejected by a risk gate (None) or
   526→        # created (Order) — but no crash, no unhandled exception.
   527→        assert order is None or hasattr(order, "order_id"), (
   528→            "submit_order must return None or an Order even with paper_balance=0"
   529→        )
   530→    finally:
   531→        # Clean up any order that was created so it doesn't leak into the
   532→        # next test via the global store (the autouse fixture also resets,
   533→        # but this is belt-and-braces).
   534→        if order is not None and hasattr(order, "order_id"):
   535→            store.open_orders.pop(order.order_id, None)
   536→        store.paper_balance = original_balance
   537→        store.kill_switch_active = original_kill

   540→# ────────────────────────────────────────────────────────────────────────────
   541→# 8. Concurrent duplicate signal (same token_id + strategy)
   542→# ────────────────────────────────────────────────────────────────────────────
   543→async def test_08_concurrent_duplicate_signal_no_double_order(caplog):
   544→    """Two ``MarketSignal``s for the same ``token_id`` from the same
   545→    strategy — ``_act_on_signal`` skips the second when the first order is
   546→    still open (``sig.token_id in self._active_signals`` and the order id
   547→    is still in ``store.open_orders``). Exactly one paper order is created.
   548→    """
   549→    strategy = SignalTraderStrategy()
   550→    token_id = "TOK_DUPLICATE"
   551→    book = _make_book(token_id=token_id)
   552→    await store.update_order_book(book)
   553→    # No existing position so the first signal can actually fire.
   554→    store.positions.pop(token_id, None)
   555→
   556→    sig = MarketSignal(
   557→        token_id=token_id,
   558→        slug="dup-test",
   559→        direction=Side.BUY,
   560→        confidence=0.90,
   561→        target_price=0.57,
   562→        size_usdc=1.0,
   563→        reason="duplicate-signal-test",
   564→        ml_score=0.85,
   565→        source="ml",
   566→        decision_id="dec-dup-1",
   567→    )
   568→
   569→    # First signal — should create exactly one paper order.
   570→    await strategy._act_on_signal(sig)
   571→    orders_after_first = len(store.open_orders)
   572→    assert orders_after_first == 1, (
   573→        f"First signal should create exactly 1 order, got {orders_after_first}"
   574→    )
   575→
   576→    # Second signal for the same token_id + strategy — must be a no-op
   577→    # (the first order is still open, so _act_on_signal returns early).
   578→    sig2 = MarketSignal(
   579→        token_id=token_id,
   580→        slug="dup-test",
   581→        direction=Side.BUY,
   582→        confidence=0.95,  # higher confidence — would normally win
   583→        target_price=0.57,
   584→        size_usdc=1.0,
   585→        reason="duplicate-signal-test",
   586→        ml_score=0.90,
   587→        source="ml",
   588→        decision_id="dec-dup-2",
   589→    )
   590→    await strategy._act_on_signal(sig2)
   591→    orders_after_second = len(store.open_orders)
   592→
   593→    # Graceful handling: no duplicate order created.
   594→    assert orders_after_second == orders_after_first, (
   595→        "Duplicate signal for the same token_id+strategy must not create a "
   596→        f"second order (first={orders_after_first}, second={orders_after_second})"
   597→    )
   598→
   599→    # The first order's id is cached in _active_signals.
   600→    assert token_id in strategy._active_signals, (
   601→        "First signal should have populated _active_signals[token_id]"
   602→    )
