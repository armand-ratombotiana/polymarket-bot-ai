   202→   202→    def _note_write(self, table: str, elapsed_ms: float, ok: bool, err: Exception | None = None) -> None:
   203→   203→        key = "inserts_ok" if ok else "inserts_failed"
   204→   204→        if table in self._telemetry[key]:
   205→   205→            self._telemetry[key][table] += 1
   206→   206→            self._telemetry["write_time_ms"][table] += elapsed_ms
   207→   207→        if err is not None:
   208→   208→            self._telemetry["last_error"] = str(err)
   209→   209→            self._telemetry["last_error_at"] = time.time()

   211→   211→    async def _write_via_sqlite(self, table: str, sql: str, params: tuple) -> bool:
   212→   212→        """Write to SQLite fallback and record telemetry."""
   213→   213→        import sqlite3
   214→   214→        started = time.perf_counter()
   215→   215→
   216→   216→        def _insert() -> None:
   217→   217→            with sqlite3.connect(self._sqlite_path) as conn:
   218→   218→                conn.execute(sql, params)
   219→   219→
   220→   220→        try:
   221→   221→            loop = asyncio.get_running_loop()
   222→   222→            await loop.run_in_executor(None, _insert)
   223→   223→            self._note_write(table, (time.perf_counter() - started) * 1000, True)
   224→   224→            return True
   225→   225→        except Exception as e:
   226→   226→            self._note_write(table, (time.perf_counter() - started) * 1000, False, e)
   227→   227→            log.error("[timescale_db] SQLite write FAILED for %s: %s", table, e)
   228→   228→            return False
   229→   229→
   230→   230→    async def _write(self, table: str, pg_sql: str, pg_params: tuple, sqlite_sql: str, sqlite_params: tuple) -> bool:
   231→   231→        """Write to PostgreSQL primary first, SQLite fallback on failure."""
   232→   232→        if self._is_postgres and self._pool:
   233→   233→            started = time.perf_counter()
   234→   234→            try:
   235→   235→                async with self._pool.acquire() as conn:
   236→   236→                    await conn.execute(pg_sql, *pg_params)
   237→   237→                self._note_write(table, (time.perf_counter() - started) * 1000, True)
   238→   238→                return True
   239→   239→            except Exception as e:
   240→   240→                self._note_write(table, (time.perf_counter() - started) * 1000, False, e)
   241→   241→                log.error("[timescale_db] PostgreSQL write FAILED for %s: %s", table, e)
   242→   242→        return await self._write_via_sqlite(table, sqlite_sql, sqlite_params)

   372→   372→            "ml_feature_store",

   380→   380→    def record_prediction(self, features: np.ndarray, p_pred: float, confidence: float, token_id: str = "") -> None:
   381→   381→        """Non-blocking recorder for ml_model.predict()."""
   382→   382→        from ml.features import (
   383→   383→            N_FEATURES,  # import here to avoid circular at module level
   384→   384→        )
   385→   385→        try:
   386→   386→            features = np.asarray(features, dtype=np.float32)
   387→   387→        except Exception:
   388→   388→            features = np.zeros(N_FEATURES, dtype=np.float32)
   389→   389→
   390→   390→        async def _recorder() -> None:
   391→   391→            try:
   392→   392→                await self.record_feature_vector(
   393→   393→                    token_id=token_id, features=features, p_pred=float(p_pred), confidence=float(confidence)
   394→   394→                )
   395→   395→            except Exception:
   396→   396→                pass
   397→   397→
   398→   398→        try:
   399→   399→            loop = asyncio.get_running_loop()
   400→   400→            loop.create_task(_recorder())
   401→   401→        except RuntimeError:
   402→   402→            try:
   403→   403→                asyncio.run(_recorder())
   404→   404→            except Exception:
   405→   405→                pass

   407→   407→    def mark_resolved_outcomes(self, token_id: str, resolved_yes: bool) -> int:
   408→   408→        """Update resolved outcome for token upon market resolution.
   409→   409→
   410→   410→        Writes to both backends:
   411→   411→        1. TimescaleDB ml.prediction.actual_outcome (when PostgreSQL is connected)
   412→   412→        2. SQLite ml_feature_store.outcome_resolved (always — canonical for training samples)
   413→   413→        """
   414→   414→        outcome = 1 if resolved_yes else 0
   415→   415→        updated = 0
   416→   416→
   417→   417→        # ── 1. TimescaleDB (async pool — schedule as fire-and-forget task) ──
   418→   418→        if self._is_postgres and self._pool:
   419→   419→            async def _pg_update() -> None:
   420→   420→                try:
   421→   421→                    async with self._pool.acquire() as conn:
   422→   422→                        await conn.execute(
   423→   423→                            """
   424→   424→                            UPDATE ml.prediction
   425→   425→                            SET actual_outcome = $1
   426→   426→                            WHERE token_id = $2 AND actual_outcome IS NULL;
   427→   427→                            """,
   428→   428→                            outcome, token_id,
   429→   429→                        )
   430→   430→                except Exception as e:
   431→   431→                    log.error("[timescale_db] mark_resolved_outcomes PG update failed for %s: %s", token_id, e)
   432→   432→
   433→   433→            try:
   434→   434→                loop = asyncio.get_running_loop()
   435→   435→                loop.create_task(_pg_update())
   436→   436→            except RuntimeError:
   437→   437→                pass  # Not in an async context — SQLite path is sufficient
   438→   438→
   439→   439→        # ── 2. SQLite fallback (canonical training sample store) ──
   440→   440→        try:
   441→   441→            import sqlite3
   442→   442→            with sqlite3.connect(self._sqlite_path) as conn:
   443→   443→                cur = conn.execute(
   444→   444→                    "UPDATE ml_feature_store SET outcome_resolved = ? WHERE token_id = ? AND outcome_resolved IS NULL;",
   445→   445→                    (outcome, token_id),
   446→   446→                )
   447→   447→                updated = cur.rowcount
   448→   448→        except Exception as e:
   449→   449→            log.error("[timescale_db] mark_resolved_outcomes SQLite update failed: %s", e)
   450→   450→
   451→   451→        return updated

   453→   453→    def fetch_recent_feature_vector(self, token_id: str) -> np.ndarray | None:
   454→   454→        """
   455→   455→        Retrieve the most recent feature vector for a token from the SQLite feature store.
   456→   456→        Used by settlement.py to feed ground-truth outcomes back to the ML online learner.
   457→   457→        Pads/trims to N_FEATURES so old stored vectors remain compatible.
   458→   458→        """
   459→   459→        from ml.features import N_FEATURES
   460→   460→        try:
   461→   461→            import sqlite3
   462→   462→            with sqlite3.connect(self._sqlite_path) as conn:
   463→   463→                cursor = conn.cursor()
   464→   464→                cursor.execute(
   465→   465→                    """
   466→   466→                    SELECT features_json FROM ml_feature_store
   467→   467→                    WHERE token_id = ? AND features_json IS NOT NULL
   468→   468→                    ORDER BY timestamp DESC LIMIT 1;
   469→   469→                    """,
   470→   470→                    (token_id,),
   471→   471→                )
   472→   472→                row = cursor.fetchone()
   473→   473→            if row:
   474→   474→                arr = np.array(json.loads(row[0]), dtype=np.float32)
   475→   475→                # Pad or trim to current N_FEATURES (handles legacy 32-feature vectors)
   476→   476→                if len(arr) < N_FEATURES:
   477→   477→                    arr = np.pad(arr, (0, N_FEATURES - len(arr)))
   478→   478→                elif len(arr) > N_FEATURES:
   479→   479→                    arr = arr[:N_FEATURES]
   480→   480→                return arr
   481→   481→        except Exception as e:
   482→   482→            log.debug("[timescale_db] fetch_recent_feature_vector(%s): %s", token_id, e)
   483→   483→        return None

   485→   485→    def fetch_training_samples(self, min_samples: int = 100) -> tuple[np.ndarray | None, np.ndarray | None]:
   486→   486→        """
   487→   487→        Extract class-balanced training samples with verified ground-truth labels.
   488→   488→
   489→   489→        Uses stratified sampling (up to 2500 YES and 2500 NO outcomes) to prevent
   490→   490→        class imbalance from biasing the model toward the majority resolution outcome.
   491→   491→        Most-recent samples within each class are preferred.
   492→   492→        """
   493→   493→        from ml.features import N_FEATURES
   494→   494→        try:
   495→   495→            import sqlite3
   496→   496→            with sqlite3.connect(self._sqlite_path) as conn:
   497→   497→                cursor = conn.cursor()
   498→   498→
   499→   499→                # Fetch YES outcomes (resolved_yes = 1)
   500→   500→                cursor.execute(
   501→   501→                    """
   502→   502→                    SELECT features_json, outcome_resolved
   503→   503→                    FROM ml_feature_store
   504→   504→                    WHERE outcome_resolved = 1 AND features_json IS NOT NULL
   505→   505→                    ORDER BY timestamp DESC LIMIT 2500;
   506→   506→                    """
   507→   507→                )
   508→   508→                yes_rows = cursor.fetchall()
   509→   509→
   510→   510→                # Fetch NO outcomes (resolved_yes = 0)
   511→   511→                cursor.execute(
   512→   512→                    """
   513→   513→                    SELECT features_json, outcome_resolved
   514→   514→                    FROM ml_feature_store
   515→   515→                    WHERE outcome_resolved = 0 AND features_json IS NOT NULL
   516→   516→                    ORDER BY timestamp DESC LIMIT 2500;
   517→   517→                    """
   518→   518→                )
   519→   519→                no_rows = cursor.fetchall()
   520→   520→
   521→   521→            rows = yes_rows + no_rows
   522→   522→            if len(rows) < min_samples:
   523→   523→                return None, None
   524→   524→
   525→   525→            log.info(
   526→   526→                "[timescale_db] Stratified training sample: %d YES + %d NO = %d total",
   527→   527→                len(yes_rows), len(no_rows), len(rows),
   528→   528→            )
   529→   529→
   530→   530→            X_list, y_list = [], []
   531→   531→            for feat_str, outcome in rows:
   532→   532→                try:
   533→   533→                    arr = np.array(json.loads(feat_str), dtype=np.float32)
   534→   534→                    # Pad/trim legacy vectors to current feature count
   535→   535→                    if len(arr) < N_FEATURES:
   536→   536→                        arr = np.pad(arr, (0, N_FEATURES - len(arr)))
   537→   537→                    elif len(arr) > N_FEATURES:
   538→   538→                        arr = arr[:N_FEATURES]
   539→   539→                    X_list.append(arr)
   540→   540→                    y_list.append(int(outcome))
   541→   541→                except Exception:
   542→   542→                    continue
   543→   543→
   544→   544→            if len(X_list) < min_samples:
   545→   545→                return None, None
   546→   546→
   547→   547→            return np.array(X_list, dtype=np.float32), np.array(y_list, dtype=int)
   548→   548→        except Exception as e:
   549→   549→            log.warning("[timescale_db] Failed to fetch training samples: %s", e)
   550→   550→            return None, None

   553→   553→        """Fetch records for database explorer."""
   554→   554→        valid_tables = set(_TABLES)
   555→   555→        if table not in valid_tables:
   556→   556→            return {"is_success": False, "error": f"Invalid table {table}", "backend": self.backend_label}
   557→   557→        try:

   560→   560→                conn.row_factory = sqlite3.Row
   561→   561→                cursor = conn.cursor()
   562→   562→                cursor.execute(f"SELECT * FROM {table} ORDER BY id DESC LIMIT ?", (limit,))
   563→   563→                rows = [dict(r) for r in cursor.fetchall()]
   564→   564→            return {
   565→   565→                "is_success": True,
   566→   566→                "table": table,

   570→   570→            }
   571→   571→        except Exception as e:
   572→   572→            log.error("[timescale_db] fetch_records(%s) FAILED: %s", table, e)
   573→   573→            return {
   574→   574→                "is_success": False,
   575→   575→                "table": table,
   576→   576→                "records": [],
   577→   577→                "count": 0,
