    61→def _valid_token(authorization: str | None) -> bool:
    62→    if not settings.api_token:
    63→        return False
    64→    scheme, _, creds = (authorization or "").partition(" ")
    65→    if scheme.lower() != "bearer" or not creds:
    66→        return False
    67→    return hmac.compare_digest(creds, settings.api_token)

    70→# ── WebSocket Connection Manager ──────────────────────────────────────────────
    71→
    72→class ConnectionManager:
    73→    def __init__(self) -> None:
    74→        self.active: list[WebSocket] = []
    75→
    76→    async def connect(self, ws: WebSocket) -> None:
    77→        await ws.accept()
    78→        self.active.append(ws)
    79→        log.info("WS client connected — total %d", len(self.active))
    80→
    81→    def disconnect(self, ws: WebSocket) -> None:
    82→        if ws in self.active:
    83→            self.active.remove(ws)
    84→
    85→    async def broadcast(self, data: dict) -> None:
    86→        dead = []
    87→        payload = json.dumps(data, default=str)
    88→        for ws in list(self.active):
    89→            try:
    90→                await ws.send_text(payload)
    91→            except Exception:
    92→                dead.append(ws)
    93→        for ws in dead:
    94→            if ws in self.active:
    95→                self.active.remove(ws)

   104→async def _seed_markets(limit: int = 60) -> list[str]:
   105→    token_ids: list[str] = []
   106→    try:
   107→        markets = await gamma_client.get_markets(active=True, limit=limit, order="volume24hr")
   108→        for mkt in markets:
   109→            slug = mkt.get("slug") or mkt.get("groupItemTitle") or ""
   110→            ids = gamma_client.extract_token_ids(mkt)
   111→            for tid in ids:
   112→                token_ids.append(tid)
   113→                if slug:
   114→                    store.market_slugs[tid] = slug
   115→                # Ingest into vector store
   116→                vector_store.add_market(tid, mkt)
   117→
   118→        vector_store.build_index()
   119→        vector_store.save_to_disk()
   120→
   121→        if token_ids:
   122→            unique_ids = list(dict.fromkeys(token_ids))
   123→            log.info("Seeded %d unique tokens from %d Gamma markets", len(unique_ids), len(markets))
   124→            await store.log_event(f"📊 Seeded {len(unique_ids)} market tokens & built Vector Embeddings index")
   125→            return unique_ids
   126→    except Exception as e:
   127→        log.error("Market seeding failed: %s", e)
   128→        await store.log_event(f"⚠ Market seed failed: {e}")
   129→    return token_ids

   358→async def _build_snapshot() -> dict:
   359→    from core.safety import kill_switch_file_exists
   360→    books = []
   361→    async with store._lock:
   362→        for tid, book in store.order_books.items():
   363→            books.append({
   364→                "token_id": tid,
   365→                "slug": store.market_slugs.get(tid, tid[:14]),
   366→                "best_bid": book.best_bid,
   367→                "best_ask": book.best_ask,
   368→                "mid": book.mid,
   369→                "spread": book.spread,
   370→                "updated_at": book.updated_at,
   371→            })
   372→
   373→    orders = []
   374→    for o in store.open_orders.values():
   375→        orders.append({
   376→            "order_id": o.order_id,
   377→            "token_id": o.token_id,
   378→            "slug": store.market_slugs.get(o.token_id, o.token_id[:14]),
   379→            "side": o.side.value,
   380→            "price": o.price,
   381→            "size": o.size,
   382→            "size_matched": o.size_matched,
   383→            "strategy": o.strategy,
   384→            "paper": o.paper,
   385→            "created_at": o.created_at,
   386→        })
   387→
   388→    positions = []
   389→    for tid, pos in store.positions.items():
   390→        if pos.yes_shares > 0 or pos.total_invested > 0 or pos.realised_pnl != 0:
   391→            # Mark-to-live-mid unrealized P&L for this position.
   392→            # Falls back to cost basis (avg_entry_price) when no live quote is tracked.
   393→            _book = store.order_books.get(tid)
   394→            current_price = _book.mid if _book and _book.mid is not None else pos.avg_entry_price
   395→            _pos_unrealized = 0.0
   396→            if pos.yes_shares > 0:
   397→                _pos_unrealized += (current_price - pos.avg_entry_price) * pos.yes_shares
   398→            if pos.no_shares > 0:
   399→                _pos_unrealized += ((1.0 - current_price) - pos.avg_entry_price) * pos.no_shares
   400→            positions.append({
   401→                "token_id": tid,
   402→                "slug": store.market_slugs.get(tid, tid[:14]),
   403→                "yes_shares": pos.yes_shares,
   404→                "avg_entry_price": pos.avg_entry_price,
   405→                "total_invested": pos.total_invested,
   406→                "realised_pnl": pos.realised_pnl,
   407→                "current_price": round(current_price, 4),
   408→                "unrealized_pnl": round(_pos_unrealized, 4),
   409→            })

   411→    trades = []
   412→    for t in store.trades[-50:]:

   414→            "trade_id": t.trade_id,
   415→            "token_id": t.token_id,

   475→_cors_origins = settings.cors_origin_list
   476→app.add_middleware(
   477→    CORSMiddleware,

   484→
   485→@app.middleware("http")
   486→async def enforce_api_auth(request: Request, call_next):
   487→    """Fail-closed bearer-token auth on every route except the liveness probe and OPTIONS preflight."""
   488→    if request.method == "OPTIONS":
   489→        return await call_next(request)
   490→
   491→    origin = request.headers.get("origin")
   492→    cors_allowed = bool(origin and (origin in settings.cors_origin_list or "*" in settings.cors_origin_list or not settings.cors_origin_list))
   493→    cors_headers = {}
   494→    if origin and cors_allowed:
   495→        cors_headers["access-control-allow-origin"] = origin
   496→        cors_headers["access-control-allow-credentials"] = "true"
   497→        cors_headers["access-control-allow-headers"] = "Authorization, Content-Type"
   498→        cors_headers["access-control-allow-methods"] = "GET, POST, PUT, DELETE, OPTIONS"
   499→
   500→    path = request.url.path
   501→    if path in PUBLIC_PATHS or path.startswith("/api/health"):
   502→        response = await call_next(request)
   503→        if origin and cors_allowed:
   504→            response.headers["access-control-allow-origin"] = origin
   505→            response.headers["access-control-allow-credentials"] = "true"
   506→        return response
   507→
   508→    if not settings.api_token:
   509→        return JSONResponse(
   510→            status_code=503,
   511→            content={"detail": "API authentication not configured — set API_TOKEN in .env", "code": "AUTH_NOT_CONFIGURED"},
   512→            headers=cors_headers,
   513→        )
   514→    if not _valid_token(request.headers.get("authorization")):
   515→        return JSONResponse(
   516→            status_code=401,
   517→            content={"detail": "Unauthorized — missing or invalid API token"},
   518→            headers=cors_headers,
   519→        )
   520→
   521→    response = await call_next(request)
   522→    if origin and cors_allowed:
   523→        response.headers["access-control-allow-origin"] = origin
   524→        response.headers["access-control-allow-credentials"] = "true"
   525→    return response

   595→@app.get("/api/analytics", tags=["system"])
   596→async def get_analytics():
   597→    trades = store.trades
   598→    total_trades = len(trades)
   599→    # Only trades that actually closed a position (pnl != 0) determine win rate.
   600→    closed_trades = [t for t in trades if t.pnl != 0]
   601→    winning_trades = sum(1 for t in closed_trades if t.pnl > 0)
   602→    losing_trades = sum(1 for t in closed_trades if t.pnl < 0)
   603→    win_rate = (winning_trades / len(closed_trades)) if closed_trades else 0.0
   604→    total_vol = sum(t.price * t.size for t in trades)
   605→
   606→    # Wilson 95% confidence interval for the win rate (sample-size honest).
   607→    n = len(closed_trades)
   608→    z = 1.96
   609→    ci_low = ci_high = None
   610→    if n > 0:
   611→        p = win_rate
   612→        denom = 1 + z * z / n
   613→        centre = (p + z * z / (2 * n)) / denom
   614→        half = z * ((p * (1 - p) / n + z * z / (4 * n * n)) ** 0.5) / denom
   615→        ci_low = max(0.0, centre - half)
   616→        ci_high = min(1.0, centre + half)
   617→
   618→    wins = sum(t.pnl for t in closed_trades if t.pnl > 0)
   619→    losses = sum(t.pnl for t in closed_trades if t.pnl < 0)
   620→    profit_factor = wins / max(1e-9, -losses) if losses else (None if not wins else float("inf"))
   621→
   622→    # Average win / loss and per-trade expectancy.
   623→    # Expectancy = (win_rate * avg_win) + (loss_rate * avg_loss) where avg_loss is negative.
   624→    avg_win = (wins / winning_trades) if winning_trades else 0.0
   625→    avg_loss = (losses / losing_trades) if losing_trades else 0.0  # negative or zero
   626→    loss_rate = 1.0 - win_rate
   627→    expectancy = (win_rate * avg_win) + (loss_rate * avg_loss)
   628→    sharpe_ratio = ml_model.sharpe_ratio
   629→
   630→    # Unrealized P&L: mark open positions to the live order-book mid when present.
   631→    unrealized_pnl = 0.0
   632→    for p in store.positions.values():
   633→        if p.current_exposure <= 0.001:
   634→            continue
   635→        book = store.order_books.get(p.token_id)
   636→        mark = book.mid if book and book.mid is not None else None
   637→        if mark is None:
   638→            mark = p.avg_entry_price  # cost-basis mark (no live quote)
   639→        unrealized_pnl += (mark - p.avg_entry_price) * p.yes_shares
   640→        unrealized_pnl += ((1.0 - mark) - p.avg_entry_price) * p.no_shares
   641→
   642→    realized_pnl = store.daily_pnl
   643→    net_pnl = realized_pnl + unrealized_pnl
   644→    equity = store.paper_balance
   645→    max_drawdown_dollars = max(0.0, store.peak_equity - equity)
   646→    max_drawdown_pct = (max_drawdown_dollars / store.peak_equity) if store.peak_equity > 0 else 0.0
   647→
   648→    exp = compute_exposure()
   649→    deployable = float(MAX_DEPLOYABLE_CAPITAL)
   650→    risk_utilization = min(1.0, exp["maximum_remaining_loss"] / deployable) if deployable > 0 else 0.0
   651→
   652→    # Data freshness: seconds since the newest tracked order-book update.
   653→    books = store.order_books.values()
   654→    freshness = max((time.time() - b.updated_at for b in books), default=0.0)
   655→
   656→    return {
   657→        "equity": round(equity, 2),
   658→        "realized_pnl": round(realized_pnl, 2),
   659→        "unrealized_pnl": round(unrealized_pnl, 2),
   660→        "net_pnl": round(net_pnl, 2),
   661→        "total_trades": total_trades,
   662→        "winning_trades": winning_trades,
   663→        "losing_trades": losing_trades,
   664→        "closed_trades": len(closed_trades),
   665→        "open_trades": total_trades - len(closed_trades),
   666→        "win_rate": round(win_rate, 4),
   667→        "win_rate_ci_low": round(ci_low, 4) if ci_low is not None else None,
   668→        "win_rate_ci_high": round(ci_high, 4) if ci_high is not None else None,
   669→        "profit_factor": round(profit_factor, 2) if isinstance(profit_factor, float) else profit_factor,
   670→        "avg_win": round(avg_win, 4),
   671→        "avg_loss": round(avg_loss, 4),
   672→        "expectancy": round(expectancy, 4),
   673→        "sharpe_ratio": round(sharpe_ratio, 4) if sharpe_ratio is not None else None,
   674→        "max_drawdown_dollars": round(max_drawdown_dollars, 2),
   675→        "max_drawdown_pct": round(max_drawdown_pct, 4),
   676→        "total_volume_usdc": round(total_vol, 2),
   677→        "open_exposure": round(exp["maximum_remaining_loss"], 2),
   678→        "open_position_count": exp["open_position_count"],
   679→        "pending_order_capital": round(exp["reserved_for_pending_orders"], 2),
   680→        "risk_utilization": round(risk_utilization, 4),
   681→        "mode": settings.trading_mode,
   682→        "data_freshness_seconds": round(freshness, 1),
   683→        "peak_equity": round(store.peak_equity, 2),
   684→        "active_strategies": list(strategy_registry.get_active_instances().keys()),
   685→    }

   745→
   746→@app.get("/api/ai/predict/{token_id}", tags=["ai"])
   747→async def get_ai_prediction(token_id: str):
   748→    """
   749→    Return the ML ensemble's directional view for a single YES token.
   750→
   751→    Reads market metadata from `market_discovery.catalog` (the universal
   752→    catalog maintained by `core/market_discovery.py`) — NOT from
   753→    `store.market_info` (that attribute does not exist on the DataStore).
   754→
   755→    Response:
   756→      * p_yes            — calibrated ensemble P(YES) ∈ (0.01, 0.99)
   757→      * confidence       — |p_yes - 0.5| * 2  ∈ [0, 1]
   758→      * market_mid       — live book mid price (or None if book is empty)
   759→      * edge             — p_yes - market_mid (positive ⇒ model is more
   760→                           bullish than the market; negative ⇒ bearish)
   761→      * edge_bps         — edge expressed in basis points
   762→      * recommended_action — BUY | SELL | HOLD (threshold-gated)
   763→      * thresholds       — the gates used so callers can audit the decision
   764→      * market           — the catalog metadata used to build the feature vector
   765→      * model_status     — whether the ensemble was actually trained
   766→
   767→    A HOLD recommendation does NOT imply no edge — it means the edge was
   768→    below the configured conviction threshold. Always inspect `edge` /
   769→    `edge_bps` before deciding to act.
   770→    """
   771→    # ── R10: pull metadata from market_discovery.catalog (NOT store.market_info) ──
   772→    from core.market_discovery import market_discovery
   773→
   774→    catalog_record = market_discovery.catalog.get(token_id)
   775→    if not catalog_record:
   776→        raise HTTPException(
   777→            status_code=404,
   778→            detail=f"token_id '{token_id}' not present in market_discovery.catalog — "
   779→                   "wait for the next catalog sync or call /api/markets/coverage",
   780→        )
   781→
   782→    # extract_features expects a dict with Gamma-style keys (volume24hr, volume,
   783→    # liquidity|liquidityNum, endDate|end_date_iso|endDateIso). The catalog
   784→    # record uses normalized snake_case keys, so bridge both shapes here.
   785→    market_for_features = dict(catalog_record)
   786→    market_for_features.setdefault(
   787→        "volume24hr", catalog_record.get("volume_24h", 0.0)
   788→    )
   789→    market_for_features.setdefault(
   790→        "volume", catalog_record.get("total_volume", 0.0)
   791→    )
   792→    # liquidity key already matches; bridge endDate aliases just in case.
   793→    if "endDate" not in market_for_features and catalog_record.get("end_date"):
   794→        market_for_features["endDate"] = catalog_record["end_date"]
   795→
   796→    # ── Live order book (needed for the microstructure feature vector) ──
   797→    book = await store.get_order_book(token_id)
   798→    if book is None:
   799→        # Hint the poller to prioritize this token, then surface a 502.
   800→        book_poller.prioritize_tokens([token_id])
   801→        raise HTTPException(
   802→            status_code=502,
   803→            detail=f"no live order book for token '{token_id}' — poller prioritized; retry shortly",
   804→        )
   805→
   806→    market_mid = book.mid
   807→    best_bid = book.best_bid
   808→    best_ask = book.best_ask
   809→
   810→    # ── Feature extraction + ensemble prediction ──
   811→    from ml.features import extract_features
   812→
   813→    features = extract_features(market_for_features, book)
   814→    if features is None:
   815→        raise HTTPException(
   816→            status_code=422,
   817→            detail=(
   818→                "feature extraction returned None — book mid is missing or "
   819→                "outside the tradeable (0.001, 0.999) band; cannot score"
   820→            ),
   821→        )
   822→
   823→    p_yes, confidence = ml_model.predict(features, token_id=token_id)
   824→    p_yes = float(p_yes)
   825→    confidence = float(confidence)
   826→
   827→    # ── Edge vs market mid ──
   828→    if market_mid is not None:
   829→        edge = float(p_yes) - float(market_mid)
   830→        edge_bps = round(edge * 10_000.0, 1)
   831→    else:
   832→        edge = None
   833→        edge_bps = None
   834→
   835→    # ── Recommendation thresholds ──
   836→    # Edge-based conviction gate (mirrors the spirit of strategies/signal_trader.py
   837→    # but expressed in terms of the model-vs-market edge rather than the absolute
   838→    # p_yes level, so a model that sees a 50/50 coin priced at 0.20 correctly
   839→    # surfaces a BUY rather than a HOLD just because |p_yes − 0.5| is small).
   840→    #
   841→    #   BUY  : edge ≥ +2ct (200 bps) AND confidence ≥ 0.10
   842→    #   SELL : edge ≤ −2ct (−200 bps) AND confidence ≥ 0.10
   843→    #   HOLD : otherwise (insufficient edge OR insufficient model conviction)
   844→    MIN_EDGE_CT = 0.02          # 2 cents
   845→    MIN_CONFIDENCE = 0.10
   846→
   847→    recommended_action = "HOLD"
   848→    if edge is None:
   849→        action_reason = "market mid unavailable — cannot compute edge"
   850→    elif confidence < MIN_CONFIDENCE:
   851→        recommended_action = "HOLD"
   852→        action_reason = (
   853→            f"confidence {confidence:.3f} < {MIN_CONFIDENCE:.2f} — model is too "
   854→            f"uncertain to act despite edge={edge*100:+.2f}ct"
   855→        )
   856→    elif edge >= MIN_EDGE_CT:
   857→        recommended_action = "BUY"
   858→        action_reason = (
   859→            f"edge=+{edge*100:.2f}ct ≥ +{MIN_EDGE_CT*100:.0f}ct AND "
   860→            f"confidence={confidence:.3f} ≥ {MIN_CONFIDENCE:.2f}"
   861→        )
   862→    elif edge <= -MIN_EDGE_CT:
   863→        recommended_action = "SELL"
   864→        action_reason = (
   865→            f"edge={edge*100:.2f}ct ≤ −{MIN_EDGE_CT*100:.0f}ct AND "
   866→            f"confidence={confidence:.3f} ≥ {MIN_CONFIDENCE:.2f}"
   867→        )
   868→    else:
   869→        recommended_action = "HOLD"
   870→        action_reason = (
   871→            f"|edge|={abs(edge)*100:.2f}ct below ±{MIN_EDGE_CT*100:.0f}ct "
   872→            f"conviction threshold"
   873→        )
   874→
   875→    # Compact market payload (avoid duplicating the full catalog record)
   876→    market_payload = {
   877→        "token_id": catalog_record.get("token_id"),
   878→        "event_id": catalog_record.get("event_id"),
   879→        "event_title": catalog_record.get("event_title"),
   880→        "question": catalog_record.get("question"),
   881→        "slug": catalog_record.get("slug") or store.market_slugs.get(token_id, ""),
   882→        "outcome": catalog_record.get("outcome"),
   883→        "category": catalog_record.get("category"),
   884→        "end_date": catalog_record.get("end_date"),
   885→        "status": catalog_record.get("status"),
   886→        "volume_24h": catalog_record.get("volume_24h"),
   887→        "total_volume": catalog_record.get("total_volume"),
   888→        "liquidity": catalog_record.get("liquidity"),
   889→        "last_synced": catalog_record.get("last_synced"),
   890→    }
   891→
   892→    return {
   893→        "token_id": token_id,
   894→        "p_yes": round(p_yes, 4),
   895→        "confidence": round(confidence, 4),
   896→        "market_mid": round(market_mid, 4) if market_mid is not None else None,
   897→        "best_bid": round(best_bid, 4) if best_bid is not None else None,
   898→        "best_ask": round(best_ask, 4) if best_ask is not None else None,
   899→        "spread": round(book.spread, 4) if book.spread is not None else None,
   900→        "edge": round(edge, 4) if edge is not None else None,
   901→        "edge_bps": edge_bps,
   902→        "recommended_action": recommended_action,
   903→        "action_reason": action_reason,
   904→        "thresholds": {
   905→            "min_edge_cents": MIN_EDGE_CT * 100.0,
   906→            "min_confidence": MIN_CONFIDENCE,
   907→        },
   908→        "model_status": {
   909→            "model_ready": ml_model.rf is not None,
   910→            "model_version": model_registry.active_version,
   911→            "brier_score": ml_model.brier_score,
   912→            "roc_auc": ml_model.roc_auc,
   913→            "ece": ml_model.ece,
   914→            "n_online_updates": ml_model._n_updates,
   915→        },
   916→        "market": market_payload,
   917→        "book_updated_at": book.updated_at,
   918→        "timestamp": time.time(),
   919→    }

   924→@app.get("/api/history/ohlcv/{token_id}", tags=["markets"])
   925→async def get_market_ohlcv(token_id: str, resolution: str = "5m", count: int = 40):
   926→    """Return OHLCV candlestick bars for visual charting.
   927→
   928→    Priority:
   929→      1. Real candles from TimescaleDB continuous aggregates (market.price_candle_*)
   930→         when TimescaleDB is connected and rows exist for this token — labeled synthetic=False.
   931→      2. Seeded random-walk anchored to live mid when no stored candles exist —
   932→         explicitly labeled synthetic=True so callers always know the data source.
   933→    """
   934→    book = await store.get_order_book(token_id)
   935→    mid = (book.mid if book else 0.5) or 0.5
   936→    slug = store.market_slugs.get(token_id, token_id[:14])
   937→    step_sec = 60 if resolution == "1m" else 300 if resolution == "5m" else 3600
   938→    view_map = {"1m": "market.price_candle_1m", "5m": "market.price_candle_5m", "1h": "market.price_candle_1h"}
   939→    pg_view = view_map.get(resolution, "market.price_candle_5m")
   940→
   941→    # ── Attempt real TimescaleDB continuous-aggregate candles ─────────────────
   942→    from core.timescale_db import timescale_db
   943→    if timescale_db._is_postgres and timescale_db._pool:
   944→        try:
   945→            async with timescale_db._pool.acquire() as conn:
   946→                rows = await conn.fetch(
   947→                    f"""
   948→                    SELECT bucket, open, high, low, close, vwap, tick_count
   949→                    FROM {pg_view}
   950→                    WHERE token_id = $1
   951→                    ORDER BY bucket DESC
   952→                    LIMIT $2;
   953→                    """,
   954→                    token_id,
   955→                    count,
   956→                )
   957→            if rows:
   958→                bars = [
   959→                    {
   960→                        "timestamp": float(r["bucket"].timestamp()),
   961→                        "open": round(float(r["open"]), 4),
   962→                        "high": round(float(r["high"]), 4),
   963→                        "low": round(float(r["low"]), 4),
   964→                        "close": round(float(r["close"]), 4),
   965→                        "volume": float(r["tick_count"]),
   966→                        "vwap": round(float(r["vwap"]), 4) if r["vwap"] is not None else None,
   967→                    }
   968→                    for r in reversed(rows)
   969→                ]
   970→                return {
   971→                    "token_id": token_id,
   972→                    "slug": slug,
   973→                    "resolution": resolution,
   974→                    "bars": bars,
   975→                    "count": len(bars),
   976→                    "synthetic": False,
   977→                    "source": pg_view,
   978→                    "data_age_seconds": round(time.time() - bars[-1]["timestamp"], 1) if bars else None,
   979→                }
   980→        except Exception as e:
   981→            log.debug("[ohlcv] TimescaleDB candle query failed for %s: %s", token_id, e)
   982→
   983→    # ── Synthetic fallback: seeded random-walk anchored to live mid ───────────
   984→    rng = np.random.RandomState(abs(hash(token_id + resolution)) % (2**31))
   985→    now = time.time()
   986→
   987→    bars = []
   988→    curr_price = max(mid * (1.0 + rng.uniform(-0.06, 0.06)), 0.05)
   989→    for i in range(count):
   990→        ts = now - (count - i) * step_sec
   991→        drift = rng.uniform(-0.012, 0.012)
   992→        open_p = curr_price
   993→        close_p = max(min(open_p + drift, 0.98), 0.02)
   994→        high_p = max(open_p, close_p) + abs(rng.uniform(0.001, 0.008))
   995→        low_p = min(open_p, close_p) - abs(rng.uniform(0.001, 0.008))
   996→        vol = float(rng.uniform(500, 15000))

  1270→@app.post("/api/positions/{token_id}/close", tags=["trading"])
  1271→async def close_position(token_id: str, req: PositionCloseRequest):
  1272→    """
  1273→    One-click marketable close of an open position.
  1274→
  1275→    Long YES positions are closed by submitting a SELL order at the current
  1276→    `best_bid` (a marketable limit — any resting bid ≥ best_bid is matched
  1277→    immediately). Long NO positions are closed by submitting a BUY order at
  1278→    the current `best_ask` (symmetric — covers the synthetic short).
  1279→
  1280→    Risk checks (`risk_manager.check_order`) are applied exactly as for a
  1281→    manual `/api/trade`, and the order flows through `paper_sim` when
  1282→    `settings.paper_trade` is true, or `clob_client` otherwise.
  1283→
  1284→    Use `dry_run=true` to preview the fill price, share count, and
  1285→    estimated realised P&L without submitting an order.
  1286→    """
  1287→    async with store._lock:
  1288→        pos = store.positions.get(token_id)
  1289→        # Snapshot the values we need before releasing the lock so we don't
  1290→        # hold store._lock across the (potentially blocking) order placement.
  1291→        if pos is None:
  1292→            long_yes = 0.0
  1293→            long_no = 0.0
  1294→            avg_entry_price = 0.0
  1295→            total_invested = 0.0
  1296→            realised_pnl = 0.0
  1297→            strategy = ""
  1298→        else:
  1299→            long_yes = float(pos.yes_shares)
  1300→            long_no = float(pos.no_shares)
  1301→            avg_entry_price = float(pos.avg_entry_price)
  1302→            total_invested = float(pos.total_invested)
  1303→            realised_pnl = float(pos.realised_pnl)
  1304→            strategy = pos.strategy or ""
  1305→
  1306→    if long_yes <= 0.0 and long_no <= 0.0:
  1307→        raise HTTPException(
  1308→            status_code=404,
  1309→            detail=f"no open position for token '{token_id}' — nothing to close",
  1310→        )
  1311→
  1312→    # ── Determine side, price, and size from the live book ──
  1313→    book = await store.get_order_book(token_id)
  1314→    if book is None or (book.best_bid is None and book.best_ask is None):
  1315→        # Treat a missing or fully empty book the same — surface 502 and hint
  1316→        # the poller to reprioritize this token so the next call succeeds.
  1317→        book_poller.prioritize_tokens([token_id])
  1318→        raise HTTPException(
  1319→            status_code=502,
  1320→            detail=f"no live order book for token '{token_id}' — poller prioritized; retry shortly",
  1321→        )
  1322→
  1323→    if long_yes > 0.0:
  1324→        # Close a long YES position: SELL into the bid ladder at best_bid.
  1325→        side = Side.SELL
  1326→        if book.best_bid is None:
  1327→            raise HTTPException(
  1328→                status_code=502,
  1329→                detail="cannot close long YES position — best_bid is empty (no resting bids)",
  1330→            )
  1331→        close_price = float(book.best_bid)
  1332→        available_shares = long_yes
  1333→    else:
  1334→        # Close a long NO position: BUY at best_ask (covers the synthetic short).
  1335→        side = Side.BUY
  1336→        if book.best_ask is None:
  1337→            raise HTTPException(
  1338→                status_code=502,
  1339→                detail="cannot close long NO position — best_ask is empty (no resting asks)",
  1340→            )
  1341→        close_price = float(book.best_ask)
  1342→        available_shares = long_no
  1343→
  1344→    # Cap the close size if the caller requested a partial scale-out.
  1345→    if req.max_size_shares is not None:
  1346→        size_shares = min(float(req.max_size_shares), available_shares)
  1347→    else:
  1348→        size_shares = available_shares
  1349→
  1350→    if size_shares <= 0.0:
  1351→        raise HTTPException(
  1352→            status_code=400,
  1353→            detail="requested close size resolves to 0 shares — nothing to close",
  1354→        )
  1355→
  1356→    # ── Estimated P&L preview (uses cost basis for long YES positions) ──
  1357→    estimated_pnl = 0.0
  1358→    if side == Side.SELL and avg_entry_price > 0.0:
  1359→        # P&L = (sell_price - avg_entry) * shares  — for YES side only.
  1360→        estimated_pnl = (close_price - avg_entry_price) * size_shares
  1361→    # (NO side P&L requires the YES/NO parity identity and is computed at fill
  1362→    #  time by store.record_fill / paper_sim._execute_fill; we don't fabricate
  1363→    #  a number here.)
  1364→
  1365→    notional_usdc = close_price * size_shares
  1366→    slug = store.market_slugs.get(token_id, token_id[:12])
  1367→
  1368→    if req.dry_run:
  1369→        return {
  1370→            "status": "dry_run",
  1371→            "token_id": token_id,
  1372→            "slug": slug,
  1373→            "side": side.value,
  1374→            "price": round(close_price, 4),
  1375→            "size_shares": round(size_shares, 4),
  1376→            "notional_usdc": round(notional_usdc, 2),
  1377→            "estimated_pnl": round(estimated_pnl, 2),
  1378→            "best_bid": round(float(book.best_bid), 4) if book.best_bid is not None else None,
  1379→            "best_ask": round(float(book.best_ask), 4) if book.best_ask is not None else None,
  1380→            "book_updated_at": book.updated_at,
  1381→            "paper_trade": settings.paper_trade,
  1382→            "remaining_position": {
  1383→                "yes_shares": max(0.0, long_yes - (size_shares if side == Side.SELL else 0.0)),
  1384→                "no_shares": max(0.0, long_no - (size_shares if side == Side.BUY else 0.0)),
  1385→                "avg_entry_price": avg_entry_price,
  1386→                "total_invested_before": total_invested,
  1387→                "realised_pnl_before": realised_pnl,
  1388→            },
  1389→            "note": "dry_run=true — no order submitted",
  1390→        }
  1391→
  1392→    # ── Risk gate (same path as /api/trade) ──
  1393→    provisional = Order(
  1394→        order_id="close-pre-check",
  1395→        token_id=token_id,
  1396→        side=side,
  1397→        price=close_price,
  1398→        size=size_shares,
  1399→        strategy=strategy or "manual_close",
  1400→        paper=settings.paper_trade,
  1401→    )

  1403→    if not allowed:
  1404→        await store.log_event(f"⚠ Risk block [close]: {reason}")
  1405→        raise HTTPException(status_code=400, detail=f"Risk rejection: {reason}")
  1406→
  1407→    # ── Submit the marketable close order ──
  1408→    args = OrderArgs(
  1409→        token_id=token_id,
  1410→        price=close_price,
  1411→        side=side,
  1412→        size=size_shares,
  1413→        # FOK = fill-or-kill: a marketable close should either fill completely
  1414→        # at the quoted top-of-book price or be rejected (no partial leaves).
  1415→        order_type="FOK",
  1416→    )
  1417→
  1418→    if settings.paper_trade:
  1419→        order = await paper_sim.create_order(args, strategy=strategy or "manual_close")
  1420→    else:
  1421→        from core.clob_client import clob_client
  1422→        order = await clob_client.create_order(args)
  1423→        if order:
  1424→            order_obj = Order(
  1425→                order_id=order.get("orderID") or order.get("order_id", f"close-{token_id[:8]}"),
  1426→                token_id=token_id,
  1427→                side=side,
  1428→                price=close_price,
  1429→                size=size_shares,
  1430→                strategy=strategy or "manual_close",
  1431→                paper=False,
  1432→            )
  1433→            await store.add_order(order_obj)
  1434→            order = order_obj
  1435→
  1436→    if not order:
  1437→        raise HTTPException(
  1438→            status_code=400,
  1439→            detail="failed to submit close order — exchange rejected or paper_sim error",

  1443→    try:
  1444→        await audit_logger.log_event(
  1445→            category="trading",
  1446→            event_type="position_close",

  1448→                f" Marketable close: {side.value} {size_shares:.4f} @ {close_price:.4f} "
  1449→                f"(${notional_usdc:.2f}) [est P&L ${estimated_pnl:+.2f}]"

  1451→            token_id=token_id,
  1452→            slug=slug,
  1453→            pnl=estimated_pnl,
  1454→            strategy=strategy or "manual_close",

  1470→        "price": round(close_price, 4),
  1471→        "size_shares": round(size_shares, 4),
  1472→        "notional_usdc": round(notional_usdc, 2),
  1473→        "estimated_pnl": round(estimated_pnl, 2),
  1474→        "best_bid": round(float(book.best_bid), 4) if book.best_bid is not None else None,
  1475→        "best_ask": round(float(book.best_ask), 4) if book.best_ask is not None else None,

  1664→    """Feed a resolved ground-truth outcome into the online SGD learner.
  1665→
  1666→    1. Backfills outcome labels in both DB backends (TimescaleDB + SQLite).
  1667→    2. Fetches the most recent stored feature vector for this token.
  1668→    3. Calls ml_model.update() to incrementally train the SGD online learner.
  1669→    """
  1670→    from core.timescale_db import timescale_db
  1671→
  1672→    # Step 1: persist outcome label
  1673→    updated = timescale_db.mark_resolved_outcomes(token_id, resolved_yes=resolved_yes)
  1674→
  1675→    # Step 2: fetch the most recent feature vector for this token
  1676→    import json as _json
  1677→    import sqlite3
  1678→
  1679→    import numpy as np
  1680→    features = None
  1681→    try:
  1682→        with sqlite3.connect(timescale_db._sqlite_path) as conn:
  1683→            row = conn.execute(
  1684→                "SELECT features_json FROM ml_feature_store WHERE token_id = ? ORDER BY timestamp DESC LIMIT 1;",
  1685→                (token_id,),
  1686→            ).fetchone()
  1687→        if row:
  1688→            features = np.array(_json.loads(row[0]), dtype=np.float32)
  1689→    except Exception as e:
  1690→        log.warning("[api/ml/learn] Could not fetch feature vector for %s: %s", token_id, e)
  1691→
  1692→    # Step 3: online update
  1693→    if features is not None:
  1694→        await asyncio.to_thread(ml_model.update, features, resolved_yes)
  1695→        await store.log_event(
  1696→            f"🧠 Online ML update: {token_id[:12]} outcome={'YES' if resolved_yes else 'NO'} "
  1697→            f"(update #{ml_model._n_updates})"
  1698→        )
  1699→    else:
  1700→        await store.log_event(
  1701→            f"🧠 ML label recorded for {token_id[:12]} (no feature vector to update — outcome stored)"
  1702→        )
  1703→
  1704→    return {
  1705→        "status": "updated",
  1706→        "token_id": token_id,
  1707→        "resolved_yes": resolved_yes,

  1826→@app.post("/api/arbitrage/execute", tags=["arbitrage"])
  1827→async def execute_arbitrage(req: ArbitrageExecuteRequest):
  1828→    """
  1829→    Execute a dual-leg Dutch-book arbitrage. Both legs pass the same risk gate
  1830→    and are hard-capped by the per-market ceiling. Live execution is only
  1831→    possible for real token ids; synthetic complementary legs are reported
  1832→    but not transmitted to the exchange.
  1833→    """
  1834→    from core.clob_client import clob_client
  1835→    from risk.manager import MAX_POSITION_PER_MARKET
  1836→
  1837→    size_usdc = min(float(req.size_usdc), float(MAX_POSITION_PER_MARKET))
  1838→    if size_usdc <= 0:
  1839→        raise HTTPException(status_code=400, detail="size_usdc must be positive")
  1840→
  1841→    results = []
  1842→    legs = [
  1843→        ("yes", req.token_id_yes),
  1844→        ("no", req.token_id_no),
  1845→    ]
  1846→    for leg, token_id in legs:
  1847→        book = store.order_books.get(token_id)
  1848→        price = book.best_ask if book and book.best_ask else 0.50
  1849→        shares = size_usdc / max(price, 0.01)
  1850→
  1851→        provisional = Order(
  1852→            order_id=f"arb-{leg}-pre",
  1853→            token_id=token_id,
  1854→            side=Side.BUY,
  1855→            price=price,
  1856→            size=shares,
  1857→            strategy="arb_scanner",
  1858→            paper=settings.paper_trade,
  1859→        )
  1860→        allowed, reason = await risk_manager.check_order(provisional)
  1861→        if not allowed:
  1862→            results.append({"leg": leg, "token_id": token_id, "status": "REJECTED", "reason": reason})
  1863→            continue
  1864→
  1865→        if settings.paper_trade:
  1866→            order = await paper_sim.create_order(
  1867→                OrderArgs(token_id=token_id, price=price, side=Side.BUY, size=shares),
  1868→                strategy="arb_scanner",
  1869→            )
  1870→            status = "PLACED_PAPER"
  1871→        else:
  1872→            if token_id.endswith("_no"):
  1873→                results.append({"leg": leg, "token_id": token_id, "status": "SKIPPED", "reason": "synthetic complementary token — not transmissible"})
  1874→                continue
  1875→            resp = await clob_client.create_order(
  1876→                OrderArgs(token_id=token_id, price=price, side=Side.BUY, size=shares)
  1877→            )
  1878→            if resp is None:
  1879→                results.append({"leg": leg, "token_id": token_id, "status": "FAILED", "reason": "exchange rejected order"})
  1880→                continue
  1881→            order = Order(
  1882→                order_id=resp.get("orderID") or resp.get("order_id", "unknown"),
  1883→                token_id=token_id,
  1884→                side=Side.BUY,
  1885→                price=price,
  1886→                size=shares,
  1887→                strategy="arb_scanner",
  1888→                paper=False,
  1889→            )
  1890→            await store.add_order(order)
  1891→            status = "PLACED_LIVE"
  1892→
  1893→        results.append({"leg": leg, "token_id": token_id, "status": status, "order_id": order.order_id})
  1894→
  1895→    slug = store.market_slugs.get(req.token_id_yes, req.token_id_yes[:12])
  1896→    await store.log_event(f"⚡ Manual arb execute: {slug} (${size_usdc:.2f}/leg): {[r['status'] for r in results]}")
  1897→    return {"status": "processed", "size_usdc": size_usdc, "legs": results, "slug": slug}
