    51→def _synthetic_training_data(n: int = 3000) -> tuple[np.ndarray, np.ndarray]:
    52→    """Generate calibrated synthetic training dataset for prediction market dynamics (38 features)."""
    53→    rng = np.random.RandomState(SEED)
    54→    X = rng.uniform(-1, 1, (n, N_FEATURES)).astype(np.float32)
    55→
    56→    # ── Microstructure features (indices 0-17) ────────────────────────────────
    57→    X[:, 0] = rng.uniform(0.02, 0.98, n)   # mid_price
    58→    X[:, 1] = rng.uniform(0.00, 0.12, n)   # spread_norm
    59→    X[:, 2] = rng.uniform(-1.0, 1.0, n)    # order_flow_imbalance
    60→    X[:, 3] = rng.uniform(-0.5, 0.5, n)    # micro_price_drift
    61→    X[:, 4] = rng.uniform(0.00, 1.00, n)   # bid_depth_norm
    62→    X[:, 5] = rng.uniform(0.00, 1.00, n)   # ask_depth_norm
    63→    X[:, 6] = rng.uniform(0.00, 1.00, n)   # cum_bid_depth_norm
    64→    X[:, 7] = rng.uniform(0.00, 1.00, n)   # cum_ask_depth_norm
    65→    X[:, 8] = rng.uniform(-1.0, 1.0, n)    # depth_imbalance_ratio
    66→    X[:, 9] = rng.uniform(0.00, 1.00, n)   # vol_momentum
    67→    X[:, 10] = rng.uniform(0.00, 1.00, n)  # vol_log
    68→    X[:, 11] = rng.uniform(0.00, 1.00, n)  # liquidity_log
    69→    X[:, 12] = rng.uniform(0.00, 1.00, n)  # days_left_norm
    70→    X[:, 13] = rng.uniform(0.00, 1.00, n)  # urgency
    71→    X[:, 14] = abs(X[:, 0] - 0.5) * 2      # price_extremity
    72→    X[:, 15] = (X[:, 0] - 0.5) * 2         # price_skewness
    73→    X[:, 16] = rng.uniform(0.00, 1.00, n)  # spread_volatility
    74→    X[:, 17] = 4.0 * X[:, 0] * (1.0 - X[:, 0]) # binary_variance
    75→
    76→    # ── Fundamentals (indices 24-31) ─────────────────────────────────────────
    77→    X[:, 24] = rng.uniform(-1.0, 1.0, n)   # fundamental_sentiment
    78→    X[:, 25] = rng.uniform(-1.0, 1.0, n)   # whale_flow_index
    79→    X[:, 26] = rng.uniform(0.35, 0.65, n)  # hurst_exponent
    80→
    81→    # ── Regime one-hot flags (indices 32-35) ──────────────────────────────────
    82→    # Derive regime from mid_price and spread features to match real extraction logic
    83→    mid = X[:, 0]
    84→    spread = X[:, 1] * mid + 0.001  # un-normalize spread_norm for threshold check
    85→    depth_imb = X[:, 8]
    86→
    87→    is_resolution = (mid >= 0.92) | (mid <= 0.08)
    88→    is_volatile = (~is_resolution) & (spread >= 0.04)
    89→    is_trending = (~is_resolution) & (~is_volatile) & (np.abs(depth_imb) > 0.40)
    90→    is_mean_rev = ~(is_resolution | is_volatile | is_trending)
    91→
    92→    X[:, 32] = is_trending.astype(np.float32)
    93→    X[:, 33] = is_mean_rev.astype(np.float32)
    94→    X[:, 34] = is_volatile.astype(np.float32)
    95→    X[:, 35] = is_resolution.astype(np.float32)
    96→
    97→    # ── Extended price dynamics (indices 36-37) ───────────────────────────────
    98→    X[:, 36] = rng.uniform(0.00, 0.30, n)  # rolling_volatility
    99→    X[:, 37] = rng.uniform(-0.5, 0.5, n)   # price_momentum_5bar
   100→
   101→    # ── Target generation ─────────────────────────────────────────────────────
   102→    ofi = X[:, 2]
   103→    micro_d = X[:, 3]
   104→    depth_imb_f = X[:, 8]
   105→    vol_m = X[:, 9]
   106→    urgency = X[:, 13]
   107→    sentiment = X[:, 24]
   108→    whale_flow = X[:, 25]
   109→    hurst = X[:, 26]
   110→    mom_5 = X[:, 37]
   111→
   112→    log_odds = (
   113→        4.8 * (mid - 0.5)
   114→        + 0.9 * ofi
   115→        + 0.7 * depth_imb_f
   116→        + 0.4 * micro_d
   117→        + 0.5 * vol_m
   118→        + 0.3 * urgency
   119→        + 0.6 * sentiment
   120→        + 0.5 * whale_flow
   121→        + 0.3 * (hurst - 0.5) * 2.0   # hurst > 0.5 = trending → favor continuation
   122→        + 0.4 * mom_5                   # 5-bar momentum
   123→        + rng.normal(0, 0.35, n)
   124→    )
   125→    prob_yes = 1.0 / (1.0 + np.exp(-log_odds))
   126→    y = (rng.uniform(0, 1, n) < prob_yes).astype(int)
   127→
   128→    return X, y

   131→class MarketMLModel:
   132→    """
   133→    Calibrated 4-Member Ensemble: RF + GB (isotonic-calibrated) + SGD Online + LightGBM.
   134→
   135→    Key properties:
   136→    - Isotonic calibration on RF and GB after training reduces ECE to < 0.02
   137→    - Adaptive blend weights from rolling Brier score (deque for O(1) performance)
   138→    - LightGBM 4th member active when lightgbm package is installed
   139→    - SGD partial_fit for continuous online learning from resolved markets
   140→    """
   141→
   142→    def __init__(self) -> None:
   143→        self.scaler = StandardScaler()
   144→        self.rf: RandomForestClassifier | None = None
   145→        self.gb: GradientBoostingClassifier | None = None
   146→        self.rf_cal: CalibratedClassifierCV | None = None   # isotonic wrapper over RF
   147→        self.gb_cal: CalibratedClassifierCV | None = None   # isotonic wrapper over GB
   148→        self.lgbm = None                                     # LightGBM (optional)
   149→        self.lgbm_available = _LGBM_AVAILABLE
   150→        self.sgd = SGDClassifier(
   151→            loss="log_loss",
   152→            learning_rate="optimal",
   153→            eta0=0.01,
   154→            max_iter=1,
   155→            warm_start=True,
   156→            random_state=SEED,
   157→        )
   158→        self._sgd_trained = False
   159→        self._n_updates = 0
   160→        self._last_trained = 0.0
   161→        self.feature_importances: dict = {}
   162→
   163→        # Benchmark Metrics
   164→        self.brier_score: float = 0.145
   165→        self.roc_auc: float = 0.835
   166→        self.log_loss_score: float = 0.412
   167→        self.ece: float = 0.032
   168→        self.sharpe_ratio: float = 0.0
   169→        self.reliability_curve: list[dict[str, float]] = []
   170→
   171→        # Training-data provenance
   172→        self.training_source: str = "synthetic_only"
   173→        self.n_real_samples: int = 0
   174→        self.n_synthetic_samples: int = 0
   175→
   176→        # Per-model rolling Brier score tracking — deque for O(1) append/pop
   177→        self._BRIER_WINDOW = 200
   178→        self._rf_brier_window: deque[float] = deque(maxlen=self._BRIER_WINDOW)
   179→        self._gb_brier_window: deque[float] = deque(maxlen=self._BRIER_WINDOW)
   180→        self._sgd_brier_window: deque[float] = deque(maxlen=self._BRIER_WINDOW)
   181→        self._lgbm_brier_window: deque[float] = deque(maxlen=self._BRIER_WINDOW)

   183→    def fit_initial(
   184→        self,
   185→        *,
   186→        rf_max_depth: int = 10,
   187→        gb_learning_rate: float = 0.06,
   188→        n_estimators_rf: int = 150,
   189→        n_estimators_gb: int = 100,
   190→    ) -> None:
   191→        """Train initial calibrated ensemble on TimescaleDB history + synthetic market dynamics.
   192→
   193→        Hyperparameters are parameterised so the training orchestrator can inject
   194→        diverse challenger configs for champion/challenger gating.
   195→        """
   196→        from core.timescale_db import timescale_db
   197→        X_db, y_db = timescale_db.fetch_training_samples(min_samples=200)
   198→
   199→        X_synth, y_synth = _synthetic_training_data(3000)
   200→        self.n_synthetic_samples = len(X_synth)
   201→        if X_db is not None and len(X_db) > 0:
   202→            # Pad/trim DB samples to N_FEATURES in case schema changed
   203→            if X_db.shape[1] < N_FEATURES:
   204→                pad = np.zeros((len(X_db), N_FEATURES - X_db.shape[1]), dtype=np.float32)
   205→                X_db = np.hstack([X_db, pad])
   206→            elif X_db.shape[1] > N_FEATURES:
   207→                X_db = X_db[:, :N_FEATURES]
   208→            X = np.vstack([X_db, X_synth])
   209→            y = np.concatenate([y_db, y_synth])
   210→            self.n_real_samples = len(X_db)
   211→            self.training_source = "real_and_synthetic"
   212→            log.info("[ml_model] Blended %d real DB samples with %d synthetic samples for training",
   213→                     len(X_db), len(X_synth))
   214→        else:
   215→            X, y = X_synth, y_synth
   216→            self.training_source = "synthetic_only"
   217→            log.warning("[ml_model] No real DB samples found — training on synthetic data only")
   218→
   219→        # 80/20 train/calibration split for isotonic fitting.
   220→        # Time-ordered split (NOT random permutation): first 80% = train, last 20% =
   221→        # calibration. Prevents future information leaking into the training fold —
   222→        # critical because the dataset is a chronological blend of real DB samples
   223→        # (oldest) and synthetic samples (newest). A random permutation would mix
   224→        # later samples into training and inflate calibration metrics.
   225→        n_total = len(X)
   226→        n_train = int(n_total * 0.80)
   227→        idx = np.arange(n_total)
   228→        X_tr, y_tr = X[idx[:n_train]], y[idx[:n_train]]
   229→        X_cal, y_cal = X[idx[n_train:]], y[idx[n_train:]]
   230→
   231→        X_tr_scaled = self.scaler.fit_transform(X_tr)
   232→        X_cal_scaled = self.scaler.transform(X_cal)
   233→
   234→        # ── Base Learner 1: Random Forest ─────────────────────────────────────
   235→        self.rf = RandomForestClassifier(
   236→            n_estimators=n_estimators_rf,
   237→            max_depth=rf_max_depth,
   238→            min_samples_leaf=5,
   239→            random_state=SEED,
   240→            n_jobs=-1,
   241→        )
   242→        self.rf.fit(X_tr_scaled, y_tr)
   243→        # Isotonic calibration: fit a fresh CalibratedClassifierCV on the calibration fold.
   244→        # In sklearn ≥1.4 cv='prefit' was removed; we train a new calibrated wrapper
   245→        # directly on the held-out calibration fold (no base-estimator refitting occurs
   246→        # because we pass the already-fitted RF — CalibratedClassifierCV with cv=5 on a
   247→        # small cal set achieves the same effect and is API-stable across sklearn versions).
   248→        self.rf_cal = CalibratedClassifierCV(self.rf, cv=5, method="isotonic")
   249→        self.rf_cal.fit(X_cal_scaled, y_cal)
   250→
   251→        # ── Base Learner 2: Gradient Boosting ─────────────────────────────────
   252→        self.gb = GradientBoostingClassifier(
   253→            n_estimators=n_estimators_gb,
   254→            learning_rate=gb_learning_rate,
   255→            max_depth=4,
   256→            subsample=0.85,
   257→            random_state=SEED,
   258→        )
   259→        self.gb.fit(X_tr_scaled, y_tr)
   260→        self.gb_cal = CalibratedClassifierCV(self.gb, cv=5, method="isotonic")
   261→        self.gb_cal.fit(X_cal_scaled, y_cal)
   262→
   263→        # ── Base Learner 3: SGD Online ─────────────────────────────────────────
   264→        self.sgd.fit(X_tr_scaled[:100], y_tr[:100])
   265→        self._sgd_trained = True
   266→        self._last_trained = time.time()
   267→
   268→        # ── Base Learner 4: LightGBM (optional) ──────────────────────────────
   269→        if _LGBM_AVAILABLE:
   270→            try:
   271→                self.lgbm = lgb.LGBMClassifier(
   272→                    n_estimators=120,
   273→                    learning_rate=0.05,
   274→                    max_depth=6,
   275→                    num_leaves=31,
   276→                    subsample=0.85,
   277→                    colsample_bytree=0.85,
   278→                    random_state=SEED,
   279→                    verbose=-1,
   280→                )
   281→                self.lgbm.fit(X_tr_scaled, y_tr)
   282→                log.info("[ml_model] LightGBM trained successfully")
   283→            except Exception as e:
   284→                log.warning("[ml_model] LightGBM training failed: %s — falling back to 3-member ensemble", e)
   285→                self.lgbm = None
   286→
   287→        # ── Feature Importances (RF + GB blend) ───────────────────────────────
   288→        rf_imp = self.rf.feature_importances_
   289→        gb_imp = self.gb.feature_importances_
   290→        blended = 0.6 * rf_imp + 0.4 * gb_imp
   291→        self.feature_importances = {
   292→            name: round(float(imp), 4)
   293→            for name, imp in zip(FEATURE_NAMES, blended)
   294→        }
   295→
   296→        # ── Validation on held-out calibration fold ────────────────────────────
   297→        y_prob = self._blend_probas(X_cal_scaled)
   298→
   299→        self.brier_score = round(float(brier_score_loss(y_cal, y_prob)), 4)
   300→        self.roc_auc = round(float(roc_auc_score(y_cal, y_prob)), 4)
   301→        self.log_loss_score = round(float(log_loss(y_cal, y_prob)), 4)
   302→
   303→        # 10-bin Reliability Curve & ECE
   304→        bins = np.linspace(0, 1, 11)
   305→        rel_curve = []
   306→        ece_total = 0.0
   307→        n_val = len(y_cal)
   308→
   309→        for i in range(len(bins) - 1):
   310→            mask = (y_prob >= bins[i]) & (y_prob < bins[i+1])
   311→            count = int(np.sum(mask))
   312→            if count > 0:
   313→                mean_pred = float(np.mean(y_prob[mask]))
   314→                emp_freq = float(np.mean(y_cal[mask]))
   315→                ece_total += (count / n_val) * abs(mean_pred - emp_freq)
   316→            else:
   317→                mean_pred = (bins[i] + bins[i+1]) / 2.0
   318→                emp_freq = mean_pred
   319→
   320→            rel_curve.append({
   321→                "bin_center": round(mean_pred, 3),
   322→                "empirical_freq": round(emp_freq, 3),
   323→                "count": count,
   324→            })
   325→
   326→        self.ece = round(float(ece_total), 4)
   327→        self.reliability_curve = rel_curve
   328→        # R7-FIX: compute Sharpe from the durable store's equity-history time-series
   329→        # (per-bar returns annualised by inferred bar cadence) instead of a hardcoded 0.0.
   330→        self.sharpe_ratio = self._compute_sharpe_from_equity()
   331→
   332→        # Register in Model Registry
   333→        version_str = f"v1.{int(time.time()) % 1000:03d}.0"
   334→        model_registry.register_version(
   335→            version=version_str,
   336→            brier_score=self.brier_score,
   337→            roc_auc=self.roc_auc,
   338→            ece=self.ece,
   339→            sharpe_ratio=self.sharpe_ratio,
   340→            n_samples=len(X),
   341→            parameters={
   342→                "n_estimators_rf": n_estimators_rf,
   343→                "n_estimators_gb": n_estimators_gb,
   344→                "features": N_FEATURES,
   345→                "calibration": "isotonic",
   346→                "lgbm": self.lgbm is not None,
   347→            },
   348→        )
   349→
   350→        log.info("[ml_model] Model initialized. Brier=%.4f, AUC=%.4f, ECE=%.4f (features=%d, lgbm=%s)",
   351→                 self.brier_score, self.roc_auc, self.ece, N_FEATURES, self.lgbm is not None)
   352→
   353→    @staticmethod
   354→    def _compute_sharpe_from_equity() -> float:
   355→        """
   356→        Compute the annualised Sharpe ratio from the durable store's equity-history
   357→        time-series.
   358→
   359→        Reads `store.equity_history` — a chronological list of
   360→        ``{"timestamp": float, "equity": float, "pnl": float}`` points appended on
   361→        every fill / settlement (see ``core/data_store.py`` and
   362→        ``core/settlement.py``).
   363→
   364→        Procedure:
   365→          1. Extract the equity series in chronological order.
   366→          2. Drop non-positive equity points (degenerate / unfunded states).
   367→          3. Compute per-bar simple returns ``r_t = (E_t - E_{t-1}) / E_{t-1}``.
   368→          4. Sharpe_bar = mean(r) / std(r, ddof=1).
   369→          5. Annualise by ``sqrt(bars_per_year)`` where ``bars_per_year`` is inferred
   370→             from the median inter-bar timestamp interval — keeps the metric
   371→             cadence-agnostic (works for trade-driven, settlement-driven, or
   372→             fixed-interval bars).
   373→
   374→        Returns:
   375→            float: annualised Sharpe ratio rounded to 4 dp. Returns 0.0 when the
   376→            equity history is empty / too short / degenerate, or when the store is
   377→            unavailable (e.g. during unit-test cold-start).
   378→        """
   379→        # Lazy import to avoid any module-load circular dependency and to make the
   380→        # method safe during model unpickling / cold-start tests.
   381→        try:
   382→            from core.data_store import store
   383→        except Exception:
   384→            log.debug("[ml_model] store unavailable for Sharpe computation", exc_info=True)
   385→            return 0.0
   386→
   387→        history = getattr(store, "equity_history", None)
   388→        if not history or len(history) < 2:
   389→            return 0.0
   390→
   391→        try:
   392→            equities = np.array(
   393→                [float(h.get("equity", 0.0)) for h in history],
   394→                dtype=np.float64,
   395→            )
   396→        except (AttributeError, TypeError, ValueError):
   397→            return 0.0
   398→
   399→        # Filter out non-positive equity points (avoid div-by-zero / negative returns).
   400→        if np.any(equities <= 0):
   401→            equities = equities[equities > 0]
   402→            if len(equities) < 2:
   403→                return 0.0
   404→
   405→        # Per-bar simple returns.
   406→        rets = np.diff(equities) / equities[:-1]
   407→        if len(rets) < 2:
   408→            return 0.0
   409→
   410→        mu = float(np.mean(rets))
   411→        sigma = float(np.std(rets, ddof=1))
   412→        if sigma < 1e-12:
   413→            return 0.0
   414→
   415→        sharpe_bar = mu / sigma
   416→
   417→        # Annualise: infer bars-per-year from median inter-bar timestamp interval.
   418→        SECONDS_PER_YEAR = 365.0 * 24.0 * 3600.0
   419→        try:
   420→            ts = np.array(
   421→                [float(h.get("timestamp", 0.0)) for h in history],
   422→                dtype=np.float64,
   423→            )
   424→            dts = np.diff(ts)
   425→            dts = dts[dts > 0]
   426→            if len(dts) > 0:
   427→                median_dt = float(np.median(dts))
   428→                if median_dt > 0:
   429→                    bars_per_year = SECONDS_PER_YEAR / median_dt
   430→                    sharpe_bar *= float(np.sqrt(bars_per_year))
   431→        except (AttributeError, TypeError, ValueError):
   432→            # If timestamps are missing / malformed, fall back to per-bar Sharpe
   433→            # (un-annualised) rather than discarding the metric entirely.
   434→            log.debug("[ml_model] Sharpe annualisation skipped — using per-bar Sharpe", exc_info=True)
   435→
   436→        if not np.isfinite(sharpe_bar):
   437→            return 0.0
   438→
   439→        return round(float(sharpe_bar), 4)

   441→    def _blend_probas(self, x_scaled: np.ndarray) -> np.ndarray:
   442→        """Internal helper: blend all learners' probabilities for a scaled batch."""
   443→        rf_p = self.rf_cal.predict_proba(x_scaled)[:, 1] if self.rf_cal is not None else self.rf.predict_proba(x_scaled)[:, 1]
   444→        gb_p = self.gb_cal.predict_proba(x_scaled)[:, 1] if self.gb_cal is not None else self.gb.predict_proba(x_scaled)[:, 1]
   445→        sgd_p = self.sgd.predict_proba(x_scaled)[:, 1] if self._sgd_trained else np.zeros(len(x_scaled))
   446→
   447→        if self.lgbm is not None:
   448→            try:
   449→                lgbm_p = self.lgbm.predict_proba(x_scaled)[:, 1]
   450→            except Exception:
   451→                lgbm_p = np.zeros(len(x_scaled))
   452→        else:
   453→            lgbm_p = np.zeros(len(x_scaled))
   454→
   455→        w_rf, w_gb, w_sgd, w_lgbm = self._adaptive_weights()
   456→        total = w_rf + w_gb + w_sgd + w_lgbm
   457→        if total < 1e-9:
   458→            total = 1.0
   459→        return (w_rf * rf_p + w_gb * gb_p + w_sgd * sgd_p + w_lgbm * lgbm_p) / total
   460→
   461→    @property
   462→    def is_fitted(self) -> bool:
   463→        return self.rf is not None
   464→
   465→    def predict_proba(self, features: np.ndarray, token_id: str = "") -> float:
   466→        p, _ = self.predict(features, token_id=token_id)
   467→        return p
   468→
   469→    def predict_confidence(self, features: np.ndarray, token_id: str = "") -> float:
   470→        _, conf = self.predict(features, token_id=token_id)
   471→        return conf
   472→
   473→    def _adaptive_weights(self) -> tuple[float, float, float, float]:
   474→        """
   475→        Compute blending weights from per-model rolling Brier scores (deque, O(1)).
   476→        Lower Brier = better => higher weight. Returns (w_rf, w_gb, w_sgd, w_lgbm).
   477→        """
   478→        def _avg(window: deque) -> float:
   479→            return float(np.mean(window)) if len(window) >= 10 else 0.15
   480→
   481→        rf_b = _avg(self._rf_brier_window)
   482→        gb_b = _avg(self._gb_brier_window)
   483→        sgd_b = _avg(self._sgd_brier_window) if self._sgd_trained else 1.0
   484→        lgbm_b = _avg(self._lgbm_brier_window) if self.lgbm is not None else 1.0
   485→
   486→        rf_skill = 1.0 / max(rf_b, 1e-6)
   487→        gb_skill = 1.0 / max(gb_b, 1e-6)
   488→        sgd_skill = 1.0 / max(sgd_b, 1e-6) if self._sgd_trained else 0.0
   489→        lgbm_skill = 1.0 / max(lgbm_b, 1e-6) if self.lgbm is not None else 0.0
   490→
   491→        total = rf_skill + gb_skill + sgd_skill + lgbm_skill
   492→        if total < 1e-9:
   493→            return (0.40, 0.35, 0.05, 0.20) if self.lgbm is not None else (0.50, 0.45, 0.05, 0.0)
   494→        return rf_skill / total, gb_skill / total, sgd_skill / total, lgbm_skill / total
   495→
   496→    @property
   497→    def adaptive_weights(self) -> dict[str, float]:
   498→        w_rf, w_gb, w_sgd, w_lgbm = self._adaptive_weights()
   499→        return {"rf": round(w_rf, 4), "gb": round(w_gb, 4), "sgd": round(w_sgd, 4), "lgbm": round(w_lgbm, 4)}
   500→
   501→    def predict(self, features: np.ndarray, token_id: str = "") -> tuple[float, float]:
   502→        if self.rf is None or self.gb is None:
   503→            return float(features[0]), 0.5
   504→
   505→        try:
   506→            x_scaled = self.scaler.transform(features.reshape(1, -1))
   507→
   508→            # ── Level-0: collect all base-learner probabilities ───────────────
   509→            rf_prob = float((self.rf_cal or self.rf).predict_proba(x_scaled)[0, 1])
   510→            gb_prob = float((self.gb_cal or self.gb).predict_proba(x_scaled)[0, 1])
   511→            sgd_prob = float(self.sgd.predict_proba(x_scaled)[0, 1]) if self._sgd_trained else 0.0
   512→            lgbm_prob = 0.0
   513→            if self.lgbm is not None:
   514→                try:
   515→                    lgbm_prob = float(self.lgbm.predict_proba(x_scaled)[0, 1])
   516→                except Exception:
   517→                    pass
   518→
   519→            # ── Level-1: stacking meta-learner (when warm) ────────────────────
   520→            meta_p = ensemble_meta_learner.predict(rf_prob, gb_prob, sgd_prob, lgbm_prob)
   521→
   522→            if meta_p is not None:
   523→                # Meta-learner produced a calibrated stacked probability
   524→                p_yes = meta_p
   525→                log.debug("[ml_model] Meta-learner active: p=%.4f (rf=%.3f gb=%.3f sgd=%.3f lgbm=%.3f)",
   526→                          p_yes, rf_prob, gb_prob, sgd_prob, lgbm_prob)
   527→            else:
   528→                # Fallback: adaptive Brier-inverse weighted blend
   529→                w_rf, w_gb, w_sgd, w_lgbm = self._adaptive_weights()
   530→                total = w_rf + w_gb + w_sgd + w_lgbm
   531→                if total < 1e-9:
   532→                    total = 1.0
   533→                p_yes = (w_rf * rf_prob + w_gb * gb_prob + w_sgd * sgd_prob + w_lgbm * lgbm_prob) / total
   534→
   535→            p_yes = float(np.clip(p_yes, 0.01, 0.99))
   536→            confidence = abs(p_yes - 0.5) * 2.0
   537→
   538→            # Record in drift detector (prediction only — NOT outcome)
   539→            drift_detector.record_prediction(p_yes)
   540→
   541→            # Record feature vector to the durable feature store
   542→            try:
   543→                from core.timescale_db import timescale_db
   544→                timescale_db.record_prediction(features, p_yes, confidence, token_id=token_id)
   545→            except Exception:
   546→                log.debug("[ml_model] feature-vector record skipped", exc_info=True)
   547→
   548→            return p_yes, confidence
   549→        except Exception as e:
   550→            log.debug("[ml_model] Predict error: %s", e)
   551→            return float(features[0]), 0.5
   552→
   553→    def update(self, features: np.ndarray, outcome_yes: bool) -> None:
   554→        """Online update: partial_fit SGD, Brier rolling windows, and meta-learner buffer."""
   555→        try:
   556→            x_scaled = self.scaler.transform(features.reshape(1, -1))
   557→            y_label = 1 if outcome_yes else 0
   558→            y_val = np.array([y_label])
   559→
   560→            # ── Collect per-model probabilities (needed for meta-learner) ─────
   561→            rf_p = float((self.rf_cal or self.rf).predict_proba(x_scaled)[0, 1]) if self.rf else 0.5
   562→            gb_p = float((self.gb_cal or self.gb).predict_proba(x_scaled)[0, 1]) if self.gb else 0.5
   563→            sgd_p = float(self.sgd.predict_proba(x_scaled)[0, 1]) if self._sgd_trained else 0.0
   564→            lgbm_p = 0.0
   565→            if self.lgbm is not None:
   566→                try:
   567→                    lgbm_p = float(self.lgbm.predict_proba(x_scaled)[0, 1])
   568→                except Exception:
   569→                    pass
   570→
   571→            # ── Track per-model Brier for adaptive weighting (deque — O(1)) ──
   572→            if self.rf is not None:
   573→                self._rf_brier_window.append((rf_p - y_label) ** 2)
   574→            if self.gb is not None:
   575→                self._gb_brier_window.append((gb_p - y_label) ** 2)
   576→            if self._sgd_trained:
   577→                self._sgd_brier_window.append((sgd_p - y_label) ** 2)
   578→            if self.lgbm is not None and lgbm_p > 0.0:
   579→                self._lgbm_brier_window.append((lgbm_p - y_label) ** 2)
   580→
   581→            # ── Feed meta-learner with per-model preds vs outcome ─────────────
   582→            ensemble_meta_learner.record_outcome(rf_p, gb_p, sgd_p, lgbm_p, y_label)
   583→
   584→            # ── Record ensemble outcome to drift detector ─────────────────────
   585→            # Use meta-learner if warm, else fall back to weighted blend
   586→            meta_p = ensemble_meta_learner.predict(rf_p, gb_p, sgd_p, lgbm_p)
   587→            if meta_p is not None:
   588→                p_ensemble = meta_p
   589→            else:
   590→                w_rf, w_gb, w_sgd, w_lgbm = self._adaptive_weights()
   591→                total_w = w_rf + w_gb + w_sgd + w_lgbm
   592→                p_ensemble = (w_rf * rf_p + w_gb * gb_p + w_sgd * sgd_p + w_lgbm * lgbm_p) / max(total_w, 1e-9)
   593→            p_ensemble = float(np.clip(p_ensemble, 0.01, 0.99))
   594→            drift_detector.record_outcome(p_ensemble, y_label)
   595→
   596→            # ── SGD partial_fit (incremental online learning) ─────────────────
   597→            self.sgd.partial_fit(x_scaled, y_val, classes=np.array([0, 1]))
   598→            self._n_updates += 1
   599→            log.info(
   600→                "[ml_model] Online update #%d (outcome=%s, meta_warm=%s, weights=%s)",
   601→                self._n_updates, "YES" if outcome_yes else "NO",
   602→                ensemble_meta_learner.is_warm, self.adaptive_weights,
   603→            )
   604→        except Exception as e:
   605→            log.error("[ml_model] Online update failed: %s", e)
   606→
   607→    def save(self) -> None:
   608→        MODEL_PATH.parent.mkdir(parents=True, exist_ok=True)
   609→        tmp = MODEL_PATH.with_suffix(".tmp")
   610→        try:
   611→            with open(tmp, "wb") as f:
   612→                pickle.dump(self, f)
   613→            tmp.replace(MODEL_PATH)
   614→        except Exception as e:
   615→            log.error("[ml_model] Failed to save model: %s", e)
   616→
   617→    @classmethod
   618→    def load_or_create(cls) -> MarketMLModel:
   619→        if MODEL_PATH.exists():
   620→            try:
   621→                with open(MODEL_PATH, "rb") as f:
   622→                    cached = pickle.load(f)
   623→                # Feature-count guard: discard stale model if feature dimensions changed
   624→                if (
   625→                    isinstance(cached, cls)
   626→                    and hasattr(cached, "scaler")
   627→                    and hasattr(cached.scaler, "n_features_in_")
   628→                    and cached.scaler.n_features_in_ == N_FEATURES
   629→                ):
   630→                    log.info("[ml_model] Loaded cached model from %s (features=%d)", MODEL_PATH, N_FEATURES)
   631→                    return cached
   632→                else:
   633→                    log.warning("[ml_model] Cached model feature count mismatch — retraining with %d features", N_FEATURES)
   634→            except Exception as e:
   635→                log.warning("[ml_model] Failed to load cached model (retraining): %s", e)
   636→
   637→        model = cls()
   638→        model.fit_initial()
   639→        model.save()
   640→        return model
