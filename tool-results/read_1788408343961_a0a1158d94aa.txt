    57→def _valid_token(authorization: str | None) -> bool:
    58→    if not settings.api_token:
    59→        return False
    60→    scheme, _, creds = (authorization or "").partition(" ")
    61→    if scheme.lower() != "bearer" or not creds:
    62→        return False
    63→    return hmac.compare_digest(creds, settings.api_token)

    66→# ── WebSocket Connection Manager ──────────────────────────────────────────────
    67→
    68→class ConnectionManager:
    69→    def __init__(self) -> None:
    70→        self.active: list[WebSocket] = []
    71→
    72→    async def connect(self, ws: WebSocket) -> None:
    73→        await ws.accept()
    74→        self.active.append(ws)
    75→        log.info("WS client connected — total %d", len(self.active))
    76→
    77→    def disconnect(self, ws: WebSocket) -> None:
    78→        if ws in self.active:
    79→            self.active.remove(ws)
    80→
    81→    async def broadcast(self, data: dict) -> None:
    82→        dead = []
    83→        payload = json.dumps(data, default=str)
    84→        for ws in list(self.active):
    85→            try:
    86→                await ws.send_text(payload)
    87→            except Exception:
    88→                dead.append(ws)
    89→        for ws in dead:
    90→            if ws in self.active:
    91→                self.active.remove(ws)

   100→async def _seed_markets(limit: int = 60) -> list[str]:
   101→    token_ids: list[str] = []
   102→    try:
   103→        markets = await gamma_client.get_markets(active=True, limit=limit, order="volume24hr")
   104→        for mkt in markets:
   105→            slug = mkt.get("slug") or mkt.get("groupItemTitle") or ""
   106→            ids = gamma_client.extract_token_ids(mkt)
   107→            for tid in ids:
   108→                token_ids.append(tid)
   109→                if slug:
   110→                    store.market_slugs[tid] = slug
   111→                # Ingest into vector store
   112→                vector_store.add_market(tid, mkt)
   113→
   114→        vector_store.build_index()
   115→        vector_store.save_to_disk()
   116→
   117→        if token_ids:
   118→            unique_ids = list(dict.fromkeys(token_ids))
   119→            log.info("Seeded %d unique tokens from %d Gamma markets", len(unique_ids), len(markets))
   120→            await store.log_event(f"📊 Seeded {len(unique_ids)} market tokens & built Vector Embeddings index")
   121→            return unique_ids
   122→    except Exception as e:
   123→        log.error("Market seeding failed: %s", e)
   124→        await store.log_event(f"⚠ Market seed failed: {e}")
   125→    return token_ids

   435→@app.middleware("http")
   436→async def enforce_api_auth(request: Request, call_next):
   437→    """Fail-closed bearer-token auth on every route except the liveness probe and OPTIONS preflight."""
   438→    if request.method == "OPTIONS":
   439→        return await call_next(request)
   440→
   441→    origin = request.headers.get("origin")
   442→    cors_allowed = bool(origin and (origin in settings.cors_origin_list or "*" in settings.cors_origin_list or not settings.cors_origin_list))
   443→    cors_headers = {}
   444→    if origin and cors_allowed:
   445→        cors_headers["access-control-allow-origin"] = origin
   446→        cors_headers["access-control-allow-credentials"] = "true"
   447→        cors_headers["access-control-allow-headers"] = "Authorization, Content-Type"
   448→        cors_headers["access-control-allow-methods"] = "GET, POST, PUT, DELETE, OPTIONS"
   449→
   450→    path = request.url.path
   451→    if path in PUBLIC_PATHS or path.startswith("/api/health"):
   452→        response = await call_next(request)
   453→        if origin and cors_allowed:
   454→            response.headers["access-control-allow-origin"] = origin
   455→            response.headers["access-control-allow-credentials"] = "true"
   456→        return response
   457→
   458→    if not settings.api_token:
   459→        return JSONResponse(
   460→            status_code=503,
   461→            content={"detail": "API authentication not configured — set API_TOKEN in .env", "code": "AUTH_NOT_CONFIGURED"},
   462→            headers=cors_headers,
   463→        )
   464→    if not _valid_token(request.headers.get("authorization")):
   465→        return JSONResponse(
   466→            status_code=401,
   467→            content={"detail": "Unauthorized — missing or invalid API token"},
   468→            headers=cors_headers,
   469→        )
   470→
   471→    response = await call_next(request)
   472→    if origin and cors_allowed:
   473→        response.headers["access-control-allow-origin"] = origin
   474→        response.headers["access-control-allow-credentials"] = "true"
   475→    return response

   547→    trades = store.trades
   548→    total_trades = len(trades)
   549→    # Only trades that actually closed a position (pnl != 0) determine win rate.
   550→    closed_trades = [t for t in trades if t.pnl != 0]
   551→    winning_trades = sum(1 for t in closed_trades if t.pnl > 0)
   552→    losing_trades = sum(1 for t in closed_trades if t.pnl < 0)
   553→    win_rate = (winning_trades / len(closed_trades)) if closed_trades else 0.0
   554→    total_vol = sum(t.price * t.size for t in trades)
   555→
   556→    # Wilson 95% confidence interval for the win rate (sample-size honest).
   557→    n = len(closed_trades)
   558→    z = 1.96
   559→    ci_low = ci_high = None
   560→    if n > 0:
   561→        p = win_rate
   562→        denom = 1 + z * z / n
   563→        centre = (p + z * z / (2 * n)) / denom
   564→        half = z * ((p * (1 - p) / n + z * z / (4 * n * n)) ** 0.5) / denom
   565→        ci_low = max(0.0, centre - half)
   566→        ci_high = min(1.0, centre + half)
   567→
   568→    wins = sum(t.pnl for t in closed_trades if t.pnl > 0)
   569→    losses = sum(t.pnl for t in closed_trades if t.pnl < 0)
   570→    profit_factor = wins / max(1e-9, -losses) if losses else (None if not wins else float("inf"))
   571→
   572→    # Unrealized P&L: mark open positions to the live order-book mid when present.
   573→    unrealized_pnl = 0.0
   574→    for p in store.positions.values():
   575→        if p.current_exposure <= 0.001:
   576→            continue
   577→        book = store.order_books.get(p.token_id)
   578→        mark = book.mid if book and book.mid is not None else None
   579→        if mark is None:
   580→            mark = p.avg_entry_price  # cost-basis mark (no live quote)
   581→        unrealized_pnl += (mark - p.avg_entry_price) * p.yes_shares
   582→        unrealized_pnl += ((1.0 - mark) - p.avg_entry_price) * p.no_shares
   583→
   584→    realized_pnl = store.daily_pnl
   585→    net_pnl = realized_pnl + unrealized_pnl
   586→    equity = store.paper_balance
   587→    max_drawdown_dollars = max(0.0, store.peak_equity - equity)
   588→    max_drawdown_pct = (max_drawdown_dollars / store.peak_equity) if store.peak_equity > 0 else 0.0
   589→
   590→    exp = compute_exposure()
   591→    deployable = float(MAX_DEPLOYABLE_CAPITAL)
   592→    risk_utilization = min(1.0, exp["maximum_remaining_loss"] / deployable) if deployable > 0 else 0.0
   593→
   594→    # Data freshness: seconds since the newest tracked order-book update.
   595→    books = store.order_books.values()
   596→    freshness = max((time.time() - b.updated_at for b in books), default=0.0)
   597→
   598→    return {
   599→        "equity": round(equity, 2),
   600→        "realized_pnl": round(realized_pnl, 2),
   601→        "unrealized_pnl": round(unrealized_pnl, 2),
   602→        "net_pnl": round(net_pnl, 2),
   603→        "total_trades": total_trades,
   604→        "winning_trades": winning_trades,
   605→        "losing_trades": losing_trades,
   606→        "closed_trades": len(closed_trades),
   607→        "open_trades": total_trades - len(closed_trades),
   608→        "win_rate": round(win_rate, 4),
   609→        "win_rate_ci_low": round(ci_low, 4) if ci_low is not None else None,
   610→        "win_rate_ci_high": round(ci_high, 4) if ci_high is not None else None,
   611→        "profit_factor": round(profit_factor, 2) if isinstance(profit_factor, float) else profit_factor,
   612→        "max_drawdown_dollars": round(max_drawdown_dollars, 2),
   613→        "max_drawdown_pct": round(max_drawdown_pct, 4),
   614→        "total_volume_usdc": round(total_vol, 2),
   615→        "open_exposure": round(exp["maximum_remaining_loss"], 2),
   616→        "open_position_count": exp["open_position_count"],
   617→        "pending_order_capital": round(exp["reserved_for_pending_orders"], 2),
   618→        "risk_utilization": round(risk_utilization, 4),
   619→        "mode": settings.trading_mode,
   620→        "data_freshness_seconds": round(freshness, 1),
   621→        "peak_equity": round(store.peak_equity, 2),
   622→        "active_strategies": list(strategy_registry.get_active_instances().keys()),
   623→    }

   684→# ── Market OHLCV & Historical Candlestick Data ────────────────────────────────
   685→
   686→@app.get("/api/history/ohlcv/{token_id}", tags=["markets"])
   687→async def get_market_ohlcv(token_id: str, resolution: str = "5m", count: int = 40):
   688→    """Return OHLCV candlestick bars for visual charting.
   689→
   690→    Priority:
   691→      1. Real candles from TimescaleDB continuous aggregates (market.price_candle_*)
   692→         when TimescaleDB is connected and rows exist for this token — labeled synthetic=False.
   693→      2. Seeded random-walk anchored to live mid when no stored candles exist —
   694→         explicitly labeled synthetic=True so callers always know the data source.
   695→    """
   696→    book = await store.get_order_book(token_id)
   697→    mid = (book.mid if book else 0.5) or 0.5
   698→    slug = store.market_slugs.get(token_id, token_id[:14])
   699→    step_sec = 60 if resolution == "1m" else 300 if resolution == "5m" else 3600
   700→    view_map = {"1m": "market.price_candle_1m", "5m": "market.price_candle_5m", "1h": "market.price_candle_1h"}
   701→    pg_view = view_map.get(resolution, "market.price_candle_5m")
   702→
   703→    # ── Attempt real TimescaleDB continuous-aggregate candles ─────────────────
   704→    from core.timescale_db import timescale_db
   705→    if timescale_db._is_postgres and timescale_db._pool:
   706→        try:
   707→            async with timescale_db._pool.acquire() as conn:
   708→                rows = await conn.fetch(
   709→                    f"""
   710→                    SELECT bucket, open, high, low, close, vwap, tick_count
   711→                    FROM {pg_view}
   712→                    WHERE token_id = $1
   713→                    ORDER BY bucket DESC
   714→                    LIMIT $2;
   715→                    """,
   716→                    token_id,
   717→                    count,
   718→                )
   719→            if rows:
   720→                bars = [
   721→                    {
   722→                        "timestamp": float(r["bucket"].timestamp()),
   723→                        "open": round(float(r["open"]), 4),
   724→                        "high": round(float(r["high"]), 4),
   725→                        "low": round(float(r["low"]), 4),
   726→                        "close": round(float(r["close"]), 4),
   727→                        "volume": float(r["tick_count"]),
   728→                        "vwap": round(float(r["vwap"]), 4) if r["vwap"] is not None else None,
   729→                    }
   730→                    for r in reversed(rows)
   731→                ]
   732→                return {
   733→                    "token_id": token_id,
   734→                    "slug": slug,
   735→                    "resolution": resolution,
   736→                    "bars": bars,
   737→                    "count": len(bars),
   738→                    "synthetic": False,
   739→                    "source": pg_view,
   740→                    "data_age_seconds": round(time.time() - bars[-1]["timestamp"], 1) if bars else None,
   741→                }
   742→        except Exception as e:
   743→            log.debug("[ohlcv] TimescaleDB candle query failed for %s: %s", token_id, e)
   744→
   745→    # ── Synthetic fallback: seeded random-walk anchored to live mid ───────────
   746→    rng = np.random.RandomState(abs(hash(token_id + resolution)) % (2**31))
   747→    now = time.time()
   748→
   749→    bars = []
   750→    curr_price = max(mid * (1.0 + rng.uniform(-0.06, 0.06)), 0.05)
   751→    for i in range(count):
   752→        ts = now - (count - i) * step_sec
   753→        drift = rng.uniform(-0.012, 0.012)
   754→        open_p = curr_price
   755→        close_p = max(min(open_p + drift, 0.98), 0.02)
   756→        high_p = max(open_p, close_p) + abs(rng.uniform(0.001, 0.008))
   757→        low_p = min(open_p, close_p) - abs(rng.uniform(0.001, 0.008))
   758→        vol = float(rng.uniform(500, 15000))
   759→        bars.append({
   760→            "timestamp": ts,
   761→            "open": round(open_p, 4),
   762→            "high": round(high_p, 4),
   763→            "low": round(low_p, 4),
   764→            "close": round(close_p, 4),
   765→            "volume": round(vol, 1),
   766→        })
   767→        curr_price = close_p
   768→
   769→    if bars:
   770→        bars[-1]["close"] = round(mid, 4)
   771→
   772→    return {
   773→        "token_id": token_id,
   774→        "slug": slug,
   775→        "resolution": resolution,
   776→        "bars": bars,
   777→        "count": len(bars),
   778→        "synthetic": True,
   779→        "synthetic_kind": "seeded_random_walk",
   780→        "disclaimer": "Synthetic bars anchored to live mid — no stored history for this token yet",
   781→    }

  1189→    """Feed a resolved ground-truth outcome into the online SGD learner.
  1190→
  1191→    1. Backfills outcome labels in both DB backends (TimescaleDB + SQLite).
  1192→    2. Fetches the most recent stored feature vector for this token.
  1193→    3. Calls ml_model.update() to incrementally train the SGD online learner.
  1194→    """

  1196→
  1197→    # Step 1: persist outcome label

  1199→

  1204→    import numpy as np

  1230→        "status": "updated",
  1231→        "token_id": token_id,

  1347→    token_id_no: str
  1348→    size_usdc: float

  1350→
  1351→@app.post("/api/arbitrage/execute", tags=["arbitrage"])
  1352→async def execute_arbitrage(req: ArbitrageExecuteRequest):
  1353→    """
  1354→    Execute a dual-leg Dutch-book arbitrage. Both legs pass the same risk gate
  1355→    and are hard-capped by the per-market ceiling. Live execution is only
  1356→    possible for real token ids; synthetic complementary legs are reported
  1357→    but not transmitted to the exchange.
  1358→    """
  1359→    from core.clob_client import clob_client
  1360→    from risk.manager import MAX_POSITION_PER_MARKET
  1361→
  1362→    size_usdc = min(float(req.size_usdc), float(MAX_POSITION_PER_MARKET))
  1363→    if size_usdc <= 0:
  1364→        raise HTTPException(status_code=400, detail="size_usdc must be positive")
  1365→
  1366→    results = []
  1367→    legs = [
  1368→        ("yes", req.token_id_yes),
  1369→        ("no", req.token_id_no),
  1370→    ]
  1371→    for leg, token_id in legs:
  1372→        book = store.order_books.get(token_id)
  1373→        price = book.best_ask if book and book.best_ask else 0.50
  1374→        shares = size_usdc / max(price, 0.01)
  1375→
  1376→        provisional = Order(
  1377→            order_id=f"arb-{leg}-pre",
  1378→            token_id=token_id,
  1379→            side=Side.BUY,
  1380→            price=price,
  1381→            size=shares,
  1382→            strategy="arb_scanner",
  1383→            paper=settings.paper_trade,
  1384→        )
  1385→        allowed, reason = await risk_manager.check_order(provisional)
  1386→        if not allowed:
  1387→            results.append({"leg": leg, "token_id": token_id, "status": "REJECTED", "reason": reason})
  1388→            continue
  1389→
  1390→        if settings.paper_trade:
  1391→            order = await paper_sim.create_order(
  1392→                OrderArgs(token_id=token_id, price=price, side=Side.BUY, size=shares),
  1393→                strategy="arb_scanner",
  1394→            )
  1395→            status = "PLACED_PAPER"
  1396→        else:
  1397→            if token_id.endswith("_no"):
  1398→                results.append({"leg": leg, "token_id": token_id, "status": "SKIPPED", "reason": "synthetic complementary token — not transmissible"})
  1399→                continue
  1400→            resp = await clob_client.create_order(
  1401→                OrderArgs(token_id=token_id, price=price, side=Side.BUY, size=shares)
  1402→            )
  1403→            if resp is None:
  1404→                results.append({"leg": leg, "token_id": token_id, "status": "FAILED", "reason": "exchange rejected order"})
  1405→                continue
  1406→            order = Order(
  1407→                order_id=resp.get("orderID") or resp.get("order_id", "unknown"),
  1408→                token_id=token_id,
  1409→                side=Side.BUY,
  1410→                price=price,
  1411→                size=shares,
  1412→                strategy="arb_scanner",
  1413→                paper=False,
  1414→            )
  1415→            await store.add_order(order)
  1416→            status = "PLACED_LIVE"
  1417→
  1418→        results.append({"leg": leg, "token_id": token_id, "status": status, "order_id": order.order_id})
  1419→
  1420→    slug = store.market_slugs.get(req.token_id_yes, req.token_id_yes[:12])
  1421→    await store.log_event(f"⚡ Manual arb execute: {slug} (${size_usdc:.2f}/leg): {[r['status'] for r in results]}")
  1422→    return {"status": "processed", "size_usdc": size_usdc, "legs": results, "slug": slug}
