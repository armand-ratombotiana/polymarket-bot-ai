    18→    18→  │ 4  │ win_rate_over_50pct           │ closed_positions win_rate > 0.50         │

    22→    22→  │ 7  │ drift_healthy                 │ drift_detector.drift_status == HEALTHY  │
    23→    23→  │ 8  │ kill_switch_tested            │ audit trail has ≥1 kill_switch_activated │

   107→   107→async def check_live_readiness() -> dict[str, Any]:
   108→   108→    """
   109→   109→    Run all 10 God Mode §82 staged checks and return a verdict dict::
   110→   110→
   111→   111→        {
   112→   112→            "passed": bool,                  # True only if ALL 10 checks pass
   113→   113→            "checks": [ {id, name, passed, severity, threshold, value, detail}, ... ],
   114→   114→            "passed_count": int,
   115→   115→            "total_count": int,
   116→   116→            "blocking_checks": [str, ...],   # ids of failed checks (empty if passed)
   117→   117→            "checked_at": float,             # epoch seconds
   118→   118→        }
   119→   119→
   120→   120→    Never raises — a check that throws records itself as failed (with the
   121→   121→    exception text in ``detail``) so the gate always returns a verdict. This
   122→   122→    is the live-trading safety contract: the gate must answer, even when a
   123→   123→    dependency is broken.
   124→   124→    """
   125→   125→    checks: list[dict[str, Any]] = []
   126→   126→    checks.append(await _check_paper_mode())
   127→   127→    checks.append(await _check_positive_expectancy())
   128→   128→    checks.append(await _check_max_drawdown())
   129→   129→    checks.append(await _check_win_rate())
   130→   130→    checks.append(await _check_closed_trades())
   131→   131→    checks.append(await _check_ml_real_data())
   132→   132→    checks.append(await _check_drift_healthy())
   133→   133→    checks.append(await _check_kill_switch_tested())
   134→   134→    checks.append(await _check_risk_limits())
   135→   135→    checks.append(await _check_api_credentials())
   136→   136→
   137→   137→    passed_count = sum(1 for c in checks if c.get("passed"))
   138→   138→    blocking = [c["id"] for c in checks if not c.get("passed")]
   139→   139→    return {
   140→   140→        "passed": len(blocking) == 0,
   141→   141→        "checks": checks,
   142→   142→        "passed_count": passed_count,
   143→   143→        "total_count": len(checks),
   144→   144→        "blocking_checks": blocking,
   145→   145→        "checked_at": time.time(),
   146→   146→    }

   221→   221→# Each returns a dict: {id, name, passed, severity, threshold, value, detail}.

   224→   224→
   225→   225→async def _check_paper_mode() -> dict[str, Any]:
   226→   226→    """Check #1: paper_mode ≥ 24h — current continuous paper session length."""
   227→   227→    name = "Paper mode soak ≥ 24h"
   228→   228→    threshold = f"trading_mode=='paper' AND session_age_s >= {PAPER_MODE_MIN_SECONDS:.0f}"
   229→   229→    try:
   230→   230→        from config import settings
   231→   231→        from core.data_store import store
   232→   232→
   233→   233→        mode = settings.trading_mode
   234→   234→        session_start = float(getattr(store, "session_start", time.time()))
   235→   235→        age_s = max(0.0, time.time() - session_start)
   236→   236→        in_paper = mode == "paper"
   237→   237→        passed = in_paper and age_s >= PAPER_MODE_MIN_SECONDS
   238→   238→        if passed:
   239→   239→            detail = f"paper session running {age_s/3600.0:.2f}h (≥24h)"
   240→   240→        elif not in_paper:
   241→   241→            detail = (
   242→   242→                f"not currently in paper mode (trading_mode={mode!r}); "
   243→   243→                f"the §82 gate requires a continuous ≥24h paper soak first. "
   244→   244→                f"Note: session_start resets on every process restart, so the "
   245→   245→                f"measured age reflects only the current continuous session."
   246→   246→            )
   247→   247→        else:
   248→   248→            detail = (
   249→   249→                f"paper session only {age_s/3600.0:.2f}h old "
   250→   250→                f"(need ≥{PAPER_MODE_MIN_SECONDS/3600.0:.0f}h); "
   251→   251→                f"session_start resets on restart."
   252→   252→            )
   253→   253→        return {
   254→   254→            "id": CHECK_PAPER_MODE, "name": name, "passed": passed,
   255→   255→            "severity": SEVERITY_BLOCKING, "threshold": threshold,
   256→   256→            "value": {"trading_mode": mode, "session_age_seconds": round(age_s, 1)},
   257→   257→            "detail": detail,
   258→   258→        }
   259→   259→    except Exception as e:
   260→   260→        log.error("[live_safety_gate] paper_mode check failed: %s", e)
   261→   261→        return _failed(CHECK_PAPER_MODE, name, threshold, f"check raised: {e}")

   264→   264→async def _check_positive_expectancy() -> dict[str, Any]:
   265→   265→    """Check #2: positive expectancy — avg PnL per closed trade > 0."""
   266→   266→    name = "Positive expectancy (avg PnL > 0)"
   267→   267→    threshold = "closed_positions.avg_pnl > 0 (count > 0)"
   268→   268→    try:
   269→   269→        from core.closed_positions import closed_positions
   270→   270→
   271→   271→        stats = await closed_positions.get_closed_stats()
   272→   272→        count = int(stats.get("count", 0))
   273→   273→        avg_pnl = float(stats.get("avg_pnl", 0.0) or 0.0)
   274→   274→        # Expectancy = mean PnL per trade = (win_rate*avg_win)+(loss_rate*avg_loss).
   275→   275→        # stats.avg_pnl is exactly this (total_pnl/count), so the >0 test is the
   276→   276→        # canonical positive-expectancy gate.
   277→   277→        passed = count > 0 and avg_pnl > 0.0
   278→   278→        if passed:
   279→   279→            detail = f"avg_pnl=${avg_pnl:+.4f} across {count} closed trade(s) — positive expectancy"
   280→   280→        elif count == 0:
   281→   281→            detail = "no closed trades yet — expectancy undefined until ≥1 trade closes"
   282→   282→        else:
   283→   283→            detail = f"avg_pnl=${avg_pnl:+.4f} across {count} closed trade(s) — non-positive expectancy"
   284→   284→        return {
   285→   285→            "id": CHECK_POSITIVE_EXPECTANCY, "name": name, "passed": passed,
   286→   286→            "severity": SEVERITY_BLOCKING, "threshold": threshold,
   287→   287→            "value": {"count": count, "avg_pnl": round(avg_pnl, 4)},
   288→   288→            "detail": detail,
   289→   289→        }
   290→   290→    except Exception as e:
   291→   291→        log.error("[live_safety_gate] positive_expectancy check failed: %s", e)
   292→   292→        return _failed(CHECK_POSITIVE_EXPECTANCY, name, threshold, f"check raised: {e}")

   377→   377→async def _check_ml_real_data() -> dict[str, Any]:
   378→   378→    """Check #6: ML model trained on real data (not synthetic-only)."""
   379→   379→    name = "ML trained on real data"
   380→   380→    threshold = "ml_model.is_fitted AND training_source contains 'real' AND n_real_samples > 0"
   381→   381→    try:
   382→   382→        from ml.model import ml_model
   383→   383→
   384→   384→        is_fitted = bool(getattr(ml_model, "is_fitted", False))
   385→   385→        training_source = str(getattr(ml_model, "training_source", "synthetic_only"))
   386→   386→        n_real = int(getattr(ml_model, "n_real_samples", 0) or 0)
   387→   387→        # ``fit_initial`` sets training_source="real_and_synthetic" when the
   388→   388→        # TimescaleDB history had ≥200 real samples; "synthetic_only" otherwise.
   389→   389→        has_real = "real" in training_source and n_real > 0
   390→   390→        passed = is_fitted and has_real
   391→   391→        if passed:
   392→   392→            detail = (
   393→   393→                f"model fitted on {n_real} real + "
   394→   394→                f"{getattr(ml_model, 'n_synthetic_samples', 0)} synthetic samples "
   395→   395→                f"(source={training_source})"

   683→   683→def register_routes(app: Any) -> None:
   684→   684→    """
   685→   685→    Append live-safety-gate endpoints to a FastAPI app.
   686→   686→
   687→   687→    Endpoints (auth-protected by the caller's existing middleware):
   688→   688→
   689→   689→      GET /api/live/readiness
   690→   690→          Run all 10 God Mode §82 staged checks and return the verdict dict
   691→   691→          ({passed, checks, passed_count, total_count, blocking_checks,
   692→   692→          checked_at}). Never 500s — a check that throws records itself as
   693→   693→          failed.
   694→   694→
   695→   695→      POST /api/live/enable
   696→   696→          Request body: {confirm: bool, reason: str}
   697→   697→          Attempt to flip the bot into live trading mode. Requires
   698→   698→          ``confirm=true`` (defence against accidental double-click). Runs
   699→   699→          ``check_live_readiness`` first; if any check fails, returns HTTP 409
   700→   700→          with the blocking-check list and the full readiness payload. On
   701→   701→          success, flips the in-memory mode flags (live_trading_enabled=True,
   702→   702→          trading_mode="live", paper_trade=False) and logs an audit event.
   703→   703→
   704→   704→          NOTE: the in-memory flip is sufficient for ``check_order`` to start
   705→   705→          admitting live orders, but it does NOT persist across process
   706→   706→          restarts. For durable activation, set TRADING_MODE=live and
   707→   707→          LIVE_TRADING_ENABLED=true in .env and restart — the response
   708→   708→          payload carries this guidance.
   709→   709→    """
   710→   710→    from fastapi import HTTPException
   711→   711→
   712→   712→    @app.get("/api/live/readiness", tags=["live"])
   713→   713→    async def _live_readiness():
   714→   714→        """Run all 10 §82 staged checks; return {passed, checks, blocking_checks, ...}."""
   715→   715→        return await check_live_readiness()

   718→   718→    async def _enable_live(req: "EnableLiveRequest"):
   719→   719→        """Attempt to enable live trading. Refuses (HTTP 409) if any §82 check fails."""
   720→   720→        if not req.confirm:
   721→   721→            raise HTTPException(
   722→   722→                status_code=400,

   737→   737→                },
   738→   738→            )
   739→   739→        # All 10 checks passed — authorise live trading in-memory.
   740→   740→        try:

   743→   743→            from core.data_store import store

   751→   751→                details=(
   752→   752→                    f"Live trading enabled via §82 safety gate "
   753→   753→                    f"(reason={req.reason or 'operator request'}). "
   754→   754→                    f"In-memory flags flipped — set TRADING_MODE=live + "
   755→   755→                    f"LIVE_TRADING_ENABLED=true in .env and restart for "
   756→   756→                    f"durable activation."
   757→   757→                ),
   758→   758→            )
   759→   759→            await store.log_event(
   760→   760→                f"🔴 LIVE TRADING ENABLED via §82 safety gate — reason: {req.reason or 'operator request'}"
   761→   761→            )
