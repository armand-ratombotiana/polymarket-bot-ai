    61→def _valid_token(authorization: str | None) -> bool:
    62→    if not settings.api_token:
    63→        return False
    64→    scheme, _, creds = (authorization or "").partition(" ")
    65→    if scheme.lower() != "bearer" or not creds:
    66→        return False
    67→    return hmac.compare_digest(creds, settings.api_token)

    70→# ── WebSocket Connection Manager ──────────────────────────────────────────────
    71→
    72→class ConnectionManager:
    73→    def __init__(self) -> None:
    74→        self.active: list[WebSocket] = []
    75→
    76→    async def connect(self, ws: WebSocket) -> None:
    77→        await ws.accept()
    78→        self.active.append(ws)
    79→        log.info("WS client connected — total %d", len(self.active))
    80→
    81→    def disconnect(self, ws: WebSocket) -> None:
    82→        if ws in self.active:
    83→            self.active.remove(ws)
    84→
    85→    async def broadcast(self, data: dict) -> None:
    86→        dead = []
    87→        payload = json.dumps(data, default=str)
    88→        for ws in list(self.active):
    89→            try:
    90→                await ws.send_text(payload)
    91→            except Exception:
    92→                dead.append(ws)
    93→        for ws in dead:
    94→            if ws in self.active:
    95→                self.active.remove(ws)

   102→# ── Market Seeding & Vector Store Ingestion ───────────────────────────────────

   104→async def _seed_markets(limit: int = 60) -> list[str]:
   105→    token_ids: list[str] = []
   106→    try:
   107→        markets = await gamma_client.get_markets(active=True, limit=limit, order="volume24hr")
   108→        for mkt in markets:
   109→            slug = mkt.get("slug") or mkt.get("groupItemTitle") or ""
   110→            ids = gamma_client.extract_token_ids(mkt)
   111→            for tid in ids:
   112→                token_ids.append(tid)
   113→                if slug:
   114→                    store.market_slugs[tid] = slug
   115→                # Ingest into vector store
   116→                vector_store.add_market(tid, mkt)
   117→
   118→        vector_store.build_index()
   119→        vector_store.save_to_disk()
   120→
   121→        if token_ids:
   122→            unique_ids = list(dict.fromkeys(token_ids))
   123→            log.info("Seeded %d unique tokens from %d Gamma markets", len(unique_ids), len(markets))
   124→            await store.log_event(f"📊 Seeded {len(unique_ids)} market tokens & built Vector Embeddings index")
   125→            return unique_ids
   126→    except Exception as e:
   127→        log.error("Market seeding failed: %s", e)
   128→        await store.log_event(f"⚠ Market seed failed: {e}")
   129→    return token_ids

   336→async def _build_snapshot() -> dict:
   337→    from core.safety import kill_switch_file_exists
   338→    books = []
   339→    async with store._lock:
   340→        for tid, book in store.order_books.items():
   341→            books.append({
   342→                "token_id": tid,
   343→                "slug": store.market_slugs.get(tid, tid[:14]),
   344→                "best_bid": book.best_bid,
   345→                "best_ask": book.best_ask,
   346→                "mid": book.mid,
   347→                "spread": book.spread,
   348→                "updated_at": book.updated_at,
   349→            })
   350→
   351→    orders = []
   352→    for o in store.open_orders.values():
   353→        orders.append({
   354→            "order_id": o.order_id,
   355→            "token_id": o.token_id,
   356→            "slug": store.market_slugs.get(o.token_id, o.token_id[:14]),
   357→            "side": o.side.value,
   358→            "price": o.price,
   359→            "size": o.size,
   360→            "size_matched": o.size_matched,
   361→            "strategy": o.strategy,
   362→            "paper": o.paper,
   363→            "created_at": o.created_at,
   364→        })
   365→
   366→    positions = []
   367→    for tid, pos in store.positions.items():
   368→        if pos.yes_shares > 0 or pos.total_invested > 0 or pos.realised_pnl != 0:
   369→            # Mark-to-live-mid unrealized P&L for this position.
   370→            # Falls back to cost basis (avg_entry_price) when no live quote is tracked.
   371→            _book = store.order_books.get(tid)
   372→            current_price = _book.mid if _book and _book.mid is not None else pos.avg_entry_price
   373→            _pos_unrealized = 0.0
   374→            if pos.yes_shares > 0:
   375→                _pos_unrealized += (current_price - pos.avg_entry_price) * pos.yes_shares
   376→            if pos.no_shares > 0:
   377→                _pos_unrealized += ((1.0 - current_price) - pos.avg_entry_price) * pos.no_shares
   378→            positions.append({
   379→                "token_id": tid,
   380→                "slug": store.market_slugs.get(tid, tid[:14]),
   381→                "yes_shares": pos.yes_shares,
   382→                "avg_entry_price": pos.avg_entry_price,
   383→                "total_invested": pos.total_invested,
   384→                "realised_pnl": pos.realised_pnl,
   385→                "current_price": round(current_price, 4),
   386→                "unrealized_pnl": round(_pos_unrealized, 4),
   387→            })
   388→
   389→    trades = []
   390→    for t in store.trades[-50:]:
   391→        trades.append({
   392→            "trade_id": t.trade_id,
   393→            "token_id": t.token_id,
   394→            "slug": store.market_slugs.get(t.token_id, t.token_id[:14]),
   395→            "side": t.side.value,
   396→            "price": t.price,
   397→            "size": t.size,
   398→            "pnl": t.pnl,
   399→            "strategy": t.strategy,
   400→            "paper": t.paper,
   401→            "timestamp": t.timestamp,
   402→        })
   403→
   404→    events = await store.get_recent_events(50)
   405→    active_strats = list(strategy_registry.get_active_instances().keys())
   406→
   407→    return {
   408→        "type": "snapshot",

   410→        "mode": settings.trading_mode,
   411→        "kill_switch": store.kill_switch_active,
   412→        "kill_switch_durable": bool(kill_switch_file_exists()),
   413→        "observation_only": risk_manager.observation_only,
   414→        "observation_reason": risk_manager.observation_reason,

   416→        "paper_balance": paper_sim.virtual_balance if settings.paper_trade else None,

   418→        "order_books": books,
   419→        "open_orders": orders,
   420→        "positions": positions,
   421→        "recent_trades": trades,
   422→        "events": events,
   423→        "ml": {
   424→            "model_ready": ml_model.rf is not None,
   425→            "brier_score": ml_model.brier_score,
   426→            "roc_auc": ml_model.roc_auc,
   427→            "ece": ml_model.ece,

   453→_cors_origins = settings.cors_origin_list

   455→    CORSMiddleware,

   463→@app.middleware("http")
   464→async def enforce_api_auth(request: Request, call_next):
   465→    """Fail-closed bearer-token auth on every route except the liveness probe and OPTIONS preflight."""
   466→    if request.method == "OPTIONS":
   467→        return await call_next(request)
   468→
   469→    origin = request.headers.get("origin")
   470→    cors_allowed = bool(origin and (origin in settings.cors_origin_list or "*" in settings.cors_origin_list or not settings.cors_origin_list))
   471→    cors_headers = {}
   472→    if origin and cors_allowed:
   473→        cors_headers["access-control-allow-origin"] = origin
   474→        cors_headers["access-control-allow-credentials"] = "true"
   475→        cors_headers["access-control-allow-headers"] = "Authorization, Content-Type"
   476→        cors_headers["access-control-allow-methods"] = "GET, POST, PUT, DELETE, OPTIONS"
   477→
   478→    path = request.url.path
   479→    if path in PUBLIC_PATHS or path.startswith("/api/health"):
   480→        response = await call_next(request)
   481→        if origin and cors_allowed:
   482→            response.headers["access-control-allow-origin"] = origin
   483→            response.headers["access-control-allow-credentials"] = "true"
   484→        return response
   485→
   486→    if not settings.api_token:
   487→        return JSONResponse(
   488→            status_code=503,
   489→            content={"detail": "API authentication not configured — set API_TOKEN in .env", "code": "AUTH_NOT_CONFIGURED"},
   490→            headers=cors_headers,
   491→        )
   492→    if not _valid_token(request.headers.get("authorization")):
   493→        return JSONResponse(
   494→            status_code=401,
   495→            content={"detail": "Unauthorized — missing or invalid API token"},
   496→            headers=cors_headers,
   497→        )
   498→
   499→    response = await call_next(request)
   500→    if origin and cors_allowed:
   501→        response.headers["access-control-allow-origin"] = origin
   502→        response.headers["access-control-allow-credentials"] = "true"
   503→    return response

   573→@app.get("/api/analytics", tags=["system"])
   574→async def get_analytics():
   575→    trades = store.trades
   576→    total_trades = len(trades)
   577→    # Only trades that actually closed a position (pnl != 0) determine win rate.
   578→    closed_trades = [t for t in trades if t.pnl != 0]
   579→    winning_trades = sum(1 for t in closed_trades if t.pnl > 0)
   580→    losing_trades = sum(1 for t in closed_trades if t.pnl < 0)
   581→    win_rate = (winning_trades / len(closed_trades)) if closed_trades else 0.0
   582→    total_vol = sum(t.price * t.size for t in trades)
   583→
   584→    # Wilson 95% confidence interval for the win rate (sample-size honest).
   585→    n = len(closed_trades)
   586→    z = 1.96
   587→    ci_low = ci_high = None
   588→    if n > 0:
   589→        p = win_rate
   590→        denom = 1 + z * z / n
   591→        centre = (p + z * z / (2 * n)) / denom
   592→        half = z * ((p * (1 - p) / n + z * z / (4 * n * n)) ** 0.5) / denom
   593→        ci_low = max(0.0, centre - half)
   594→        ci_high = min(1.0, centre + half)
   595→
   596→    wins = sum(t.pnl for t in closed_trades if t.pnl > 0)
   597→    losses = sum(t.pnl for t in closed_trades if t.pnl < 0)
   598→    profit_factor = wins / max(1e-9, -losses) if losses else (None if not wins else float("inf"))
   599→
   600→    # Average win / loss and per-trade expectancy.
   601→    # Expectancy = (win_rate * avg_win) + (loss_rate * avg_loss) where avg_loss is negative.
   602→    avg_win = (wins / winning_trades) if winning_trades else 0.0
   603→    avg_loss = (losses / losing_trades) if losing_trades else 0.0  # negative or zero
   604→    loss_rate = 1.0 - win_rate
   605→    expectancy = (win_rate * avg_win) + (loss_rate * avg_loss)
   606→    sharpe_ratio = ml_model.sharpe_ratio
   607→
   608→    # Unrealized P&L: mark open positions to the live order-book mid when present.
   609→    unrealized_pnl = 0.0
   610→    for p in store.positions.values():
   611→        if p.current_exposure <= 0.001:
   612→            continue
   613→        book = store.order_books.get(p.token_id)
   614→        mark = book.mid if book and book.mid is not None else None
   615→        if mark is None:
   616→            mark = p.avg_entry_price  # cost-basis mark (no live quote)
   617→        unrealized_pnl += (mark - p.avg_entry_price) * p.yes_shares
   618→        unrealized_pnl += ((1.0 - mark) - p.avg_entry_price) * p.no_shares
   619→
   620→    realized_pnl = store.daily_pnl
   621→    net_pnl = realized_pnl + unrealized_pnl
   622→    equity = store.paper_balance
   623→    max_drawdown_dollars = max(0.0, store.peak_equity - equity)
   624→    max_drawdown_pct = (max_drawdown_dollars / store.peak_equity) if store.peak_equity > 0 else 0.0
   625→
   626→    exp = compute_exposure()
   627→    deployable = float(MAX_DEPLOYABLE_CAPITAL)
   628→    risk_utilization = min(1.0, exp["maximum_remaining_loss"] / deployable) if deployable > 0 else 0.0
   629→
   630→    # Data freshness: seconds since the newest tracked order-book update.
   631→    books = store.order_books.values()
   632→    freshness = max((time.time() - b.updated_at for b in books), default=0.0)
   633→
   634→    return {
   635→        "equity": round(equity, 2),
   636→        "realized_pnl": round(realized_pnl, 2),
   637→        "unrealized_pnl": round(unrealized_pnl, 2),
   638→        "net_pnl": round(net_pnl, 2),
   639→        "total_trades": total_trades,
   640→        "winning_trades": winning_trades,
   641→        "losing_trades": losing_trades,
   642→        "closed_trades": len(closed_trades),
   643→        "open_trades": total_trades - len(closed_trades),
   644→        "win_rate": round(win_rate, 4),
   645→        "win_rate_ci_low": round(ci_low, 4) if ci_low is not None else None,
   646→        "win_rate_ci_high": round(ci_high, 4) if ci_high is not None else None,
   647→        "profit_factor": round(profit_factor, 2) if isinstance(profit_factor, float) else profit_factor,
   648→        "avg_win": round(avg_win, 4),
   649→        "avg_loss": round(avg_loss, 4),
   650→        "expectancy": round(expectancy, 4),
   651→        "sharpe_ratio": round(sharpe_ratio, 4) if sharpe_ratio is not None else None,
   652→        "max_drawdown_dollars": round(max_drawdown_dollars, 2),
   653→        "max_drawdown_pct": round(max_drawdown_pct, 4),
   654→        "total_volume_usdc": round(total_vol, 2),
   655→        "open_exposure": round(exp["maximum_remaining_loss"], 2),
   656→        "open_position_count": exp["open_position_count"],
   657→        "pending_order_capital": round(exp["reserved_for_pending_orders"], 2),
   658→        "risk_utilization": round(risk_utilization, 4),
   659→        "mode": settings.trading_mode,
   660→        "data_freshness_seconds": round(freshness, 1),
   661→        "peak_equity": round(store.peak_equity, 2),
   662→        "active_strategies": list(strategy_registry.get_active_instances().keys()),
   663→    }

   723→
   724→@app.get("/api/ai/predict/{token_id}", tags=["ai"])
   725→async def get_ai_prediction(token_id: str):
   726→    """
   727→    Return the ML ensemble's directional view for a single YES token.
   728→
   729→    Reads market metadata from `market_discovery.catalog` (the universal
   730→    catalog maintained by `core/market_discovery.py`) — NOT from
   731→    `store.market_info` (that attribute does not exist on the DataStore).
   732→
   733→    Response:
   734→      * p_yes            — calibrated ensemble P(YES) ∈ (0.01, 0.99)
   735→      * confidence       — |p_yes - 0.5| * 2  ∈ [0, 1]
   736→      * market_mid       — live book mid price (or None if book is empty)
   737→      * edge             — p_yes - market_mid (positive ⇒ model is more
   738→                           bullish than the market; negative ⇒ bearish)
   739→      * edge_bps         — edge expressed in basis points
   740→      * recommended_action — BUY | SELL | HOLD (threshold-gated)
   741→      * thresholds       — the gates used so callers can audit the decision
   742→      * market           — the catalog metadata used to build the feature vector
   743→      * model_status     — whether the ensemble was actually trained
   744→
   745→    A HOLD recommendation does NOT imply no edge — it means the edge was
   746→    below the configured conviction threshold. Always inspect `edge` /
   747→    `edge_bps` before deciding to act.
   748→    """
   749→    # ── R10: pull metadata from market_discovery.catalog (NOT store.market_info) ──
   750→    from core.market_discovery import market_discovery
   751→
   752→    catalog_record = market_discovery.catalog.get(token_id)
   753→    if not catalog_record:
   754→        raise HTTPException(
   755→            status_code=404,
   756→            detail=f"token_id '{token_id}' not present in market_discovery.catalog — "
   757→                   "wait for the next catalog sync or call /api/markets/coverage",
   758→        )
   759→
   760→    # extract_features expects a dict with Gamma-style keys (volume24hr, volume,
   761→    # liquidity|liquidityNum, endDate|end_date_iso|endDateIso). The catalog
   762→    # record uses normalized snake_case keys, so bridge both shapes here.
   763→    market_for_features = dict(catalog_record)
   764→    market_for_features.setdefault(
   765→        "volume24hr", catalog_record.get("volume_24h", 0.0)
   766→    )
   767→    market_for_features.setdefault(
   768→        "volume", catalog_record.get("total_volume", 0.0)
   769→    )
   770→    # liquidity key already matches; bridge endDate aliases just in case.
   771→    if "endDate" not in market_for_features and catalog_record.get("end_date"):
   772→        market_for_features["endDate"] = catalog_record["end_date"]
   773→
   774→    # ── Live order book (needed for the microstructure feature vector) ──
   775→    book = await store.get_order_book(token_id)
   776→    if book is None:
   777→        # Hint the poller to prioritize this token, then surface a 502.
   778→        book_poller.prioritize_tokens([token_id])
   779→        raise HTTPException(
   780→            status_code=502,
   781→            detail=f"no live order book for token '{token_id}' — poller prioritized; retry shortly",
   782→        )
   783→
   784→    market_mid = book.mid
   785→    best_bid = book.best_bid
   786→    best_ask = book.best_ask
   787→
   788→    # ── Feature extraction + ensemble prediction ──
   789→    from ml.features import extract_features
   790→
   791→    features = extract_features(market_for_features, book)
   792→    if features is None:
   793→        raise HTTPException(
   794→            status_code=422,
   795→            detail=(
   796→                "feature extraction returned None — book mid is missing or "
   797→                "outside the tradeable (0.001, 0.999) band; cannot score"
   798→            ),
   799→        )
   800→
   801→    p_yes, confidence = ml_model.predict(features, token_id=token_id)
   802→    p_yes = float(p_yes)
   803→    confidence = float(confidence)
   804→
   805→    # ── Edge vs market mid ──
   806→    if market_mid is not None:
   807→        edge = float(p_yes) - float(market_mid)
   808→        edge_bps = round(edge * 10_000.0, 1)
   809→    else:
   810→        edge = None
   811→        edge_bps = None
   812→
   813→    # ── Recommendation thresholds ──
   814→    # Edge-based conviction gate (mirrors the spirit of strategies/signal_trader.py
   815→    # but expressed in terms of the model-vs-market edge rather than the absolute
   816→    # p_yes level, so a model that sees a 50/50 coin priced at 0.20 correctly
   817→    # surfaces a BUY rather than a HOLD just because |p_yes − 0.5| is small).
   818→    #
   819→    #   BUY  : edge ≥ +2ct (200 bps) AND confidence ≥ 0.10
   820→    #   SELL : edge ≤ −2ct (−200 bps) AND confidence ≥ 0.10
   821→    #   HOLD : otherwise (insufficient edge OR insufficient model conviction)
   822→    MIN_EDGE_CT = 0.02          # 2 cents
   823→    MIN_CONFIDENCE = 0.10
   824→
   825→    recommended_action = "HOLD"
   826→    if edge is None:
   827→        action_reason = "market mid unavailable — cannot compute edge"
   828→    elif confidence < MIN_CONFIDENCE:
   829→        recommended_action = "HOLD"
   830→        action_reason = (
   831→            f"confidence {confidence:.3f} < {MIN_CONFIDENCE:.2f} — model is too "
   832→            f"uncertain to act despite edge={edge*100:+.2f}ct"
   833→        )
   834→    elif edge >= MIN_EDGE_CT:
   835→        recommended_action = "BUY"
   836→        action_reason = (
   837→            f"edge=+{edge*100:.2f}ct ≥ +{MIN_EDGE_CT*100:.0f}ct AND "
   838→            f"confidence={confidence:.3f} ≥ {MIN_CONFIDENCE:.2f}"
   839→        )
   840→    elif edge <= -MIN_EDGE_CT:
   841→        recommended_action = "SELL"
   842→        action_reason = (
   843→            f"edge={edge*100:.2f}ct ≤ −{MIN_EDGE_CT*100:.0f}ct AND "
   844→            f"confidence={confidence:.3f} ≥ {MIN_CONFIDENCE:.2f}"
   845→        )
   846→    else:
   847→        recommended_action = "HOLD"
   848→        action_reason = (
   849→            f"|edge|={abs(edge)*100:.2f}ct below ±{MIN_EDGE_CT*100:.0f}ct "
   850→            f"conviction threshold"
   851→        )
   852→
   853→    # Compact market payload (avoid duplicating the full catalog record)
   854→    market_payload = {
   855→        "token_id": catalog_record.get("token_id"),
   856→        "event_id": catalog_record.get("event_id"),
   857→        "event_title": catalog_record.get("event_title"),
   858→        "question": catalog_record.get("question"),
   859→        "slug": catalog_record.get("slug") or store.market_slugs.get(token_id, ""),
   860→        "outcome": catalog_record.get("outcome"),
   861→        "category": catalog_record.get("category"),
   862→        "end_date": catalog_record.get("end_date"),
   863→        "status": catalog_record.get("status"),
   864→        "volume_24h": catalog_record.get("volume_24h"),
   865→        "total_volume": catalog_record.get("total_volume"),
   866→        "liquidity": catalog_record.get("liquidity"),
   867→        "last_synced": catalog_record.get("last_synced"),
   868→    }
   869→
   870→    return {
   871→        "token_id": token_id,
   872→        "p_yes": round(p_yes, 4),
   873→        "confidence": round(confidence, 4),
   874→        "market_mid": round(market_mid, 4) if market_mid is not None else None,
   875→        "best_bid": round(best_bid, 4) if best_bid is not None else None,
   876→        "best_ask": round(best_ask, 4) if best_ask is not None else None,
   877→        "spread": round(book.spread, 4) if book.spread is not None else None,
   878→        "edge": round(edge, 4) if edge is not None else None,
   879→        "edge_bps": edge_bps,
   880→        "recommended_action": recommended_action,
   881→        "action_reason": action_reason,
   882→        "thresholds": {
   883→            "min_edge_cents": MIN_EDGE_CT * 100.0,
   884→            "min_confidence": MIN_CONFIDENCE,
   885→        },
   886→        "model_status": {
   887→            "model_ready": ml_model.rf is not None,
   888→            "model_version": model_registry.active_version,
   889→            "brier_score": ml_model.brier_score,
   890→            "roc_auc": ml_model.roc_auc,
   891→            "ece": ml_model.ece,
   892→            "n_online_updates": ml_model._n_updates,
   893→        },
   894→        "market": market_payload,
   895→        "book_updated_at": book.updated_at,
   896→        "timestamp": time.time(),
   897→    }

   900→# ── Market OHLCV & Historical Candlestick Data ────────────────────────────────

   902→@app.get("/api/history/ohlcv/{token_id}", tags=["markets"])
   903→async def get_market_ohlcv(token_id: str, resolution: str = "5m", count: int = 40):
   904→    """Return OHLCV candlestick bars for visual charting.
   905→
   906→    Priority:
   907→      1. Real candles from TimescaleDB continuous aggregates (market.price_candle_*)
   908→         when TimescaleDB is connected and rows exist for this token — labeled synthetic=False.
   909→      2. Seeded random-walk anchored to live mid when no stored candles exist —
   910→         explicitly labeled synthetic=True so callers always know the data source.
   911→    """
   912→    book = await store.get_order_book(token_id)
   913→    mid = (book.mid if book else 0.5) or 0.5
   914→    slug = store.market_slugs.get(token_id, token_id[:14])
   915→    step_sec = 60 if resolution == "1m" else 300 if resolution == "5m" else 3600
   916→    view_map = {"1m": "market.price_candle_1m", "5m": "market.price_candle_5m", "1h": "market.price_candle_1h"}
   917→    pg_view = view_map.get(resolution, "market.price_candle_5m")
   918→
   919→    # ── Attempt real TimescaleDB continuous-aggregate candles ─────────────────
   920→    from core.timescale_db import timescale_db
   921→    if timescale_db._is_postgres and timescale_db._pool:
   922→        try:
   923→            async with timescale_db._pool.acquire() as conn:
   924→                rows = await conn.fetch(
   925→                    f"""
   926→                    SELECT bucket, open, high, low, close, vwap, tick_count
   927→                    FROM {pg_view}
   928→                    WHERE token_id = $1
   929→                    ORDER BY bucket DESC
   930→                    LIMIT $2;
   931→                    """,
   932→                    token_id,
   933→                    count,
   934→                )
   935→            if rows:
   936→                bars = [
   937→                    {
   938→                        "timestamp": float(r["bucket"].timestamp()),
   939→                        "open": round(float(r["open"]), 4),
   940→                        "high": round(float(r["high"]), 4),
   941→                        "low": round(float(r["low"]), 4),
   942→                        "close": round(float(r["close"]), 4),
   943→                        "volume": float(r["tick_count"]),
   944→                        "vwap": round(float(r["vwap"]), 4) if r["vwap"] is not None else None,
   945→                    }
   946→                    for r in reversed(rows)
   947→                ]
   948→                return {
   949→                    "token_id": token_id,
   950→                    "slug": slug,
   951→                    "resolution": resolution,
   952→                    "bars": bars,
   953→                    "count": len(bars),
   954→                    "synthetic": False,
   955→                    "source": pg_view,
   956→                    "data_age_seconds": round(time.time() - bars[-1]["timestamp"], 1) if bars else None,
   957→                }
   958→        except Exception as e:
   959→            log.debug("[ohlcv] TimescaleDB candle query failed for %s: %s", token_id, e)
   960→
   961→    # ── Synthetic fallback: seeded random-walk anchored to live mid ───────────
   962→    rng = np.random.RandomState(abs(hash(token_id + resolution)) % (2**31))
   963→    now = time.time()
   964→
   965→    bars = []
   966→    curr_price = max(mid * (1.0 + rng.uniform(-0.06, 0.06)), 0.05)
   967→    for i in range(count):
   968→        ts = now - (count - i) * step_sec
   969→        drift = rng.uniform(-0.012, 0.012)
   970→        open_p = curr_price
   971→        close_p = max(min(open_p + drift, 0.98), 0.02)
   972→        high_p = max(open_p, close_p) + abs(rng.uniform(0.001, 0.008))
   973→        low_p = min(open_p, close_p) - abs(rng.uniform(0.001, 0.008))
   974→        vol = float(rng.uniform(500, 15000))
   975→        bars.append({
   976→            "timestamp": ts,
   977→            "open": round(open_p, 4),
   978→            "high": round(high_p, 4),
   979→            "low": round(low_p, 4),
   980→            "close": round(close_p, 4),
   981→            "volume": round(vol, 1),
   982→        })
   983→        curr_price = close_p
   984→
   985→    if bars:
   986→        bars[-1]["close"] = round(mid, 4)
   987→
   988→    return {
   989→        "token_id": token_id,
   990→        "slug": slug,
   991→        "resolution": resolution,
   992→        "bars": bars,
   993→        "count": len(bars),
   994→        "synthetic": True,
   995→        "synthetic_kind": "seeded_random_walk",
   996→        "disclaimer": "Synthetic bars anchored to live mid — no stored history for this token yet",

  1235→class PositionCloseRequest(BaseModel):
  1236→    """Optional request body for one-click position close.
  1237→
  1238→    All fields are optional — when omitted, the endpoint closes the full
  1239→    position size at the current best bid (long YES → SELL) / best ask
  1240→    (long NO → BUY). Callers may override `max_size_shares` to scale out
  1241→    of a position incrementally, or set `dry_run=true` to preview the
  1242→    marketable close without submitting an order.
  1243→    """
  1244→    max_size_shares: float | None = Field(default=None, ge=0.0)
  1245→    dry_run: bool = False
  1246→
  1247→
  1248→@app.post("/api/positions/{token_id}/close", tags=["trading"])
  1249→async def close_position(token_id: str, req: PositionCloseRequest):
  1250→    """
  1251→    One-click marketable close of an open position.
  1252→
  1253→    Long YES positions are closed by submitting a SELL order at the current
  1254→    `best_bid` (a marketable limit — any resting bid ≥ best_bid is matched
  1255→    immediately). Long NO positions are closed by submitting a BUY order at
  1256→    the current `best_ask` (symmetric — covers the synthetic short).
  1257→
  1258→    Risk checks (`risk_manager.check_order`) are applied exactly as for a
  1259→    manual `/api/trade`, and the order flows through `paper_sim` when
  1260→    `settings.paper_trade` is true, or `clob_client` otherwise.
  1261→
  1262→    Use `dry_run=true` to preview the fill price, share count, and
  1263→    estimated realised P&L without submitting an order.
  1264→    """
  1265→    async with store._lock:
  1266→        pos = store.positions.get(token_id)
  1267→        # Snapshot the values we need before releasing the lock so we don't
  1268→        # hold store._lock across the (potentially blocking) order placement.
  1269→        if pos is None:
  1270→            long_yes = 0.0
  1271→            long_no = 0.0
  1272→            avg_entry_price = 0.0
  1273→            total_invested = 0.0
  1274→            realised_pnl = 0.0
  1275→            strategy = ""
  1276→        else:
  1277→            long_yes = float(pos.yes_shares)
  1278→            long_no = float(pos.no_shares)
  1279→            avg_entry_price = float(pos.avg_entry_price)
  1280→            total_invested = float(pos.total_invested)
  1281→            realised_pnl = float(pos.realised_pnl)
  1282→            strategy = pos.strategy or ""
  1283→
  1284→    if long_yes <= 0.0 and long_no <= 0.0:
  1285→        raise HTTPException(
  1286→            status_code=404,
  1287→            detail=f"no open position for token '{token_id}' — nothing to close",
  1288→        )
  1289→
  1290→    # ── Determine side, price, and size from the live book ──
  1291→    book = await store.get_order_book(token_id)
  1292→    if book is None or (book.best_bid is None and book.best_ask is None):
  1293→        # Treat a missing or fully empty book the same — surface 502 and hint
  1294→        # the poller to reprioritize this token so the next call succeeds.
  1295→        book_poller.prioritize_tokens([token_id])
  1296→        raise HTTPException(
  1297→            status_code=502,
  1298→            detail=f"no live order book for token '{token_id}' — poller prioritized; retry shortly",
  1299→        )
  1300→
  1301→    if long_yes > 0.0:
  1302→        # Close a long YES position: SELL into the bid ladder at best_bid.
  1303→        side = Side.SELL
  1304→        if book.best_bid is None:
  1305→            raise HTTPException(
  1306→                status_code=502,
  1307→                detail="cannot close long YES position — best_bid is empty (no resting bids)",
  1308→            )
  1309→        close_price = float(book.best_bid)
  1310→        available_shares = long_yes
  1311→    else:
  1312→        # Close a long NO position: BUY at best_ask (covers the synthetic short).
  1313→        side = Side.BUY
  1314→        if book.best_ask is None:
  1315→            raise HTTPException(
  1316→                status_code=502,
  1317→                detail="cannot close long NO position — best_ask is empty (no resting asks)",
  1318→            )
  1319→        close_price = float(book.best_ask)
  1320→        available_shares = long_no
  1321→
  1322→    # Cap the close size if the caller requested a partial scale-out.
  1323→    if req.max_size_shares is not None:
  1324→        size_shares = min(float(req.max_size_shares), available_shares)
  1325→    else:
  1326→        size_shares = available_shares
  1327→
  1328→    if size_shares <= 0.0:
  1329→        raise HTTPException(
  1330→            status_code=400,
  1331→            detail="requested close size resolves to 0 shares — nothing to close",
  1332→        )
  1333→
  1334→    # ── Estimated P&L preview (uses cost basis for long YES positions) ──
  1335→    estimated_pnl = 0.0
  1336→    if side == Side.SELL and avg_entry_price > 0.0:
  1337→        # P&L = (sell_price - avg_entry) * shares  — for YES side only.
  1338→        estimated_pnl = (close_price - avg_entry_price) * size_shares
  1339→    # (NO side P&L requires the YES/NO parity identity and is computed at fill
  1340→    #  time by store.record_fill / paper_sim._execute_fill; we don't fabricate
  1341→    #  a number here.)
  1342→
  1343→    notional_usdc = close_price * size_shares
  1344→    slug = store.market_slugs.get(token_id, token_id[:12])
  1345→
  1346→    if req.dry_run:
  1347→        return {
  1348→            "status": "dry_run",
  1349→            "token_id": token_id,
  1350→            "slug": slug,
  1351→            "side": side.value,
  1352→            "price": round(close_price, 4),
  1353→            "size_shares": round(size_shares, 4),
  1354→            "notional_usdc": round(notional_usdc, 2),
  1355→            "estimated_pnl": round(estimated_pnl, 2),
  1356→            "best_bid": round(float(book.best_bid), 4) if book.best_bid is not None else None,
  1357→            "best_ask": round(float(book.best_ask), 4) if book.best_ask is not None else None,
  1358→            "book_updated_at": book.updated_at,
  1359→            "paper_trade": settings.paper_trade,
  1360→            "remaining_position": {
  1361→                "yes_shares": max(0.0, long_yes - (size_shares if side == Side.SELL else 0.0)),
  1362→                "no_shares": max(0.0, long_no - (size_shares if side == Side.BUY else 0.0)),
  1363→                "avg_entry_price": avg_entry_price,
  1364→                "total_invested_before": total_invested,
  1365→                "realised_pnl_before": realised_pnl,
  1366→            },
  1367→            "note": "dry_run=true — no order submitted",
  1368→        }
  1369→
  1370→    # ── Risk gate (same path as /api/trade) ──
  1371→    provisional = Order(
  1372→        order_id="close-pre-check",
  1373→        token_id=token_id,
  1374→        side=side,
  1375→        price=close_price,
  1376→        size=size_shares,
  1377→        strategy=strategy or "manual_close",
  1378→        paper=settings.paper_trade,
  1379→    )
  1380→    allowed, reason = await risk_manager.check_order(provisional)
  1381→    if not allowed:
  1382→        await store.log_event(f"⚠ Risk block [close]: {reason}")
  1383→        raise HTTPException(status_code=400, detail=f"Risk rejection: {reason}")
  1384→
  1385→    # ── Submit the marketable close order ──
  1386→    args = OrderArgs(
  1387→        token_id=token_id,
  1388→        price=close_price,
  1389→        side=side,
  1390→        size=size_shares,
  1391→        # FOK = fill-or-kill: a marketable close should either fill completely
  1392→        # at the quoted top-of-book price or be rejected (no partial leaves).
  1393→        order_type="FOK",
  1394→    )
  1395→
  1396→    if settings.paper_trade:
  1397→        order = await paper_sim.create_order(args, strategy=strategy or "manual_close")
  1398→    else:
  1399→        from core.clob_client import clob_client
  1400→        order = await clob_client.create_order(args)
  1401→        if order:
  1402→            order_obj = Order(
  1403→                order_id=order.get("orderID") or order.get("order_id", f"close-{token_id[:8]}"),
  1404→                token_id=token_id,
  1405→                side=side,
  1406→                price=close_price,
  1407→                size=size_shares,
  1408→                strategy=strategy or "manual_close",
  1409→                paper=False,
  1410→            )
  1411→            await store.add_order(order_obj)
  1412→            order = order_obj
  1413→
  1414→    if not order:
  1415→        raise HTTPException(
  1416→            status_code=400,
  1417→            detail="failed to submit close order — exchange rejected or paper_sim error",
  1418→        )
  1419→
  1420→    # ── Audit trail entry ──
  1421→    try:
  1422→        await audit_logger.log_event(
  1423→            category="trading",
  1424→            event_type="position_close",
  1425→            details=(
  1426→                f" Marketable close: {side.value} {size_shares:.4f} @ {close_price:.4f} "
  1427→                f"(${notional_usdc:.2f}) [est P&L ${estimated_pnl:+.2f}]"
  1428→            ),
  1429→            token_id=token_id,
  1430→            slug=slug,
  1431→            pnl=estimated_pnl,
  1432→            strategy=strategy or "manual_close",
  1433→        )
  1434→    except Exception as e:

  1437→    await store.log_event(

  1440→    )
  1441→
  1442→    return {

  1444→        "token_id": token_id,

  1450→        "notional_usdc": round(notional_usdc, 2),

  1452→        "best_bid": round(float(book.best_bid), 4) if book.best_bid is not None else None,

  1455→        "paper_trade": settings.paper_trade,
  1456→        "remaining_position": {

  1640→@app.post("/api/ml/learn", tags=["ml"])

  1642→    """Feed a resolved ground-truth outcome into the online SGD learner.
  1643→
  1644→    1. Backfills outcome labels in both DB backends (TimescaleDB + SQLite).
  1645→    2. Fetches the most recent stored feature vector for this token.
  1646→    3. Calls ml_model.update() to incrementally train the SGD online learner.
  1647→    """
  1648→    from core.timescale_db import timescale_db
  1649→
  1650→    # Step 1: persist outcome label
  1651→    updated = timescale_db.mark_resolved_outcomes(token_id, resolved_yes=resolved_yes)
  1652→
  1653→    # Step 2: fetch the most recent feature vector for this token
  1654→    import json as _json
  1655→    import sqlite3
  1656→
  1657→    import numpy as np
  1658→    features = None
  1659→    try:
  1660→        with sqlite3.connect(timescale_db._sqlite_path) as conn:
  1661→            row = conn.execute(
  1662→                "SELECT features_json FROM ml_feature_store WHERE token_id = ? ORDER BY timestamp DESC LIMIT 1;",
  1663→                (token_id,),
  1664→            ).fetchone()
  1665→        if row:
  1666→            features = np.array(_json.loads(row[0]), dtype=np.float32)
  1667→    except Exception as e:
  1668→        log.warning("[api/ml/learn] Could not fetch feature vector for %s: %s", token_id, e)
  1669→
  1670→    # Step 3: online update
  1671→    if features is not None:
  1672→        await asyncio.to_thread(ml_model.update, features, resolved_yes)
  1673→        await store.log_event(
  1674→            f"🧠 Online ML update: {token_id[:12]} outcome={'YES' if resolved_yes else 'NO'} "
  1675→            f"(update #{ml_model._n_updates})"
  1676→        )
  1677→    else:
  1678→        await store.log_event(
  1679→            f"🧠 ML label recorded for {token_id[:12]} (no feature vector to update — outcome stored)"
  1680→        )
  1681→
  1682→    return {
  1683→        "status": "updated",
  1684→        "token_id": token_id,
  1685→        "resolved_yes": resolved_yes,

  1804→@app.post("/api/arbitrage/execute", tags=["arbitrage"])
  1805→async def execute_arbitrage(req: ArbitrageExecuteRequest):
  1806→    """
  1807→    Execute a dual-leg Dutch-book arbitrage. Both legs pass the same risk gate
  1808→    and are hard-capped by the per-market ceiling. Live execution is only
  1809→    possible for real token ids; synthetic complementary legs are reported
  1810→    but not transmitted to the exchange.
  1811→    """
  1812→    from core.clob_client import clob_client
  1813→    from risk.manager import MAX_POSITION_PER_MARKET
  1814→
  1815→    size_usdc = min(float(req.size_usdc), float(MAX_POSITION_PER_MARKET))
  1816→    if size_usdc <= 0:
  1817→        raise HTTPException(status_code=400, detail="size_usdc must be positive")
  1818→
  1819→    results = []
  1820→    legs = [
  1821→        ("yes", req.token_id_yes),
  1822→        ("no", req.token_id_no),
  1823→    ]
  1824→    for leg, token_id in legs:
  1825→        book = store.order_books.get(token_id)
  1826→        price = book.best_ask if book and book.best_ask else 0.50
  1827→        shares = size_usdc / max(price, 0.01)
  1828→
  1829→        provisional = Order(
  1830→            order_id=f"arb-{leg}-pre",
  1831→            token_id=token_id,
  1832→            side=Side.BUY,
  1833→            price=price,
  1834→            size=shares,
  1835→            strategy="arb_scanner",
  1836→            paper=settings.paper_trade,
  1837→        )
  1838→        allowed, reason = await risk_manager.check_order(provisional)
  1839→        if not allowed:
  1840→            results.append({"leg": leg, "token_id": token_id, "status": "REJECTED", "reason": reason})
  1841→            continue
  1842→
  1843→        if settings.paper_trade:
  1844→            order = await paper_sim.create_order(
  1845→                OrderArgs(token_id=token_id, price=price, side=Side.BUY, size=shares),
  1846→                strategy="arb_scanner",
  1847→            )
  1848→            status = "PLACED_PAPER"
  1849→        else:
  1850→            if token_id.endswith("_no"):
  1851→                results.append({"leg": leg, "token_id": token_id, "status": "SKIPPED", "reason": "synthetic complementary token — not transmissible"})
  1852→                continue
  1853→            resp = await clob_client.create_order(
  1854→                OrderArgs(token_id=token_id, price=price, side=Side.BUY, size=shares)
  1855→            )
  1856→            if resp is None:
  1857→                results.append({"leg": leg, "token_id": token_id, "status": "FAILED", "reason": "exchange rejected order"})
  1858→                continue
  1859→            order = Order(
  1860→                order_id=resp.get("orderID") or resp.get("order_id", "unknown"),
  1861→                token_id=token_id,
  1862→                side=Side.BUY,
  1863→                price=price,
  1864→                size=shares,
  1865→                strategy="arb_scanner",
  1866→                paper=False,
  1867→            )
  1868→            await store.add_order(order)
  1869→            status = "PLACED_LIVE"
  1870→
  1871→        results.append({"leg": leg, "token_id": token_id, "status": status, "order_id": order.order_id})
  1872→
  1873→    slug = store.market_slugs.get(req.token_id_yes, req.token_id_yes[:12])
  1874→    await store.log_event(f"⚡ Manual arb execute: {slug} (${size_usdc:.2f}/leg): {[r['status'] for r in results]}")
  1875→    return {"status": "processed", "size_usdc": size_usdc, "legs": results, "slug": slug}
